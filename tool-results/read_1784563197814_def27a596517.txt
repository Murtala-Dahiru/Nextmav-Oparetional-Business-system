    36→    36→function formatDateTime(d: string) {
    37→    37→  return new Date(d).toLocaleString('en-US', {
    38→    38→    month: 'short', day: 'numeric', hour: 'numeric', minute: '2-digit',
    39→    39→  });
    40→    40→}
    41→    41→
    42→    42→function timeAgo(d: string) {
    43→    43→  const diff = Date.now() - new Date(d).getTime();
    44→    44→  const mins = Math.floor(diff / 60000);
    45→    45→  if (mins < 60) return `${mins}m ago`;
    46→    46→  const hrs = Math.floor(mins / 60);
    47→    47→  if (hrs < 24) return `${hrs}h ago`;
    48→    48→  const days = Math.floor(hrs / 24);
    49→    49→  return `${days}d ago`;
    50→    50→}
