   134→async function fetchList<T>(url: string): Promise<{ data: T[]; meta: ApiMeta }> {
   135→  const res = await fetch(url);
   136→  if (!res.ok) {
   137→    const e = await res.json().catch(() => ({ message: 'Request failed' }));
   138→    throw new Error((e as any).message || `Error ${res.status}`);
   139→  }
   140→  return res.json();
   141→}
   142→
   143→async function createRecord<T>(url: string, data: unknown): Promise<T> {
   144→  const res = await fetch(url, {
   145→    method: 'POST',
   146→    headers: { 'Content-Type': 'application/json' },
   147→    body: JSON.stringify(data),
   148→  });
   149→  if (!res.ok) {
   150→    const e = await res.json().catch(() => ({ message: 'Create failed' }));
   151→    throw new Error((e as any).message || 'Create failed');
   152→  }
   153→  const json = await res.json();
   154→  return json.data;
   155→}
   156→
   157→async function updateRecord<T>(url: string, data: unknown): Promise<T> {
   158→  const res = await fetch(url, {
   159→    method: 'PUT',
   160→    headers: { 'Content-Type': 'application/json' },
   161→    body: JSON.stringify(data),
   162→  });
   163→  if (!res.ok) {
   164→    const e = await res.json().catch(() => ({ message: 'Update failed' }));
   165→    throw new Error((e as any).message || 'Update failed');
   166→  }
   167→  const json = await res.json();
   168→  return json.data;
   169→}

   210→function OverviewTab({ onSwitchTab }: { onSwitchTab: (tab: string) => void }) {

   214→
   215→  useEffect(() => {
   216→    (async () => {
   217→      try {
   218→        const [invRes, expRes] = await Promise.all([
   219→          fetchList<InvoiceRecord>('/api/finance/invoices?pageSize=1000'),
   220→          fetchList<ExpenseRecord>('/api/finance/expenses?pageSize=1000'),
   221→        ]);
   222→        setInvoices(invRes.data);
   223→        setExpenses(expRes.data);
   224→      } catch (e: any) {
   225→        toast.error(e.message || 'Failed to load finance data');
   226→      } finally {
   227→        setLoading(false);
   228→      }
   229→    })();
   230→  }, []);
   231→
   232→  // Computed stats
   233→  const totalRevenue = useMemo(() => invoices.filter((i) => i.status === 'paid').reduce((s, i) => s + i.total, 0), [invoices]);
   234→  const outstanding = useMemo(() => invoices.filter((i) => i.status === 'sent' || i.status === 'overdue').reduce((s, i) => s + i.total, 0), [invoices]);
   235→  const totalExpenses = useMemo(() => expenses.reduce((s, e) => s + e.amount, 0), [expenses]);
   236→  const netIncome = totalRevenue - totalExpenses;
   237→
   238→  // Revenue by month (AreaChart data)
   239→  const revenueByMonth = useMemo(() => {
   240→    const monthMap: Record<string, number> = {};
   241→    invoices.forEach((inv) => {
   242→      if (inv.status !== 'paid') return;
   243→      const d = new Date(inv.createdAt);
   244→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   245→      monthMap[key] = (monthMap[key] || 0) + inv.total;
   246→    });
   247→    return Object.entries(monthMap)
   248→      .sort(([a], [b]) => a.localeCompare(b))
   249→      .map(([key, revenue]) => {
   250→        const [y, m] = key.split('-');
   251→        return { month: `${MONTH_NAMES[parseInt(m) - 1]} ${y}`, revenue };
   252→      });
   253→  }, [invoices]);
   254→
   255→  // Expense by category (PieChart data)
   256→  const expenseByCategory = useMemo(() => {
   257→    const catMap: Record<string, number> = {};
   258→    expenses.forEach((exp) => {
   259→      catMap[exp.category] = (catMap[exp.category] || 0) + exp.amount;
   260→    });
   261→    return Object.entries(catMap).map(([name, value]) => ({ name, value }));
   262→  }, [expenses]);
   263→
   264→  // Recent invoices (top 5 by date)
   265→  const recentInvoices = useMemo(
   266→    () => [...invoices].sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()).slice(0, 5),
   267→    [invoices],
   268→  );
   269→
   270→  if (loading) {
   271→    return (
   272→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   273→        {[0, 1, 2, 3].map((i) => (
   274→          <Card key={i}><CardContent className="p-4"><div className="h-20 animate-pulse bg-muted rounded" /></CardContent></Card>
   275→        ))}
   276→      </div>
   277→    );
   278→  }
   279→
   280→  return (
   281→    <div className="flex flex-col gap-6">
   282→      <PageHeader title="Finance" icon={DollarSign} />
   283→
   284→      {/* Stat Cards */}
   285→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   286→        <StatCard label="Total Revenue" value={formatCurrency(totalRevenue)} icon={TrendingUp} />
   287→        <StatCard label="Outstanding" value={formatCurrency(outstanding)} icon={Wallet} />
   288→        <StatCard label="Total Expenses" value={formatCurrency(totalExpenses)} icon={Receipt} />
   289→        <StatCard label="Net Income" value={formatCurrency(netIncome)} icon={netIncome >= 0 ? TrendingUp : TrendingDown} />
   290→      </div>
   291→
   292→      {/* Charts */}
   293→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   294→        {/* Revenue by month */}
   295→        <Card>
   296→          <CardHeader className="pb-2">
   297→            <CardTitle className="text-base font-semibold">Revenue by Month</CardTitle>
   298→          </CardHeader>
   299→          <CardContent>
   300→            {revenueByMonth.length === 0 ? (
   301→              <div className="flex items-center justify-center h-[250px] text-muted-foreground text-sm">
   302→                No paid invoices to display
   303→              </div>
   304→            ) : (
   305→              <ResponsiveContainer width="100%" height={250}>
   306→                <AreaChart data={revenueByMonth}>
   307→                  <defs>
   308→                    <linearGradient id="revenueGrad" x1="0" y1="0" x2="0" y2="1">
   309→                      <stop offset="5%" stopColor="#10b981" stopOpacity={0.3} />
   310→                      <stop offset="95%" stopColor="#10b981" stopOpacity={0} />
   311→                    </linearGradient>
   312→                  </defs>
   313→                  <CartesianGrid strokeDasharray="3 3" className="stroke-muted" />
   314→                  <XAxis dataKey="month" tick={{ fontSize: 12 }} className="text-muted-foreground" />
   315→                  <YAxis tick={{ fontSize: 12 }} className="text-muted-foreground" tickFormatter={(v) => `$${(v / 1000).toFixed(0)}k`} />
   316→                  <Tooltip formatter={(value: number) => [formatCurrency(value), 'Revenue']} />
   317→                  <Area type="monotone" dataKey="revenue" stroke="#10b981" fill="url(#revenueGrad)" strokeWidth={2} />
   318→                </AreaChart>
   319→              </ResponsiveContainer>
   320→            )}
   321→          </CardContent>
   322→        </Card>
   323→
   324→        {/* Expense breakdown */}
   325→        <Card>
   326→          <CardHeader className="pb-2">
   327→            <CardTitle className="text-base font-semibold">Expense Breakdown</CardTitle>
   328→          </CardHeader>
   329→          <CardContent>
   330→            {expenseByCategory.length === 0 ? (
   331→              <div className="flex items-center justify-center h-[250px] text-muted-foreground text-sm">
   332→                No expenses to display
   333→              </div>

   335→              <div className="flex items-center gap-6">

   342→                      innerRadius={50}

   375→          </Button>

   377→        <CardContent>
   378→          {recentInvoices.length === 0 ? (
   379→            <p className="text-muted-foreground text-sm py-8 text-center">No invoices yet</p>
