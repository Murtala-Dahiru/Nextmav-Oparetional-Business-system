   143→function progressColor(pct: number): string {
   144→  if (pct > 60) return '[&>div]:bg-emerald-500';

   153→async function apiFetch<T>(url: string, init?: RequestInit): Promise<{ data: T; meta?: ApiMeta }> {
   154→  const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...init });
   155→  const json = await res.json();
   156→  if (json.error) throw new Error(json.error.message || 'Request failed');
   157→  return json;
   158→}

   194→function TasksTab() {
   195→  // Data state
   196→  const [tasks, setTasks] = useState<Task[]>([]);
   197→  const [total, setTotal] = useState(0);
   198→  const [loading, setLoading] = useState(true);
   199→  const [users, setUsers] = useState<User[]>([]);
   200→  const [projects, setProjects] = useState<Project[]>([]);
   201→
   202→  // Table state
   203→  const [page, setPage] = useState(0);
   204→  const [pageSize, setPageSize] = useState(20);
   205→  const [search, setSearch] = useState('');
   206→  const [columnFilters, setColumnFilters] = useState<ColumnFiltersState>([]);
   207→  const [sorting, setSorting] = useState<SortingState>([]);
   208→
   209→  // Dialog state
   210→  const [taskDialogOpen, setTaskDialogOpen] = useState(false);
   211→  const [editingTask, setEditingTask] = useState<Task | null>(null);
   212→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   213→  const [deletingTask, setDeletingTask] = useState<Task | null>(null);
   214→  const [saving, setSaving] = useState(false);
   215→  const [deleting, setDeleting] = useState(false);
   216→
   217→  // Stats
   218→  const [stats, setStats] = useState({ total: 0, inProgress: 0, completed: 0 });
   219→
   220→  // Fetch tasks
   221→  const fetchTasks = useCallback(async () => {
   222→    setLoading(true);
   223→    try {
   224→      const params = new URLSearchParams();
   225→      params.set('page', String(page + 1));
   226→      params.set('pageSize', String(pageSize));
   227→      if (search) params.set('search', search);
   228→
   229→      // Apply column filters
   230→      const statusFilter = columnFilters.find((f) => f.id === 'status')?.value as string | undefined;
   231→      const priorityFilter = columnFilters.find((f) => f.id === 'priority')?.value as string | undefined;
   232→      const projectFilter = columnFilters.find((f) => f.id === 'projectId')?.value as string | undefined;
   233→      if (statusFilter) params.set('status', statusFilter);
   234→      if (priorityFilter) params.set('priority', priorityFilter);
   235→      if (projectFilter) params.set('projectId', projectFilter);
   236→
   237→      // Apply sorting
   238→      if (sorting.length > 0) {
   239→        params.set('sort', sorting[0].id);
   240→        params.set('sortDir', sorting[0].desc ? 'desc' : 'asc');
   241→      }
   242→
   243→      const res = await apiFetch<Task[]>(`/api/projects/tasks?${params.toString()}`);
   244→      setTasks(res.data || []);
   245→      setTotal(res.meta?.total || 0);
   246→    } catch (err: any) {
   247→      toast.error(err.message || 'Failed to load tasks');
   248→    } finally {
   249→      setLoading(false);
   250→    }
   251→  }, [page, pageSize, search, columnFilters, sorting]);
   252→
   253→  // Fetch all tasks for stats (no pagination)
   254→  const fetchStats = useCallback(async () => {
   255→    try {
   256→      const params = new URLSearchParams();
   257→      params.set('page', '1');
   258→      params.set('pageSize', '1');
   259→      const res = await apiFetch<Task[]>(`/api/projects/tasks?${params.toString()}`);
   260→      setStats((prev) => ({ ...prev, total: res.meta?.total || 0 }));
   261→
   262→      // In progress count
   263→      const res2 = await apiFetch<Task[]>('/api/projects/tasks?status=in-progress&page=1&pageSize=1');
   264→      setStats((prev) => ({ ...prev, inProgress: res2.meta?.total || 0 }));
   265→
   266→      // Completed count
   267→      const res3 = await apiFetch<Task[]>('/api/projects/tasks?status=done&page=1&pageSize=1');
   268→      setStats((prev) => ({ ...prev, completed: res3.meta?.total || 0 }));
   269→    } catch {
   270→      // silent
   271→    }
   272→  }, []);
   273→
   274→  // Fetch users and projects for dropdowns
   275→  const fetchDropdowns = useCallback(async () => {
   276→    try {
   277→      const [usersRes, projectsRes] = await Promise.all([
   278→        apiFetch<User[]>('/api/admin/users?pageSize=100'),
   279→        apiFetch<Project[]>('/api/projects/projects?page=1&pageSize=100'),
   280→      ]);
   281→      setUsers(usersRes.data || []);
   282→      setProjects(projectsRes.data || []);
   283→    } catch {
   284→      // silent
   285→    }
   286→  }, []);
   287→
   288→  useEffect(() => { fetchTasks(); }, [fetchTasks]);
   289→  useEffect(() => { fetchStats(); fetchDropdowns(); }, [fetchStats, fetchDropdowns]);
   290→
   291→  // Form
   292→  const { register, handleSubmit, control, reset, formState: { errors } } = useForm<TaskFormValues>({
   293→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   294→    resolver: zodResolver(createTaskSchema) as any,
   295→    defaultValues: {
   296→      title: '', description: '', status: 'todo', priority: 'medium',
   297→      assigneeId: 'u1', projectId: '', dueDate: null as any, estimatedHours: 0, loggedHours: 0,
   298→    },
   299→  });
   300→
   301→  // Open create dialog
   302→  const openCreate = useCallback(() => {
   303→    setEditingTask(null);
   304→    reset({
   305→      title: '', description: '', status: 'todo', priority: 'medium',
   306→      assigneeId: 'u1', projectId: projects[0]?.id || '', dueDate: null as any,
   307→      estimatedHours: 0, loggedHours: 0,
   308→    });
   309→    setTaskDialogOpen(true);
   310→  }, [reset, projects]);
   311→
   312→  // Open edit dialog
   313→  const openEdit = useCallback((task: Task) => {
   314→    setEditingTask(task);
   315→    reset({
   316→      title: task.title,
   317→      description: task.description,
   318→      status: task.status,
   319→      priority: task.priority,
   320→      assigneeId: task.assigneeId,
   321→      projectId: task.projectId,
   322→      dueDate: task.dueDate ? task.dueDate.slice(0, 10) : null as any,
   323→      estimatedHours: task.estimatedHours,
   324→      loggedHours: task.loggedHours,
   325→    });
   326→    setTaskDialogOpen(true);
   327→  }, [reset]);
   328→
   329→  // Submit
   330→  const onSubmit = useCallback(async (values: TaskFormValues) => {
   331→    setSaving(true);
   332→    try {
   333→      const payload = {
   334→        ...values,
   335→        dueDate: values.dueDate || null,
   336→        loggedHours: values.loggedHours ?? 0,
   337→        sortOrder: values.sortOrder ?? 0,
   338→      };
   339→      if (editingTask) {
   340→        await apiFetch(`/api/projects/tasks/${editingTask.id}`, { method: 'PUT', body: JSON.stringify(payload) });
   341→        toast.success('Task updated');
   342→      } else {
   343→        await apiFetch('/api/projects/tasks', { method: 'POST', body: JSON.stringify(payload) });
   344→        toast.success('Task created');
   345→      }
   346→      setTaskDialogOpen(false);
   347→      fetchTasks();
   348→      fetchStats();
   349→    } catch (err: any) {
   350→      toast.error(err.message || 'Save failed');
   351→    } finally {
   352→      setSaving(false);
   353→    }
   354→  }, [editingTask, fetchTasks, fetchStats]);
   355→
   356→  // Delete
   357→  const confirmDelete = useCallback(() => {
   358→    if (!deletingTask) return;
   359→    setDeleting(true);
   360→    apiFetch(`/api/projects/tasks/${deletingTask.id}`, { method: 'DELETE' })
   361→      .then(() => {
   362→        toast.success('Task deleted');
   363→        setDeleteDialogOpen(false);
   364→        setDeletingTask(null);
   365→        fetchTasks();
   366→        fetchStats();
   367→      })
   368→      .catch((err: any) => toast.error(err.message || 'Delete failed'))
   369→      .finally(() => setDeleting(false));
   370→  }, [deletingTask, fetchTasks, fetchStats]);
   371→
   372→  // Table columns
   373→  const columns = useMemo<ColumnDef<Task, any>[]>(() => [
   374→    {
   375→      accessorKey: 'title',
   376→      header: 'Title',
   377→      cell: ({ row }) => (
   378→        <span className="font-medium text-foreground">{row.original.title}</span>
   379→      ),
   380→    },
   381→    {
   382→      accessorKey: 'project.name',
   383→      id: 'project',
   384→      header: 'Project',
   385→      cell: ({ row }) => row.original.project ? (
   386→        <Badge variant="outline" className="font-normal text-xs">{row.original.project.name}</Badge>
   387→      ) : <span className="text-muted-foreground text-sm">—</span>,
   388→    },
   389→    {
   390→      accessorKey: 'assignee',
   391→      id: 'assignee',
   392→      header: 'Assignee',
   393→      cell: ({ row }) => {
   394→        const a = row.original.assignee;
   395→        return a ? (
   396→          <div className="flex items-center gap-2">
   397→            <Avatar className="size-6">
   398→              <AvatarFallback className="text-[10px] bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300">
   399→                {getInitials(a.firstName, a.lastName)}
   400→              </AvatarFallback>
   401→            </Avatar>
   402→            <span className="text-sm">{a.firstName} {a.lastName}</span>
   403→          </div>
   404→        ) : <span className="text-muted-foreground text-sm">—</span>;
   405→      },
   406→    },
   407→    {
   408→      accessorKey: 'status',
   409→      header: 'Status',
   410→      cell: ({ row }) => {
   411→        const s = row.original.status;
   412→        return (
   413→          <Badge variant="secondary" className={TASK_STATUS_COLORS[s] || ''}>
   414→            {TASK_STATUS_LABELS[s] || s}
   415→          </Badge>
   416→        );
   417→      },
   418→    },
   419→    {
   420→      accessorKey: 'priority',
   421→      header: 'Priority',
   422→      cell: ({ row }) => {
   423→        const p = row.original.priority;
   424→        return (
   425→          <Badge variant="secondary" className={PRIORITY_COLORS[p] || ''}>
   426→            {PRIORITY_LABELS[p] || p}
   427→          </Badge>
   428→        );
   429→      },
   430→    },
   431→    {
   432→      accessorKey: 'dueDate',
   433→      header: 'Due Date',
   434→      cell: ({ row }) => row.original.dueDate
   435→        ? <span className="text-sm">{formatDate(row.original.dueDate)}</span>
   436→        : <span className="text-muted-foreground text-sm">—</span>,
   437→    },
   438→    {
   439→      id: 'hours',
   440→      header: 'Hours',
   441→      cell: ({ row }) => (
   442→        <span className="text-sm tabular-nums">
   443→          {row.original.loggedHours}/{row.original.estimatedHours}h
   444→        </span>
   445→      ),
   446→    },
   447→    {
   448→      id: 'actions',
   449→      header: '',
   450→      size: 50,
   451→      cell: ({ row }) => (
   452→        <DropdownMenu>
   453→          <DropdownMenuTrigger asChild>
   454→            <Button variant="ghost" size="icon" className="size-8">
   455→              <MoreHorizontal className="size-4" />
   456→            </Button>
   457→          </DropdownMenuTrigger>
   458→          <DropdownMenuContent align="end">
   459→            <DropdownMenuItem onClick={() => openEdit(row.original)}>
   460→              <Pencil className="size-4 mr-2" /> Edit
   461→            </DropdownMenuItem>
   462→            <DropdownMenuItem
   463→              className="text-destructive focus:text-destructive"
   464→              onClick={() => { setDeletingTask(row.original); setDeleteDialogOpen(true); }}
   465→            >
   466→              <Trash2 className="size-4 mr-2" /> Delete
   467→            </DropdownMenuItem>
   468→          </DropdownMenuContent>
   469→        </DropdownMenu>
   470→      ),
   471→    },
   472→  ], [openEdit]);
   473→
   474→  // Filters for DataTable
   475→  const filters = useMemo<DataTableFilter[]>(() => [
   476→    {
   477→      key: 'status',
   478→      label: 'Status',
   479→      options: TASK_STATUSES.map((s) => ({ value: s, label: TASK_STATUS_LABELS[s] || s })),
   480→    },
   481→    {
   482→      key: 'priority',
   483→      label: 'Priority',
   484→      options: ['low', 'medium', 'high', 'critical'].map((p) => ({ value: p, label: PRIORITY_LABELS[p] || p })),
   485→    },
   486→    {
   487→      key: 'projectId',
   488→      label: 'Project',
   489→      options: projects.map((p) => ({ value: p.id, label: p.name })),
   490→    },
   491→  ], [projects]);
   492→
   493→  return (
   494→    <div className="flex flex-col gap-6">
   495→      {/* Header */}
   496→      <PageHeader title="Tasks" icon={ListTodo}>
   497→        <Button onClick={openCreate} className="bg-emerald-600 text-white hover:bg-emerald-700 gap-1.5">
   498→          <Plus className="size-4" /> New Task
   499→        </Button>
   500→      </PageHeader>
   501→
   502→      {/* Stat Cards */}
   503→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   504→        <StatCard label="Total Tasks" value={stats.total} icon={ListTodo} />
   505→        <StatCard label="In Progress" value={stats.inProgress} icon={Clock} />
   506→        <StatCard label="Completed" value={stats.completed} icon={CheckCircle2} />
   507→      </div>
   508→
   509→      {/* Data Table */}
   510→      <DataTable
   511→        columns={columns}
   512→        data={tasks}
   513→        searchKey="title"
   514→        searchPlaceholder="Search tasks..."
   515→        filters={filters}
   516→        isLoading={loading}
   517→        emptyMessage="No tasks found. Create your first task!"
   518→        total={total}
   519→        page={page}
   520→        pageSize={pageSize}
   521→        onPageChange={setPage}
   522→        onPageSizeChange={(s) => { setPageSize(s); setPage(0); }}
   523→        onSearchChange={(v) => { setSearch(v); setPage(0); }}
   524→        onFilterChange={(f) => { setColumnFilters(f); setPage(0); }}
   525→        onSortChange={(s) => { setSorting(s); setPage(0); }}
   526→      />
   527→
   528→      {/* Create / Edit Dialog */}
   529→      <Dialog open={taskDialogOpen} onOpenChange={setTaskDialogOpen}>
   530→        <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">
   531→          <DialogHeader>
   532→            <DialogTitle>{editingTask ? 'Edit Task' : 'New Task'}</DialogTitle>
