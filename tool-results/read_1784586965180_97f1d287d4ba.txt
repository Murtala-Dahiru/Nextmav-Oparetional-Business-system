   125→async function fetchList<T>(url: string): Promise<{ data: T[]; meta: ApiMeta }> {
   126→  const res = await fetch(url);
   127→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Request failed' })); throw new Error((e as any).message || `Error ${res.status}`); }
   128→  return res.json();
   129→}
   130→
   131→async function createRecord<T>(url: string, data: unknown): Promise<T> {
   132→  const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
   133→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Create failed' })); throw new Error((e as any).message || 'Create failed'); }
   134→  const json = await res.json();
   135→  return json.data;
   136→}
   137→
   138→async function updateRecord<T>(url: string, data: unknown): Promise<T> {
   139→  const res = await fetch(url, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
   140→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Update failed' })); throw new Error((e as any).message || 'Update failed'); }
   141→  const json = await res.json();
   142→  return json.data;
   143→}
   144→
   145→async function deleteRecord(url: string): Promise<void> {
   146→  const res = await fetch(url, { method: 'DELETE' });
   147→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Delete failed' })); throw new Error((e as any).message || 'Delete failed'); }
   148→}

   169→  );

   198→function LeadsTab() {
   199→  const [leads, setLeads] = useState<Lead[]>([]);
   200→  const [loading, setLoading] = useState(true);
   201→  const [meta, setMeta] = useState<ApiMeta>({ total: 0, page: 1, pageSize: 20, totalPages: 0 });
   202→  const [search, setSearch] = useState('');
   203→  const [page, setPage] = useState(0);
   204→  const [pageSize, setPageSize] = useState(20);
   205→  const [sort, setSort] = useState('createdAt');
   206→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc');
   207→  const [statusFilter, setStatusFilter] = useState('');
   208→
   209→  const [allLeads, setAllLeads] = useState<Lead[]>([]);
   210→  const [statsLoading, setStatsLoading] = useState(true);
   211→
   212→  const [open, setOpen] = useState(false);
   213→  const [editing, setEditing] = useState<Lead | null>(null);
   214→  const [submitting, setSubmitting] = useState(false);
   215→  const [deleteOpen, setDeleteOpen] = useState(false);
   216→  const [deletingItem, setDeletingItem] = useState<Lead | null>(null);
   217→  const [deleting, setDeleting] = useState(false);
   218→
   219→  const form = useForm<LeadFormValues>({ resolver: zodResolver(createLeadSchema) as any, defaultValues: LEAD_DEFAULTS });
   220→
   221→  const fetchLeads = useCallback(async () => {
   222→    setLoading(true);
   223→    try {
   224→      const p = new URLSearchParams({ page: String(page + 1), pageSize: String(pageSize), sort, sortDir });
   225→      if (search) p.set('search', search);
   226→      if (statusFilter) p.set('status', statusFilter);
   227→      const res = await fetchList<Lead>(`/api/crm/leads?${p}`);
   228→      setLeads(res.data); setMeta(res.meta);
   229→    } catch (err: any) { toast.error(err.message || 'Failed to load leads'); }
   230→    finally { setLoading(false); }
   231→  }, [page, pageSize, search, sort, sortDir, statusFilter]);
   232→
   233→  const fetchStats = useCallback(async () => {
   234→    setStatsLoading(true);
   235→    try {
   236→      const res = await fetchList<Lead>('/api/crm/leads?pageSize=100&sort=createdAt&sortDir=desc');
   237→      setAllLeads(res.data);
   238→    } catch { /* silent */ }
   239→    finally { setStatsLoading(false); }
   240→  }, []);
   241→
   242→  useEffect(() => { fetchLeads(); }, [fetchLeads]);
   243→  useEffect(() => { fetchStats(); }, [fetchStats]);
   244→
   245→  const stats = useMemo(() => ({
   246→    total: allLeads.length,
   247→    new: allLeads.filter(l => l.status === 'new').length,
   248→    qualified: allLeads.filter(l => l.status === 'qualified').length,
   249→    won: allLeads.filter(l => l.status === 'won').length,
   250→  }), [allLeads]);
   251→
   252→  const statusFilterOption: DataTableFilter = useMemo(() => ({
   253→    key: 'status', label: 'Status',
   254→    options: LEAD_STATUSES.map(s => ({ value: s, label: s.charAt(0).toUpperCase() + s.slice(1) })),
   255→  }), []);
   256→
   257→  const columns: ColumnDef<Lead>[] = useMemo(() => [
   258→    { accessorKey: 'firstName', header: 'Name', cell: ({ row }) => {
   259→      const l = row.original;
   260→      return (
   261→        <div className="flex items-center gap-2">
   262→          <div className="bg-emerald-500/10 text-emerald-600 flex size-8 shrink-0 items-center justify-center rounded-full text-xs font-medium">
   263→            {getInitials(l.firstName, l.lastName)}
   264→          </div>
   265→          <span className="font-medium">{l.firstName} {l.lastName}</span>
   266→        </div>
   267→      );
   268→    }},
   269→    { accessorKey: 'email', header: 'Email', cell: ({ row }) => <span className="text-muted-foreground">{row.original.email}</span> },
   270→    { accessorKey: 'company', header: 'Company' },
   271→    { accessorKey: 'source', header: 'Source', cell: ({ row }) => <Badge variant="outline">{row.original.source}</Badge> },
   272→    { accessorKey: 'status', header: 'Status', cell: ({ row }) => <StatusBadge status={row.original.status} colorMap={LEAD_STATUS_COLORS} /> },
   273→    { accessorKey: 'score', header: 'Score', cell: ({ row }) => {
   274→      const s = row.original.score;
   275→      return (
   276→        <div className="flex items-center gap-2 min-w-[80px]">
   277→          <div className="flex-1 h-2 bg-muted rounded-full overflow-hidden">
   278→            <div className={`h-full rounded-full transition-all ${s > 60 ? 'bg-emerald-500' : s > 30 ? 'bg-amber-500' : 'bg-red-500'}`} style={{ width: `${s}%` }} />
   279→          </div>
   280→          <span className="text-xs text-muted-foreground w-6 text-right">{s}</span>
   281→        </div>
   282→      );
   283→    }},
   284→    { accessorKey: 'value', header: 'Value', cell: ({ row }) => <span className="font-medium">{formatCurrency(row.original.value)}</span> },
   285→    { accessorKey: 'createdAt', header: 'Created', cell: ({ row }) => <span className="text-muted-foreground text-sm">{formatRelativeTime(row.original.createdAt)}</span> },
   286→    { id: 'actions', size: 50, cell: ({ row }) => (
   287→      <DropdownMenu>
   288→        <DropdownMenuTrigger asChild><Button variant="ghost" size="icon" className="size-8"><MoreHorizontal className="size-4" /></Button></DropdownMenuTrigger>
   289→        <DropdownMenuContent align="end">
   290→          <DropdownMenuItem onClick={() => { setEditing(row.original); form.reset({ firstName: row.original.firstName, lastName: row.original.lastName, email: row.original.email, phone: row.original.phone, company: row.original.company, title: row.original.title, source: row.original.source, status: row.original.status, score: row.original.score, value: row.original.value, notes: row.original.notes, ownerId: row.original.ownerId }); setOpen(true); }}>
   291→            <Pencil className="size-4 mr-2" /> Edit
   292→          </DropdownMenuItem>
   293→          <DropdownMenuItem onClick={() => { setDeletingItem(row.original); setDeleteOpen(true); }} className="text-red-600 focus:text-red-600">
   294→            <Trash2 className="size-4 mr-2" /> Delete
   295→          </DropdownMenuItem>
   296→        </DropdownMenuContent>
   297→      </DropdownMenu>
   298→    )},
   299→  ], []);
   300→
   301→  const onSubmit = async (values: LeadFormValues) => {
   302→    setSubmitting(true);
   303→    try {
   304→      if (editing) { await updateRecord<Lead>(`/api/crm/leads/${editing.id}`, values); toast.success('Lead updated'); }
   305→      else { await createRecord<Lead>('/api/crm/leads', values); toast.success('Lead created'); }
   306→      setOpen(false); setEditing(null); fetchLeads(); fetchStats();
   307→    } catch (err: any) { toast.error(err.message || 'Operation failed'); }
   308→    finally { setSubmitting(false); }
   309→  };
   310→
   311→  const handleDelete = async () => {
   312→    if (!deletingItem) return;
   313→    setDeleting(true);
   314→    try { await deleteRecord(`/api/crm/leads/${deletingItem.id}`); toast.success('Lead deleted'); setDeleteOpen(false); setDeletingItem(null); fetchLeads(); fetchStats(); }
   315→    catch (err: any) { toast.error(err.message || 'Delete failed'); }
   316→    finally { setDeleting(false); }
   317→  };
   318→
   319→  return (
   320→    <div className="flex flex-col gap-4">
   321→      <PageHeader title="Leads" description="Track and manage your sales leads">
   322→        <Button onClick={() => { setEditing(null); form.reset(LEAD_DEFAULTS); setOpen(true); }} className="bg-emerald-600 hover:bg-emerald-700">
   323→          <Plus className="size-4 mr-2" /> New Lead
   324→        </Button>
   325→      </PageHeader>
   326→
   327→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   328→        <StatCard label="Total Leads" value={stats.total} icon={Users} />
   329→        <StatCard label="New" value={stats.new} icon={Sparkles} />
   330→        <StatCard label="Qualified" value={stats.qualified} icon={Target} />
   331→        <StatCard label="Won" value={stats.won} icon={TrendingUp} />
   332→      </div>
   333→
   334→      <DataTable
   335→        columns={columns} data={leads} isLoading={loading}
   336→        searchPlaceholder="Search leads..." filters={[statusFilterOption]}
   337→        onSearchChange={v => { setSearch(v); setPage(0); }}
   338→        onSortChange={s => { if (s.length) { setSort(s[0].id); setSortDir(s[0].desc ? 'desc' : 'asc'); } }}
   339→        onFilterChange={f => { setStatusFilter((f.find(x => x.id === 'status')?.value as string) || ''); setPage(0); }}
   340→        onPageChange={setPage} onPageSizeChange={s => { setPageSize(s); setPage(0); }}
   341→        page={page} pageSize={pageSize} total={meta.total}
   342→        emptyMessage="No leads found" emptyIcon={Users}
   343→      />
   344→
   345→      <Dialog open={open} onOpenChange={setOpen}>
   346→        <DialogContent className="sm:max-w-lg max-h-[85vh] overflow-y-auto">
   347→          <DialogHeader>
   348→            <DialogTitle>{editing ? 'Edit Lead' : 'New Lead'}</DialogTitle>
   349→            <DialogDescription>{editing ? 'Update lead information.' : 'Add a new lead to your pipeline.'}</DialogDescription>
   350→          </DialogHeader>
   351→          <form onSubmit={form.handleSubmit(onSubmit)} className="flex flex-col gap-4">
   352→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   353→              <Field label="First Name" error={form.formState.errors.firstName?.message}>
   354→                <Input {...form.register('firstName')} placeholder="John" />
   355→              </Field>
   356→              <Field label="Last Name" error={form.formState.errors.lastName?.message}>
   357→                <Input {...form.register('lastName')} placeholder="Doe" />
   358→              </Field>
   359→            </div>
   360→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   361→              <Field label="Email" error={form.formState.errors.email?.message}>
   362→                <Input type="email" {...form.register('email')} placeholder="john@example.com" />
   363→              </Field>
   364→              <Field label="Phone" error={form.formState.errors.phone?.message}>
   365→                <Input {...form.register('phone')} placeholder="+1 234 567 8900" />
   366→              </Field>
   367→            </div>
   368→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   369→              <Field label="Company" error={form.formState.errors.company?.message}>
   370→                <Input {...form.register('company')} placeholder="Acme Inc" />
   371→              </Field>
   372→              <Field label="Job Title" error={form.formState.errors.title?.message}>
   373→                <Input {...form.register('title')} placeholder="CEO" />
   374→              </Field>
   375→            </div>
   376→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   377→              <Field label="Source" error={form.formState.errors.source?.message}>
   378→                <Controller control={form.control} name="source" render={({ field }) => (
   379→                  <Select value={field.value} onValueChange={field.onChange}>
   380→                    <SelectTrigger><SelectValue placeholder="Select" /></SelectTrigger>
   381→                    <SelectContent>{SOURCE_OPTIONS.map(o => <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>)}</SelectContent>
   382→                  </Select>
   383→                )} />
   384→              </Field>
   385→              <Field label="Score (0-100)" error={form.formState.errors.score?.message}>
   386→                <Input type="number" min={0} max={100} {...form.register('score', { valueAsNumber: true })} />
   387→              </Field>
   388→              <Field label="Value ($)" error={form.formState.errors.value?.message}>
   389→                <Input type="number" min={0} {...form.register('value', { valueAsNumber: true })} />
   390→              </Field>
   391→            </div>
   392→            <Field label="Notes" error={form.formState.errors.notes?.message}>
   393→              <Textarea {...form.register('notes')} rows={3} placeholder="Additional notes..." />
   394→            </Field>
   395→            <DialogFooter>
   396→              <Button type="button" variant="outline" onClick={() => setOpen(false)}>Cancel</Button>
   397→              <Button type="submit" disabled={submitting} className="bg-emerald-600 hover:bg-emerald-700">
   398→                {submitting && <Loader2 className="size-4 mr-2 animate-spin" />}
   399→                {editing ? 'Update' : 'Create'}
   400→              </Button>
   401→            </DialogFooter>
   402→          </form>
   403→        </DialogContent>
   404→      </Dialog>
   405→
   406→      <ConfirmDialog open={deleteOpen} onOpenChange={setDeleteOpen} title="Delete Lead"

   409→    </div>

   417→function ContactsTab() {

   424→  const [sort, setSort] = useState('createdAt');
   425→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc');
   426→
   427→  const [open, setOpen] = useState(false);
   428→  const [editing, setEditing] = useState<Contact | null>(null);
   429→  const [submitting, setSubmitting] = useState(false);
   430→  const [deleteOpen, setDeleteOpen] = useState(false);

   434→  const form = useForm<ContactFormValues>({ resolver: zodResolver(createContactSchema) as any, defaultValues: CONTACT_DEFAULTS });
   435→
   436→  const fetchData = useCallback(async () => {

   439→      const p = new URLSearchParams({ page: String(page + 1), pageSize: String(pageSize), sort, sortDir });
   440→      if (search) p.set('search', search);
   441→      const res = await fetchList<Contact>(`/api/crm/contacts?${p}`);
   442→      setContacts(res.data); setMeta(res.meta);
   443→    } catch (err: any) { toast.error(err.message || 'Failed to load contacts'); }
   444→    finally { setLoading(false); }
   445→  }, [page, pageSize, search, sort, sortDir]);
   446→
   447→  useEffect(() => { fetchData(); }, [fetchData]);
   448→
   449→  const columns: ColumnDef<Contact>[] = useMemo(() => [

   453→        <div className="flex items-center gap-2">

   461→    { accessorKey: 'email', header: 'Email', cell: ({ row }) => <span className="text-muted-foreground">{row.original.email}</span> },

   465→    { accessorKey: 'source', header: 'Source', cell: ({ row }) => <Badge variant="outline">{row.original.source}</Badge> },

   467→      <Badge variant="secondary" className={row.original.isActive ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300' : 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-300'}>
   468→        {row.original.isActive ? 'Active' : 'Inactive'}
   469→      </Badge>
   470→    )},
   471→    { accessorKey: 'createdAt', header: 'Created', cell: ({ row }) => <span className="text-muted-foreground text-sm">{formatRelativeTime(row.original.createdAt)}</span> },
   472→    { id: 'actions', size: 50, cell: ({ row }) => (
   473→      <DropdownMenu>
   474→        <DropdownMenuTrigger asChild><Button variant="ghost" size="icon" className="size-8"><MoreHorizontal className="size-4" /></Button></DropdownMenuTrigger>
   475→        <DropdownMenuContent align="end">

   477→            <Pencil className="size-4 mr-2" /> Edit

   479→          <DropdownMenuItem onClick={() => { setDeletingItem(row.original); setDeleteOpen(true); }} className="text-red-600 focus:text-red-600">
   480→            <Trash2 className="size-4 mr-2" /> Delete

   492→      setOpen(false); setEditing(null); fetchData();
   493→    } catch (err: any) { toast.error(err.message || 'Operation failed'); }
   494→    finally { setSubmitting(false); }
   495→  };

   497→  const handleDelete = async () => {

   500→    try { await deleteRecord(`/api/crm/contacts/${deletingItem.id}`); toast.success('Contact deleted'); setDeleteOpen(false); setDeletingItem(null); fetchData(); }

   520→        emptyMessage="No contacts found" emptyIcon={Users}
   521→      />
   522→
   523→      <Dialog open={open} onOpenChange={setOpen}>

   528→          </DialogHeader>

   530→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">

   539→              <Field label="Email" error={form.formState.errors.email?.message}>
   540→                <Input type="email" {...form.register('email')} placeholder="john@example.com" />
   541→              </Field>

   543→                <Input {...form.register('phone')} placeholder="+1 234 567 8900" />

   578→                {submitting && <Loader2 className="size-4 mr-2 animate-spin" />}
