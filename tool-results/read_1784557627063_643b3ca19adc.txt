    42→function formatLastSeen(d: string) {
    43→  const date = new Date(d);
    44→  const now = new Date('2026-07-20T11:00:00Z');
    45→  const diffMs = now.getTime() - date.getTime();
    46→  const diffMins = Math.floor(diffMs / 60000);
    47→  if (diffMins < 60) return `${diffMins}m ago`;
    48→  const diffHrs = Math.floor(diffMins / 60);
    49→  if (diffHrs < 24) return `${diffHrs}h ago`;
    50→  const diffDays = Math.floor(diffHrs / 24);
    51→  return `${diffDays}d ago`;
    52→}
