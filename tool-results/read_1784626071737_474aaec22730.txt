    93→    93→async function apiFetch<T>(url: string, init?: RequestInit): Promise<{ data: T; meta?: { total: number; page: number; pageSize: number; totalPages: number } }> {
    94→    94→  const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...init });
    95→    95→  const json = await res.json();
    96→    96→  if (json.error) throw new Error(json.error.message || 'Request failed');
    97→    97→  return json;
    98→    98→}

   103→   103→
   104→   104→export default function WorkspaceModule() {
   105→   105→  // Data state
   106→   106→  const [pages, setPages] = useState<WorkspacePage[]>([]);
   107→   107→  const [loading, setLoading] = useState(true);
   108→   108→
   109→   109→  // Selection state
   110→   110→  const [selectedPageId, setSelectedPageId] = useState<string | null>(null);
   111→   111→  const [sidebarSearch, setSidebarSearch] = useState('');
   112→   112→  const [collapsedFolders, setCollapsedFolders] = useState<Set<string>>(new Set());
   113→   113→
   114→   114→  // Full page content
   115→   115→  const [fullPage, setFullPage] = useState<WorkspacePage | null>(null);
   116→   116→  const [pageLoading, setPageLoading] = useState(false);
   117→   117→  const [editing, setEditing] = useState(false);
   118→   118→  const [editContent, setEditContent] = useState('');
   119→   119→  const [savingContent, setSavingContent] = useState(false);
   120→   120→
   121→   121→  // Inline title editing
   122→   122→  const [isEditingTitle, setIsEditingTitle] = useState(false);
   123→   123→  const [editTitle, setEditTitle] = useState('');
   124→   124→
   125→   125→  // Dialog state
   126→   126→  const [createDialogOpen, setCreateDialogOpen] = useState(false);
   127→   127→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   128→   128→  const [deletingPage, setDeletingPage] = useState<WorkspacePage | null>(null);
   129→   129→  const [saving, setSaving] = useState(false);
   130→   130→  const [deleting, setDeleting] = useState(false);
   131→   131→
   132→   132→  // Fetch all pages for sidebar (large page size)
   133→   133→  const fetchPages = useCallback(async () => {
   134→   134→    setLoading(true);
   135→   135→    try {
   136→   136→      const res = await apiFetch<WorkspacePage[]>('/api/workspace/pages?page=1&pageSize=200');
   137→   137→      setPages(res.data || []);
   138→   138→    } catch (err: any) {
   139→   139→      toast.error(err.message || 'Failed to load pages');
   140→   140→    } finally {
   141→   141→      setLoading(false);
   142→   142→    }
   143→   143→  }, []);
   144→   144→
   145→   145→  useEffect(() => { fetchPages(); }, [fetchPages]);
   146→   146→
   147→   147→  // Fetch full page
   148→   148→  const fetchFullPage = useCallback(async (id: string) => {
   149→   149→    setPageLoading(true);
   150→   150→    setSelectedPageId(id);
   151→   151→    setEditing(false);
   152→   152→    try {
   153→   153→      const res = await apiFetch<WorkspacePage>(`/api/workspace/pages/${id}`);
   154→   154→      setFullPage(res.data);
   155→   155→      setEditContent(res.data.content);
   156→   156→      setEditTitle(res.data.title);
   157→   157→    } catch (err: any) {
   158→   158→      toast.error(err.message || 'Failed to load page');
   159→   159→    } finally {
   160→   160→      setPageLoading(false);
   161→   161→    }
   162→   162→  }, []);
   163→   163→
   164→   164→  // Derived: filtered pages
   165→   165→  const filteredPages = useMemo(() => {
   166→   166→    if (!sidebarSearch) return pages;
   167→   167→    const q = sidebarSearch.toLowerCase();
   168→   168→    return pages.filter((p) => p.title.toLowerCase().includes(q));
   169→   169→  }, [pages, sidebarSearch]);
   170→   170→
   171→   171→  // Derived: folders and non-folder pages
   172→   172→  const rootPages = useMemo(() => {
   173→   173→    const folders = filteredPages.filter((p) => p.isFolder && !p.parentId);
   174→   174→    const items = filteredPages.filter((p) => !p.isFolder && !p.parentId);
   175→   175→    return { folders, items };
   176→   176→  }, [filteredPages]);
   177→   177→
   178→   178→  // Get children of a folder
   179→   179→  const getFolderChildren = useCallback((folderId: string) => {
   180→   180→    return filteredPages.filter((p) => p.parentId === folderId && !p.isFolder);
   181→   181→  }, [filteredPages]);
   182→   182→
   183→   183→  // Toggle folder collapse
   184→   184→  const toggleFolder = useCallback((folderId: string) => {
   185→   185→    setCollapsedFolders((prev) => {
   186→   186→      const next = new Set(prev);
   187→   187→      if (next.has(folderId)) next.delete(folderId);
   188→   188→      else next.add(folderId);
   189→   189→      return next;
   190→   190→    });
   191→   191→  }, []);
   192→   192→
   193→   193→  // Save page content
   194→   194→  const saveContent = useCallback(async () => {
   195→   195→    if (!fullPage) return;
   196→   196→    setSavingContent(true);
   197→   197→    try {
   198→   198→      const res = await apiFetch<WorkspacePage>(`/api/workspace/pages/${fullPage.id}`, {
   199→   199→        method: 'PUT',
   200→   200→        body: JSON.stringify({ content: editContent }),
   201→   201→      });
   202→   202→      setFullPage(res.data);
   203→   203→      setEditing(false);
   204→   204→      fetchPages();
   205→   205→      toast.success('Page saved');
   206→   206→    } catch (err: any) {
   207→   207→      toast.error(err.message || 'Save failed');
   208→   208→    } finally {
   209→   209→      setSavingContent(false);
   210→   210→    }
   211→   211→  }, [fullPage, editContent, fetchPages]);
   212→   212→
   213→   213→  // Save page title (inline)
   214→   214→  const saveTitle = useCallback(async () => {
   215→   215→    if (!fullPage) return;
   216→   216→    try {
   217→   217→      const res = await apiFetch<WorkspacePage>(`/api/workspace/pages/${fullPage.id}`, {
   218→   218→        method: 'PUT',
   219→   219→        body: JSON.stringify({ title: editTitle }),
   220→   220→      });
   221→   221→      setFullPage(res.data);
   222→   222→      setIsEditingTitle(false);
   223→   223→      fetchPages();
   224→   224→    } catch (err: any) {
   225→   225→      toast.error(err.message || 'Failed to update title');
   226→   226→    }
   227→   227→  }, [fullPage, editTitle, fetchPages]);
   228→   228→
   229→   229→  // Toggle star
   230→   230→  const toggleStar = useCallback(async (page: WorkspacePage) => {
   231→   231→    try {
   232→   232→      await apiFetch(`/api/workspace/pages/${page.id}`, {
   233→   233→        method: 'PUT',
   234→   234→        body: JSON.stringify({ isStarred: !page.isStarred }),
   235→   235→      });
   236→   236→      fetchPages();
   237→   237→      if (selectedPageId === page.id) {
   238→   238→        setFullPage((prev) => prev ? { ...prev, isStarred: !prev.isStarred } : null);
   239→   239→      }
   240→   240→    } catch (err: any) {
   241→   241→      toast.error(err.message || 'Failed to toggle star');
   242→   242→    }
   243→   243→  }, [fetchPages, selectedPageId]);
   244→   244→
   245→   245→  // Delete
   246→   246→  const confirmDelete = useCallback(() => {
   247→   247→    if (!deletingPage) return;
   248→   248→    setDeleting(true);
   249→   249→    apiFetch(`/api/workspace/pages/${deletingPage.id}`, { method: 'DELETE' })
   250→   250→      .then(() => {
   251→   251→        toast.success('Page deleted');
   252→   252→        setDeleteDialogOpen(false);
   253→   253→        setDeletingPage(null);
   254→   254→        if (selectedPageId === deletingPage.id) {
   255→   255→          setSelectedPageId(null);
   256→   256→          setFullPage(null);
   257→   257→        }
   258→   258→        fetchPages();
   259→   259→      })
   260→   260→      .catch((err: any) => toast.error(err.message || 'Delete failed'))
   261→   261→      .finally(() => setDeleting(false));
   262→   262→  }, [deletingPage, fetchPages, selectedPageId]);
   263→   263→
   264→   264→  // Sort: starred first, then by updatedAt desc
   265→   265→  const sortedItems = useMemo(() => {
   266→   266→    return [...rootPages.items].sort((a, b) => {
   267→   267→      if (a.isStarred && !b.isStarred) return -1;
   268→   268→      if (!a.isStarred && b.isStarred) return 1;
   269→   269→      return new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime();
   270→   270→    });
   271→   271→  }, [rootPages.items]);
   272→   272→
   273→   273→  const sortedFolders = useMemo(() => {
   274→   274→    return [...rootPages.folders].sort((a, b) => {
   275→   275→      if (a.isStarred && !b.isStarred) return -1;
   276→   276→      if (!a.isStarred && b.isStarred) return 1;
   277→   277→      return new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime();
   278→   278→    });
   279→   279→  }, [rootPages.folders]);
   280→   280→
   281→   281→  // Form for create dialog
   282→   282→  const { register, handleSubmit, control, reset, watch, formState: { errors } } = useForm<PageFormValues>({
   283→   283→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   284→   284→    resolver: zodResolver(createPageSchema) as any,
   285→   285→    defaultValues: {
   286→   286→      title: '', content: '', icon: 'file-text', color: '#10b981',
   287→   287→      parentId: '', isFolder: false, isStarred: false, lastEditedBy: '',
   288→   288→    },
   289→   289→  });
   290→   290→
   291→   291→  const watchIsFolder = watch('isFolder');
   292→   292→
   293→   293→  // Open create dialog
   294→   294→  const openCreate = useCallback(() => {
   295→   295→    reset({
   296→   296→      title: '', content: '# New Page\n\nStart writing here...', icon: 'file-text', color: '#10b981',
   297→   297→      parentId: '', isFolder: false, isStarred: false, lastEditedBy: '',
   298→   298→    });
   299→   299→    setCreateDialogOpen(true);
   300→   300→  }, [reset]);
   301→   301→
   302→   302→  // Submit create
   303→   303→  const onSubmit = useCallback(async (values: PageFormValues) => {
   304→   304→    setSaving(true);
   305→   305→    try {
   306→   306→      const payload = {
   307→   307→        ...values,
   308→   308→        parentId: values.parentId || '',
   309→   309→      };
   310→   310→      const res = await apiFetch<WorkspacePage>('/api/workspace/pages', {
   311→   311→        method: 'POST',
   312→   312→        body: JSON.stringify(payload),
   313→   313→      });
   314→   314→      toast.success(values.isFolder ? 'Folder created' : 'Page created');
   315→   315→      setCreateDialogOpen(false);
   316→   316→      fetchPages();
   317→   317→      // Auto-select the new page
   318→   318→      if (!values.isFolder) {
   319→   319→        fetchFullPage(res.data.id);
   320→   320→      }
   321→   321→    } catch (err: any) {
   322→   322→      toast.error(err.message || 'Create failed');
   323→   323→    } finally {
   324→   324→      setSaving(false);
   325→   325→    }
   326→   326→  }, [fetchPages, fetchFullPage]);
   327→   327→
   328→   328→  // Page icon component
   329→   329→  const PageIcon = ({ icon, color, className }: { icon: string; color: string; className?: string }) => {
   330→   330→    const IconComp = ICON_MAP[icon] || FileText;
   331→   331→    return <IconComp className={className} style={{ color }} />;
   332→   332→  };
   333→   333→
   334→   334→  // Render sidebar page item
   335→   335→  const renderPageItem = (page: WorkspacePage, indent = 0) => (
   336→   336→    <button
   337→   337→      key={page.id}
   338→   338→      onClick={() => fetchFullPage(page.id)}
   339→   339→      className={`w-full flex items-center gap-2 px-3 py-1.5 text-sm rounded-md transition-colors text-left hover:bg-accent ${
   340→   340→        selectedPageId === page.id ? 'bg-accent text-accent-foreground' : 'text-foreground'
   341→   341→      }`}
   342→   342→      style={{ paddingLeft: `${12 + indent * 20}px` }}
   343→   343→    >
   344→   344→      <PageIcon icon={page.icon} color={page.color} className="size-4 shrink-0" />
   345→   345→      <span className="truncate flex-1">{page.title}</span>
   346→   346→      {page.isStarred && <Star className="size-3 text-amber-500 fill-amber-500 shrink-0" />}
   347→   347→      <span className="text-xs text-muted-foreground shrink-0 hidden sm:inline">
   348→   348→        {formatRelativeTime(page.updatedAt)}
   349→   349→      </span>
   350→   350→    </button>
   351→   351→  );
   352→   352→
   353→   353→  // Render folder group
   354→   354→  const renderFolder = (folder: WorkspacePage) => {
   355→   355→    const children = getFolderChildren(folder.id);
   356→   356→    const isCollapsed = collapsedFolders.has(folder.id);
   357→   357→
   358→   358→    return (
   359→   359→      <div key={folder.id}>
   360→   360→        <button
   361→   361→          onClick={() => toggleFolder(folder.id)}
   362→   362→          className="w-full flex items-center gap-2 px-3 py-1.5 text-sm rounded-md transition-colors text-left hover:bg-accent text-foreground"
   363→   363→        >
   364→   364→          {isCollapsed ? (
   365→   365→            <ChevronRight className="size-3.5 text-muted-foreground shrink-0" />
   366→   366→          ) : (
   367→   367→            <ChevronDown className="size-3.5 text-muted-foreground shrink-0" />
   368→   368→          )}
   369→   369→          <PageIcon icon={folder.icon} color={folder.color} className="size-4 shrink-0" />
   370→   370→          <span className="truncate flex-1 font-medium">{folder.title}</span>
   371→   371→          {folder.isStarred && <Star className="size-3 text-amber-500 fill-amber-500 shrink-0" />}
   372→   372→          <span className="text-xs text-muted-foreground shrink-0">{children.length}</span>
   373→   373→        </button>
   374→   374→        {!isCollapsed && (
   375→   375→          <div className="mt-0.5 space-y-0.5">
   376→   376→            {children.map((child) => renderPageItem(child, 1))}
   377→   377→            {children.length === 0 && (
   378→   378→              <p className="text-xs text-muted-foreground pl-12 py-1">Empty folder</p>
   379→   379→            )}
   380→   380→          </div>
   381→   381→        )}
   382→   382→      </div>
   383→   383→    );
   384→   384→  };
   385→   385→
   386→   386→  // Current selected page icon for header
   387→   387→  const currentIcon = fullPage ? (ICON_MAP[fullPage.icon] || FileText) : FileText;
   388→   388→
   389→   389→  return (
   390→   390→    <TooltipProvider>
   391→   391→      <div className="flex flex-1 overflow-hidden h-full">
   392→   392→        {/* ─── Left Sidebar ─── */}
   393→   393→        <aside className="w-64 border-r bg-card flex flex-col shrink-0 hidden sm:flex">
   394→   394→          {/* New Page Button */}
   395→   395→          <div className="p-3">
   396→   396→            <Button
   397→   397→              onClick={openCreate}
   398→   398→              variant="outline"
   399→   399→              className="w-full justify-start gap-2 text-emerald-600 border-emerald-200 hover:bg-emerald-50 dark:border-emerald-800 dark:hover:bg-emerald-950/30"
   400→   400→            >
   401→   401→              <Plus className="size-4" /> New Page
   402→   402→            </Button>
   403→   403→          </div>
   404→   404→
   405→   405→          {/* Search */}
   406→   406→          <div className="px-3 pb-2">
   407→   407→            <div className="relative">
   408→   408→              <Search className="text-muted-foreground absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5" />
   409→   409→              <Input
   410→   410→                placeholder="Search pages..."
   411→   411→                value={sidebarSearch}
   412→   412→                onChange={(e) => setSidebarSearch(e.target.value)}
   413→   413→                className="pl-8 h-8 text-sm"
   414→   414→              />
   415→   415→            </div>
   416→   416→          </div>
   417→   417→
   418→   418→          <Separator />
   419→   419→
   420→   420→          {/* Page List */}
   421→   421→          <ScrollArea className="flex-1 px-2 py-2">
   422→   422→            {loading ? (
   423→   423→              <div className="space-y-2 px-1">
   424→   424→                {Array.from({ length: 8 }).map((_, i) => (
   425→   425→                  <Skeleton key={i} className="h-8 w-full" />
   426→   426→                ))}
   427→   427→              </div>
   428→   428→            ) : filteredPages.length === 0 ? (
   429→   429→              <p className="text-xs text-muted-foreground text-center py-8">
   430→   430→                {sidebarSearch ? 'No pages match your search.' : 'No pages yet.'}
   431→   431→              </p>
   432→   432→            ) : (
   433→   433→              <div className="space-y-0.5">
   434→   434→                {/* Folders */}
   435→   435→                {sortedFolders.length > 0 && (
   436→   436→                  <div className="space-y-0.5 mb-2">
   437→   437→                    {sortedFolders.map(renderFolder)}
   438→   438→                  </div>
   439→   439→                )}
   440→   440→
   441→   441→                {/* Divider if both exist */}
   442→   442→                {sortedFolders.length > 0 && sortedItems.length > 0 && (
   443→   443→                  <Separator className="my-2" />
   444→   444→                )}
