   189→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ name: string; value: number; color: string }>; label?: string }) {
   190→  if (!active || !payload?.length) return null;
   191→  return (
   192→    <div className="rounded-lg border border-gray-200 bg-white/95 px-3 py-2 shadow-lg dark:border-gray-700 dark:bg-gray-900/95">
   193→      <p className="mb-1 text-sm font-semibold text-gray-900 dark:text-gray-100">{label}</p>
   194→      {payload.map((entry, i) => (
   195→        <p key={i} className="text-xs text-gray-600 dark:text-gray-400">
   196→          <span className="mr-1.5 inline-block h-2 w-2 rounded-full" style={{ backgroundColor: entry.color }} />
   197→          {entry.name}: <span className="font-medium text-gray-900 dark:text-gray-100">${(entry.value / 1000).toFixed(1)}K</span>
   198→        </p>
   199→      ))}
   200→    </div>
   201→  );
   202→}

   284→function LeadConversionFunnel() {
   285→  const maxCount = funnelStages[0].count;
   286→  return (
   287→    <Card className="border-0 shadow-sm">
   288→      <CardHeader className="pb-2">
   289→        <CardTitle className="flex items-center gap-2 text-base font-semibold text-gray-900 dark:text-gray-100">
   290→          <Filter className="h-4 w-4 text-emerald-600" />
   291→          Lead Conversion Funnel
   292→        </CardTitle>
   293→        <CardDescription>Visual representation of your pipeline from lead to close</CardDescription>
   294→      </CardHeader>
   295→      <CardContent className="pt-2">
   296→        <div className="relative flex flex-col items-center gap-0">
   297→          {funnelStages.map((stage, idx) => {
   298→            const widthPercent = 30 + ((stage.count / maxCount) * 70);
   299→            const conversionFromPrev = idx > 0 ? ((stage.count / funnelStages[idx - 1].count) * 100).toFixed(1) : '100';
   300→            const isLast = idx === funnelStages.length - 1;
   301→            return (
   302→              <div key={stage.label} className="relative flex w-full flex-col items-center">
   303→                {/* Connector arrow */}
   304→                {idx > 0 && (
   305→                  <motion.div
   306→                    initial={{ scaleY: 0 }}
   307→                    animate={{ scaleY: 1 }}
   308→                    transition={{ duration: 0.3, delay: idx * 0.1 }}
   309→                    className="flex items-center gap-2 py-1"
   310→                  >
   311→                    <div className="h-4 w-px bg-gradient-to-b from-gray-300 to-gray-400 dark:from-gray-600 dark:to-gray-500" />
   312→                    <ChevronRight className="h-3 w-3 -rotate-90 text-gray-400" />
   313→                    <span className="text-[10px] font-bold text-emerald-600 dark:text-emerald-400">
   314→                      {conversionFromPrev}%
   315→                    </span>
   316→                  </motion.div>
   317→                )}
   318→                {/* Funnel bar */}
   319→                <motion.div
   320→                  initial={{ scaleX: 0, opacity: 0 }}
   321→                  animate={{ scaleX: 1, opacity: 1 }}
   322→                  transition={{ duration: 0.5, delay: idx * 0.1, ease: 'easeOut' }}
   323→                  style={{ width: `${widthPercent}%` }}
   324→                  className="relative"
   325→                >
   326→                  <div
   327→                    className={cn(
   328→                      'group relative flex items-center justify-between rounded-sm px-5 py-3 transition-all duration-300',
   329→                      isLast
   330→                        ? 'bg-gradient-to-r from-emerald-500 to-teal-400 shadow-lg shadow-emerald-500/25'
   331→                        : 'hover:brightness-110',
   332→                    )}
   333→                    style={
   334→                      isLast
   335→                        ? undefined
   336→                        : {
   337→                            background: `linear-gradient(135deg, ${stage.color}, ${stage.color}dd)`,
   338→                            boxShadow: `0 2px 8px ${stage.color}30`,
   339→                          }
   340→                    }
   341→                  >
   342→                    {/* Animated shimmer effect */}
   343→                    <div className="absolute inset-0 overflow-hidden rounded-sm">
   344→                      <div className="absolute inset-0 -translate-x-full bg-gradient-to-r from-transparent via-white/10 to-transparent animate-[shimmer_3s_infinite]" />
   345→                    </div>
   346→                    {/* Stage label and count */}
   347→                    <div className="relative z-10 flex items-center gap-3">
   348→                      <div className={cn(
   349→                        'flex h-7 w-7 items-center justify-center rounded-full text-xs font-bold',
   350→                        isLast ? 'bg-white/25 text-white' : 'bg-white/15 text-white/90',
   351→                      )}>
   352→                        {idx + 1}
   353→                      </div>
   354→                      <div>
   355→                        <span className={cn(
   356→                          'block text-sm font-semibold tracking-wide',
   357→                          isLast ? 'text-white' : 'text-white/95',
   358→                        )}>
   359→                          {stage.label}
   360→                        </span>
   361→                      </div>
   362→                    </div>
   363→                    <div className="relative z-10 text-right">
   364→                      <span className={cn(
   365→                        'block text-xl font-bold tabular-nums',
   366→                        isLast ? 'text-white' : 'text-white/95',

   369→                      </span>
   370→                    </div>
   371→                    {/* Drop percentage */}
   372→                    {idx > 0 && (
   373→                      <div className="absolute -right-1 top-1/2 -translate-y-1/2 translate-x-full px-2 py-0.5">
   374→                        <Badge variant="secondary" className="text-[10px] font-bold text-emerald-700 bg-emerald-50 dark:text-emerald-400 dark:bg-emerald-950/50 border-0">
   375→                          ↓ {((1 - stage.count / funnelStages[idx - 1].count) * 100).toFixed(1)}%
   376→                        </Badge>
   377→                      </div>
   378→                    )}
   379→                  </div>
   380→                </motion.div>
   381→              </div>
   382→            );
   383→          })}
   384→          {/* Overall conversion rate callout */}
   385→          <motion.div
   386→            initial={{ opacity: 0, scale: 0.9 }}
   387→            animate={{ opacity: 1, scale: 1 }}
   388→            transition={{ duration: 0.5, delay: 0.8 }}
   389→            className="mt-4 rounded-xl border border-emerald-200 bg-gradient-to-r from-emerald-50 to-teal-50 px-6 py-3 text-center dark:border-emerald-800 dark:from-emerald-950/40 dark:to-teal-950/40"
