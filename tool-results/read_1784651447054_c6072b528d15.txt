   124→async function fetchList<T>(url: string): Promise<{ data: T[]; meta: ApiMeta }> {
   125→  const res = await fetch(url);
   126→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Request failed' })); throw new Error((e as any).message || `Error ${res.status}`); }
   127→  return res.json();
   128→}
   129→
   130→async function createRecord<T>(url: string, data: unknown): Promise<T> {
   131→  const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
   132→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Create failed' })); throw new Error((e as any).message || 'Create failed'); }
   133→  const json = await res.json();
   134→  return json.data;
   135→}
   136→
   137→async function updateRecord<T>(url: string, data: unknown): Promise<T> {
   138→  const res = await fetch(url, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
   139→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Update failed' })); throw new Error((e as any).message || 'Update failed'); }
   140→  const json = await res.json();
   141→  return json.data;
   142→}
   143→
   144→async function deleteRecord(url: string): Promise<void> {
   145→  const res = await fetch(url, { method: 'DELETE' });
   146→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Delete failed' })); throw new Error((e as any).message || 'Delete failed'); }
   147→}

   168→  );

   197→function LeadsTab() {
   198→  const [leads, setLeads] = useState<Lead[]>([]);
   199→  const [loading, setLoading] = useState(true);
   200→  const [meta, setMeta] = useState<ApiMeta>({ total: 0, page: 1, pageSize: 20, totalPages: 0 });
   201→  const [search, setSearch] = useState('');
   202→  const [page, setPage] = useState(0);
   203→  const [pageSize, setPageSize] = useState(20);
   204→  const [sort, setSort] = useState('createdAt');
   205→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc');
   206→  const [statusFilter, setStatusFilter] = useState('');
   207→
   208→  const [allLeads, setAllLeads] = useState<Lead[]>([]);
   209→  const [statsLoading, setStatsLoading] = useState(true);
   210→
   211→  const [open, setOpen] = useState(false);
   212→  const [editing, setEditing] = useState<Lead | null>(null);
   213→  const [submitting, setSubmitting] = useState(false);
   214→  const [deleteOpen, setDeleteOpen] = useState(false);
   215→  const [deletingItem, setDeletingItem] = useState<Lead | null>(null);
   216→  const [deleting, setDeleting] = useState(false);
   217→
   218→  const form = useForm<LeadFormValues>({ resolver: zodResolver(createLeadSchema) as any, defaultValues: LEAD_DEFAULTS });
   219→
   220→  const fetchLeads = useCallback(async () => {
   221→    setLoading(true);
   222→    try {
   223→      const p = new URLSearchParams({ page: String(page + 1), pageSize: String(pageSize), sort, sortDir });
   224→      if (search) p.set('search', search);
   225→      if (statusFilter) p.set('status', statusFilter);
   226→      const res = await fetchList<Lead>(`/api/crm/leads?${p}`);
   227→      setLeads(res.data); setMeta(res.meta);
   228→    } catch (err: any) { toast.error(err.message || 'Failed to load leads'); }
   229→    finally { setLoading(false); }
   230→  }, [page, pageSize, search, sort, sortDir, statusFilter]);
   231→
   232→  const fetchStats = useCallback(async () => {
   233→    setStatsLoading(true);
   234→    try {
   235→      const res = await fetchList<Lead>('/api/crm/leads?pageSize=100&sort=createdAt&sortDir=desc');
   236→      setAllLeads(res.data);
   237→    } catch { /* non-critical */ }
   238→    finally { setStatsLoading(false); }
   239→  }, []);
   240→
   241→  useEffect(() => { fetchLeads(); }, [fetchLeads]);
   242→  useEffect(() => { fetchStats(); }, [fetchStats]);
   243→
   244→  const stats = useMemo(() => ({
   245→    total: allLeads.length,
   246→    new: allLeads.filter(l => l.status === 'new').length,
   247→    qualified: allLeads.filter(l => l.status === 'qualified').length,
   248→    won: allLeads.filter(l => l.status === 'won').length,
   249→  }), [allLeads]);
   250→
   251→  const statusFilterOption: DataTableFilter = useMemo(() => ({
   252→    key: 'status', label: 'Status',
   253→    options: LEAD_STATUSES.map(s => ({ value: s, label: s.charAt(0).toUpperCase() + s.slice(1) })),
   254→  }), []);
   255→
   256→  const columns: ColumnDef<Lead>[] = useMemo(() => [
   257→    { accessorKey: 'firstName', header: 'Name', cell: ({ row }) => {
   258→      const l = row.original;
   259→      return (
   260→        <div className="flex items-center gap-2">
   261→          <div className="bg-emerald-500/10 text-emerald-600 flex size-8 shrink-0 items-center justify-center rounded-full text-xs font-medium">
   262→            {getInitials(l.firstName, l.lastName)}
   263→          </div>
   264→          <span className="font-medium">{l.firstName} {l.lastName}</span>
   265→        </div>
   266→      );
   267→    }},
   268→    { accessorKey: 'email', header: 'Email', cell: ({ row }) => <span className="text-muted-foreground">{row.original.email}</span> },
   269→    { accessorKey: 'company', header: 'Company' },
   270→    { accessorKey: 'source', header: 'Source', cell: ({ row }) => <Badge variant="outline">{row.original.source}</Badge> },
   271→    { accessorKey: 'status', header: 'Status', cell: ({ row }) => <StatusBadge status={row.original.status} colorMap={LEAD_STATUS_COLORS} /> },
   272→    { accessorKey: 'score', header: 'Score', cell: ({ row }) => {
   273→      const s = row.original.score;
   274→      return (
   275→        <div className="flex items-center gap-2 min-w-[80px]">
   276→          <div className="flex-1 h-2 bg-muted rounded-full overflow-hidden">
   277→            <div className={`h-full rounded-full transition-all ${s > 60 ? 'bg-emerald-500' : s > 30 ? 'bg-amber-500' : 'bg-red-500'}`} style={{ width: `${s}%` }} />
   278→          </div>
   279→          <span className="text-xs text-muted-foreground w-6 text-right">{s}</span>
   280→        </div>
   281→      );
   282→    }},
   283→    { accessorKey: 'value', header: 'Value', cell: ({ row }) => <span className="font-medium">{formatCurrency(row.original.value)}</span> },
   284→    { accessorKey: 'createdAt', header: 'Created', cell: ({ row }) => <span className="text-muted-foreground text-sm">{formatRelativeTime(row.original.createdAt)}</span> },
   285→    { id: 'actions', size: 50, cell: ({ row }) => (
   286→      <DropdownMenu>
   287→        <DropdownMenuTrigger asChild><Button variant="ghost" size="icon" className="size-8" aria-label="Actions"><MoreHorizontal className="size-4" /></Button></DropdownMenuTrigger>
   288→        <DropdownMenuContent align="end">
   289→          <DropdownMenuItem onClick={() => { setEditing(row.original); form.reset({ firstName: row.original.firstName, lastName: row.original.lastName, email: row.original.email, phone: row.original.phone, company: row.original.company, title: row.original.title, source: row.original.source, status: row.original.status, score: row.original.score, value: row.original.value, notes: row.original.notes, ownerId: row.original.ownerId }); setOpen(true); }}>
   290→            <Pencil className="size-4 mr-2" /> Edit
   291→          </DropdownMenuItem>
   292→          <DropdownMenuItem onClick={() => { setDeletingItem(row.original); setDeleteOpen(true); }} className="text-red-600 focus:text-red-600">
   293→            <Trash2 className="size-4 mr-2" /> Delete
   294→          </DropdownMenuItem>
   295→        </DropdownMenuContent>
   296→      </DropdownMenu>
   297→    )},
   298→  ], []);
   299→
   300→  const onSubmit = async (values: LeadFormValues) => {
   301→    setSubmitting(true);
   302→    try {
   303→      if (editing) { await updateRecord<Lead>(`/api/crm/leads/${editing.id}`, values); toast.success('Lead updated'); }
   304→      else { await createRecord<Lead>('/api/crm/leads', values); toast.success('Lead created'); }
   305→      setOpen(false); setEditing(null); fetchLeads(); fetchStats();
   306→    } catch (err: any) { toast.error(err.message || 'Operation failed'); }
   307→    finally { setSubmitting(false); }
   308→  };
   309→
   310→  const handleDelete = async () => {
   311→    if (!deletingItem) return;
   312→    setDeleting(true);
   313→    try { await deleteRecord(`/api/crm/leads/${deletingItem.id}`); toast.success('Lead deleted'); setDeleteOpen(false); setDeletingItem(null); fetchLeads(); fetchStats(); }
   314→    catch (err: any) { toast.error(err.message || 'Delete failed'); }
   315→    finally { setDeleting(false); }
   316→  };
   317→
   318→  return (
   319→    <div className="flex flex-col gap-4">
   320→      <PageHeader title="Leads" description="Track and manage your sales leads">
   321→        <Button onClick={() => { setEditing(null); form.reset(LEAD_DEFAULTS); setOpen(true); }} className="bg-emerald-600 hover:bg-emerald-700">
   322→          <Plus className="size-4 mr-2" /> New Lead
   323→        </Button>
   324→      </PageHeader>
   325→
   326→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   327→        <StatCard label="Total Leads" value={stats.total} icon={Users} />
   328→        <StatCard label="New" value={stats.new} icon={Sparkles} />
   329→        <StatCard label="Qualified" value={stats.qualified} icon={Target} />
   330→        <StatCard label="Won" value={stats.won} icon={TrendingUp} />
   331→      </div>
   332→
   333→      <DataTable
   334→        columns={columns} data={leads} isLoading={loading}
   335→        searchPlaceholder="Search leads..." filters={[statusFilterOption]}
   336→        onSearchChange={v => { setSearch(v); setPage(0); }}
   337→        onSortChange={s => { if (s.length) { setSort(s[0].id); setSortDir(s[0].desc ? 'desc' : 'asc'); } }}
   338→        onFilterChange={f => { setStatusFilter((f.find(x => x.id === 'status')?.value as string) || ''); setPage(0); }}
   339→        onPageChange={setPage} onPageSizeChange={s => { setPageSize(s); setPage(0); }}
   340→        page={page} pageSize={pageSize} total={meta.total}
   341→        emptyMessage="No leads found" emptyIcon={Users}
   342→      />
   343→
   344→      <Dialog open={open} onOpenChange={setOpen}>
   345→        <DialogContent className="sm:max-w-lg max-h-[85vh] overflow-y-auto">
   346→          <DialogHeader>
   347→            <DialogTitle>{editing ? 'Edit Lead' : 'New Lead'}</DialogTitle>
   348→            <DialogDescription>{editing ? 'Update lead information.' : 'Add a new lead to your pipeline.'}</DialogDescription>
   349→          </DialogHeader>
   350→          <form onSubmit={form.handleSubmit(onSubmit)} className="flex flex-col gap-4">
   351→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   352→              <Field label="First Name" error={form.formState.errors.firstName?.message}>
   353→                <Input {...form.register('firstName')} placeholder="John" />
   354→              </Field>
   355→              <Field label="Last Name" error={form.formState.errors.lastName?.message}>
   356→                <Input {...form.register('lastName')} placeholder="Doe" />
   357→              </Field>
   358→            </div>
   359→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   360→              <Field label="Email" error={form.formState.errors.email?.message}>
   361→                <Input type="email" {...form.register('email')} placeholder="john@example.com" />
   362→              </Field>
   363→              <Field label="Phone" error={form.formState.errors.phone?.message}>
   364→                <Input {...form.register('phone')} placeholder="+1 234 567 8900" />
   365→              </Field>
   366→            </div>
   367→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   368→              <Field label="Company" error={form.formState.errors.company?.message}>
   369→                <Input {...form.register('company')} placeholder="Acme Inc" />
   370→              </Field>
   371→              <Field label="Job Title" error={form.formState.errors.title?.message}>
   372→                <Input {...form.register('title')} placeholder="CEO" />
   373→              </Field>
   374→            </div>
   375→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   376→              <Field label="Source" error={form.formState.errors.source?.message}>
   377→                <Controller control={form.control} name="source" render={({ field }) => (
   378→                  <Select value={field.value} onValueChange={field.onChange}>
   379→                    <SelectTrigger><SelectValue placeholder="Select" /></SelectTrigger>
   380→                    <SelectContent>{SOURCE_OPTIONS.map(o => <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>)}</SelectContent>
   381→                  </Select>
   382→                )} />
   383→              </Field>
   384→              <Field label="Score (0-100)" error={form.formState.errors.score?.message}>
   385→                <Input type="number" min={0} max={100} {...form.register('score', { valueAsNumber: true })} />
   386→              </Field>
   387→              <Field label="Value ($)" error={form.formState.errors.value?.message}>
   388→                <Input type="number" min={0} {...form.register('value', { valueAsNumber: true })} />
   389→              </Field>
   390→            </div>
   391→            <Field label="Notes" error={form.formState.errors.notes?.message}>
   392→              <Textarea {...form.register('notes')} rows={3} placeholder="Additional notes..." />
   393→            </Field>
   394→            <DialogFooter>
   395→              <Button type="button" variant="outline" onClick={() => setOpen(false)}>Cancel</Button>
   396→              <Button type="submit" disabled={submitting} className="bg-emerald-600 hover:bg-emerald-700">
   397→                {submitting && <Loader2 className="size-4 mr-2 animate-spin" />}
   398→                {editing ? 'Update' : 'Create'}
   399→              </Button>
   400→            </DialogFooter>
   401→          </form>
   402→        </DialogContent>
   403→      </Dialog>
   404→
   405→      <ConfirmDialog open={deleteOpen} onOpenChange={setDeleteOpen} title="Delete Lead"
   406→        description={`Are you sure you want to delete ${deletingItem?.firstName} ${deletingItem?.lastName}? This action cannot be undone.`}
   407→        confirmLabel="Delete" variant="destructive" onConfirm={handleDelete} isLoading={deleting} />
   408→    </div>
   409→  );
   410→}
