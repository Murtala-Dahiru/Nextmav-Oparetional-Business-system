   238→function ListViewTab({ tasks }: { tasks: ProjectTaskItem[] }) {
   239→  const [search, setSearch] = useState('');
   240→  const [sortKey, setSortKey] = useState<'priority' | 'dueDate' | 'title'>('priority');
   241→  const priorityOrder: Record<string, number> = { critical: 0, high: 1, medium: 2, low: 3 };
   242→
   243→  const filtered = useMemo(() => {
   244→    let result = tasks;
   245→    if (search) {
   246→      const q = search.toLowerCase();
   247→      result = result.filter(
   248→        (t) =>
   249→          t.title.toLowerCase().includes(q) ||
   250→          t.projectName.toLowerCase().includes(q) ||
   251→          t.assigneeName.toLowerCase().includes(q)
   252→      );
   253→    }
   254→    return [...result].sort((a, b) => {
   255→      if (sortKey === 'priority') return (priorityOrder[a.priority] ?? 9) - (priorityOrder[b.priority] ?? 9);

   258→    });

   421→/* ---- Tab 4: Timeline ---- */
   422→
   423→function TimelineTab({ items }: { items: ProjectItem[] }) {
   424→  const allDates = useMemo(() => {
   425→    const starts = items.map((p) => new Date(p.startDate).getTime());
   426→    const ends = items.map((p) => new Date(p.endDate).getTime());
   427→    return { min: Math.min(...starts), max: Math.max(...ends) };
   428→  }, [items]);
   429→
   430→  const totalSpan = allDates.max - allDates.min;
   431→
   432→  function getLeft(startDate: string) {
   433→    const ms = new Date(startDate).getTime() - allDates.min;
   434→    return totalSpan > 0 ? (ms / totalSpan) * 100 : 0;
   435→  }
   436→
   437→  function getWidth(startDate: string, endDate: string) {
   438→    const ms = new Date(endDate).getTime() - new Date(startDate).getTime();
   439→    return totalSpan > 0 ? Math.max((ms / totalSpan) * 100, 3) : 3;
   440→  }
   441→
   442→  const months = useMemo(() => {
   443→    const start = new Date(allDates.min);
   444→    const end = new Date(allDates.max);
   445→    const result: string[] = [];
   446→    const current = new Date(start.getFullYear(), start.getMonth(), 1);
   447→    while (current <= end) {
   448→      result.push(current.toLocaleDateString('en-US', { month: 'short', year: '2-digit' }));
   449→      current.setMonth(current.getMonth() + 1);
   450→    }
   451→    return result;
   452→  }, [allDates]);
   453→
   454→  return (
   455→    <div>
   456→      <TabHeader title="Timeline" description="Visualize project schedules and progress on a Gantt-like timeline" actionLabel="Add Project" />
   457→      <div className="bg-white rounded-xl border border-gray-200 overflow-hidden">
   458→        {/* Month header */}
   459→        <div className="flex border-b border-gray-100">
   460→          <div className="w-52 shrink-0 px-4 py-3 border-r border-gray-100">
   461→            <span className="text-xs font-semibold text-gray-500 uppercase tracking-wider">Project</span>
   462→          </div>
   463→          <div className="flex-1 flex relative">
   464→            {months.map((m, i) => {
   465→              const monthStart = new Date(allDates.min);
   466→              monthStart.setMonth(monthStart.getMonth() + i);
   467→              const monthStartMs = monthStart.getTime() - allDates.min;
   468→              const nextMonth = new Date(monthStart);
   469→              nextMonth.setMonth(nextMonth.getMonth() + 1);
   470→              const nextMonthMs = Math.min(nextMonth.getTime() - allDates.min, totalSpan);
   471→              const widthPct = totalSpan > 0 ? ((nextMonthMs - monthStartMs) / totalSpan) * 100 : 0;
   472→              return (
   473→                <div
   474→                  key={m}
   475→                  className="flex items-center justify-center text-[11px] text-muted-foreground font-medium border-r border-gray-50 last:border-r-0"
   476→                  style={{ width: `${widthPct}%` }}
   477→                >
   478→                  {m}
   479→                </div>
   480→              );
   481→            })}
   482→          </div>
   483→        </div>
   484→        {/* Rows */}
   485→        {items.map((project, i) => (
   486→          <div
   487→            key={project.id}
   488→            className={cn('flex items-center border-b border-gray-50 last:border-b-0', i % 2 === 0 ? 'bg-white' : 'bg-gray-50/30')}
   489→          >
   490→            <div className="w-52 shrink-0 px-4 py-3.5 border-r border-gray-100">
   491→              <div className="flex items-center gap-2">
   492→                <ProjectStatusBadge status={project.status} />
   493→                <span className="text-sm font-medium text-gray-900 truncate">{project.name}</span>
   494→              </div>
   495→            </div>
   496→            <div className="flex-1 py-3.5 px-2 relative" style={{ minHeight: '44px' }}>
   497→              <div
   498→                className="relative h-7 rounded-md overflow-hidden"
   499→                style={{
   500→                  position: 'absolute',
   501→                  left: `${getLeft(project.startDate)}%`,
   502→                  width: `${getWidth(project.startDate, project.endDate)}%`,
   503→                  top: '50%',
   504→                  transform: 'translateY(-50%)',
   505→                }}
   506→              >
   507→                {/* Background bar */}
   508→                <div className={cn('absolute inset-0 rounded-md opacity-20', projectBarColors[project.status] ?? 'bg-gray-400')} />
   509→                {/* Progress fill */}
   510→                <div
   511→                  className={cn('absolute inset-y-0 left-0 rounded-md', projectBarColors[project.status] ?? 'bg-gray-400')}
   512→                  style={{ width: `${project.progress}%` }}
   513→                />
   514→                {/* Label */}
   515→                <span className="absolute inset-0 flex items-center px-2 text-[10px] font-semibold text-gray-700 truncate">
   516→                  {project.progress}% — {formatDate(project.startDate).replace(/,\s*\d{4}$/, '')} to {formatDate(project.endDate).replace(/,\s*\d{4}$/, '')}
   517→                </span>
   518→              </div>
   519→            </div>
   520→          </div>
   521→        ))}
   522→      </div>
   523→    </div>
   524→  );
   525→}
