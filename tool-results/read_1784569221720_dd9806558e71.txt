   171→   171→function getAvatarColor(name: string): string {
   172→   172→  let hash = 0;
   173→   173→  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);

   192→   192→    const q = searchQuery.toLowerCase();
   193→   193→    return folderTree
   194→   194→      .map((folder) => {
   195→   195→        const filteredChildren = folder.children.filter(
   196→   196→          (child) =>
   197→   197→            child.name.toLowerCase().includes(q) ||
   198→   198→            (child as DocumentItem).tags?.some((tag: string) => tag.toLowerCase().includes(q)),
   199→   199→        );

   201→   201→          return { ...folder, children: filteredChildren.length > 0 ? filteredChildren : folder.children };
   202→   202→        }
   203→   203→        return null;
   204→   204→      })

   212→   212→      else next.add(folderId);
   213→   213→      return next;
   214→   214→    });
