   105→   105→function formatDate(): string {
   106→   106→  return new Date().toLocaleDateString('en-US', {
   107→   107→    weekday: 'long',
   108→   108→    year: 'numeric',
   109→   109→    month: 'long',
   110→   110→    day: 'numeric',
   111→   111→  });
   112→   112→}
   113→   113→
   114→   114→function timeAgo(dateStr: string): string {
   115→   115→  const now = new Date();
   116→   116→  const date = new Date(dateStr);
   117→   117→  const diffMs = now.getTime() - date.getTime();
   118→   118→  const diffMin = Math.floor(diffMs / 60000);
   119→   119→  if (diffMin < 1) return 'Just now';
   120→   120→  if (diffMin < 60) return `${diffMin}m ago`;
   121→   121→  const diffHr = Math.floor(diffMin / 60);
   122→   122→  if (diffHr < 24) return `${diffHr}h ago`;
   123→   123→  const diffDay = Math.floor(diffHr / 24);
   124→   124→  return `${diffDay}d ago`;
   125→   125→}
   126→   126→
   127→   127→function formatTime(dateStr: string): string {
   128→   128→  const date = new Date(dateStr);
   129→   129→  return date.toLocaleTimeString('en-US', {
   130→   130→    hour: 'numeric',
   131→   131→    minute: '2-digit',
   132→   132→    hour12: true,
   133→   133→  });
   134→   134→}
   135→   135→
   136→   136→function formatShortDate(dateStr: string): string {

   139→   139→    month: 'short',
   140→   140→    day: 'numeric',

   142→   142→}
