   582→          const barColor = budgetBarColor(pct);

   584→            <motion.div key={b.department} {...fadeUp} transition={{ duration: 0.3, delay: 0.1 + i * 0.06 }}>

   620→                      initial={{ width: 0 }}
   621→                      animate={{ width: `${pct}%` }}
   622→                      transition={{ duration: 0.7, ease: 'easeOut', delay: 0.2 + i * 0.06 }}
   623→                    />
   624→                  </div>
   625→
   626→                  {pct > 90 && (
