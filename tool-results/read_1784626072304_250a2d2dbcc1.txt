   373→   373→      params.set('page', String(page + 1));

   379→   379→      }
   380→   380→      columnFilters.forEach((f) => {
   381→   381→        params.set(f.id, String(f.value));
   382→   382→      });
   383→   383→
   384→   384→      const res = await fetch(`/api/support/tickets?${params}`);
   385→   385→      const json = await res.json();
   386→   386→      if (json.error) { toast.error(json.error.message); return; }
   387→   387→

   389→   389→      setTotal(json.meta?.total ?? 0);

   400→   400→      });

   402→   402→      toast.error('Failed to load tickets');
   403→   403→    } finally {
   404→   404→      setTicketsLoading(false);
   405→   405→    }
   406→   406→  }, [page, pageSize, search, sorting, columnFilters]);

   422→   422→      contactEmail: '',
   423→   423→      assigneeId: null as string | null,

   433→   433→      const res = await fetch('/api/support/tickets', {
   434→   434→        method: 'POST',
   435→   435→        headers: { 'Content-Type': 'application/json' },
   436→   436→        body: JSON.stringify({ ...values, ticketNumber }),
   437→   437→      });
   438→   438→      const json = await res.json();
   439→   439→      if (json.error) { toast.error(json.error.message); return; }
   440→   440→      toast.success('Ticket created');
   441→   441→      setCreateOpen(false);
   442→   442→      createForm.reset();
   443→   443→      fetchTickets();
   444→   444→    } catch {
   445→   445→      toast.error('Failed to create ticket');
   446→   446→    } finally {
   447→   447→      setCreating(false);
   448→   448→    }
   449→   449→  }, [total, createForm, fetchTickets]);
   450→   450→
   451→   451→  // ─── Edit ticket ──────────────────────────────────────────────────
   452→   452→  const editForm = useForm({
   453→   453→    resolver: zodResolver(updateTicketSchema),
   454→   454→  });
   455→   455→
   456→   456→  useEffect(() => {
   457→   457→    if (editTicket) {
   458→   458→      editForm.reset({
   459→   459→        subject: editTicket.subject,
   460→   460→        description: editTicket.description,
   461→   461→        priority: editTicket.priority,
