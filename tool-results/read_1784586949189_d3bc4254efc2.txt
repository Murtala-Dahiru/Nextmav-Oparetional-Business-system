   131→async function fetchList<T>(url: string): Promise<{ data: T[]; meta: ApiMeta }> {
   132→  const res = await fetch(url);
   133→  if (!res.ok) {
   134→    const e = await res.json().catch(() => ({ message: 'Request failed' }));
   135→    throw new Error((e as any).message || `Error ${res.status}`);
   136→  }
   137→  return res.json();
   138→}
   139→
   140→async function createRecord<T>(url: string, data: unknown): Promise<T> {
   141→  const res = await fetch(url, {
   142→    method: 'POST',
   143→    headers: { 'Content-Type': 'application/json' },
   144→    body: JSON.stringify(data),
   145→  });
   146→  if (!res.ok) {
   147→    const e = await res.json().catch(() => ({ message: 'Create failed' }));
   148→    throw new Error((e as any).message || 'Create failed');
   149→  }
   150→  const json = await res.json();
   151→  return json.data;
   152→}
   153→
   154→async function updateRecord<T>(url: string, data: unknown): Promise<T> {
   155→  const res = await fetch(url, {
   156→    method: 'PUT',
   157→    headers: { 'Content-Type': 'application/json' },
   158→    body: JSON.stringify(data),
   159→  });
   160→  if (!res.ok) {
   161→    const e = await res.json().catch(() => ({ message: 'Update failed' }));
   162→    throw new Error((e as any).message || 'Update failed');
   163→  }
   164→  const json = await res.json();
   165→  return json.data;
   166→}
   167→
   168→async function deleteRecord(url: string): Promise<void> {
   169→  const res = await fetch(url, { method: 'DELETE' });
   170→  if (!res.ok) {
   171→    const e = await res.json().catch(() => ({ message: 'Delete failed' }));
   172→    throw new Error((e as any).message || 'Delete failed');
   173→  }
   174→}

   198→function computeDuration(start: string, end: string): number {
   199→  const s = new Date(start);
   200→  const e = new Date(end);
   201→  const diffMs = e.getTime() - s.getTime();
   202→  return Math.max(1, Math.ceil(diffMs / (1000 * 60 * 60 * 24)) + 1);
   203→}
   204→
   205→function isDateInRange(dateStr: string, start: string, end: string): boolean {
   206→  const d = new Date(dateStr);
   207→  const s = new Date(start);
   208→  const e = new Date(end);
   209→  return d >= s && d <= e;
   210→}

   619→interface LeaveFormData {

   621→  type: string;
   622→  startDate: string;
   623→  endDate: string;
   624→  reason: string;

   628→  requesterId: '', type: 'vacation', startDate: '', endDate: '', reason: '',
   629→};
   630→
   631→function LeaveTab() {
   632→  // Data state
   633→  const [leaves, setLeaves] = useState<LeaveRequest[]>([]);
   634→  const [leavesLoading, setLeavesLoading] = useState(false);
   635→
   636→  // All leaves (for stats)
   637→  const [allLeaves, setAllLeaves] = useState<LeaveRequest[]>([]);
   638→
   639→  // Users (for requester select)
   640→  const [usersList, setUsersList] = useState<UserRecord[]>([]);
   641→
   642→  // Dialogs
   643→  const [createOpen, setCreateOpen] = useState(false);
   644→  const [form, setForm] = useState<LeaveFormData>(LEAVE_DEFAULTS);
   645→  const [formLoading, setFormLoading] = useState(false);
   646→
   647→  // Stats
   648→  const [pendingCount, setPendingCount] = useState(0);
   649→  const [approvedThisMonth, setApprovedThisMonth] = useState(0);
   650→  const [rejectedCount, setRejectedCount] = useState(0);
   651→
   652→  // Fetch all leaves for stats
   653→  const fetchStats = useCallback(async () => {
   654→    try {
   655→      const res = await fetchList<LeaveRequest>('/api/hr/leave?pageSize=1000');
   656→      setAllLeaves(res.data);
   657→      const now = new Date();
   658→      const currentMonth = now.getMonth();
   659→      const currentYear = now.getFullYear();
   660→      setPendingCount(res.data.filter((l) => l.status === 'pending').length);
   661→      setApprovedThisMonth(
   662→        res.data.filter((l) => {
   663→          if (l.status !== 'approved') return false;
   664→          const d = new Date(l.startDate);
   665→          return d.getMonth() === currentMonth && d.getFullYear() === currentYear;
   666→        }).length,
   667→      );
   668→      setRejectedCount(res.data.filter((l) => l.status === 'rejected').length);
   669→    } catch {
   670→      // non-critical
   671→    }
   672→  }, []);
   673→
   674→  // Fetch leaves for table (client-side, use all data)
   675→  const fetchLeaves = useCallback(async () => {
   676→    setLeavesLoading(true);
   677→    try {
   678→      const res = await fetchList<LeaveRequest>('/api/hr/leave?pageSize=1000');
   679→      setLeaves(res.data);
   680→    } catch (e: any) {
   681→      toast.error(e.message || 'Failed to load leave requests');
   682→    } finally {
   683→      setLeavesLoading(false);
   684→    }

   691→      setUsersList(res.data);
   692→      if (res.data.length > 0 && !form.requesterId) {
   693→        setForm((p) => ({ ...p, requesterId: res.data[0].id }));
   694→      }
   695→    } catch {
   696→      // non-critical
   697→    }
   698→  }, []);
   699→
   700→  useEffect(() => {
   701→    fetchStats();
   702→    fetchLeaves();
   703→    fetchUsers();
   704→  }, [fetchStats, fetchLeaves, fetchUsers]);
   705→
   706→  // Column definitions
   707→  const columns: ColumnDef<LeaveRequest>[] = useMemo(() => [
   708→    {
   709→      accessorKey: 'requester',
   710→      header: 'Requester',
   711→      cell: ({ row }) => {
   712→        const r = row.original.requester;
   713→        if (!r) return '—';
   714→        return (
   715→          <div className="flex items-center gap-2.5">
   716→            <Avatar className="size-8">
   717→              <AvatarFallback className="text-xs bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300">
   718→                {getInitials(r.firstName, r.lastName)}
   719→              </AvatarFallback>
   720→            </Avatar>
   721→            <span className="font-medium whitespace-nowrap">{r.firstName} {r.lastName}</span>
   722→          </div>
   723→        );
   724→      },
   725→    },
   726→    {
   727→      accessorKey: 'type',
   728→      header: 'Type',
   729→      cell: ({ row }) => <StatusBadge status={row.original.type} colorMap={LEAVE_TYPE_COLORS} />,
   730→    },
   731→    {
   732→      accessorKey: 'startDate',
   733→      header: 'Start Date',
   734→      cell: ({ row }) => <span className="whitespace-nowrap">{formatDate(row.original.startDate)}</span>,
   735→    },
   736→    {
   737→      accessorKey: 'endDate',
   738→      header: 'End Date',
   739→      cell: ({ row }) => <span className="whitespace-nowrap">{formatDate(row.original.endDate)}</span>,
   740→    },
   741→    {
   742→      id: 'duration',
   743→      header: 'Duration',
   744→      cell: ({ row }) => (
   745→        <span>{computeDuration(row.original.startDate, row.original.endDate)} days</span>
   746→      ),
   747→    },
   748→    {
   749→      accessorKey: 'status',
   750→      header: 'Status',
   751→      cell: ({ row }) => <StatusBadge status={row.original.status} colorMap={LEAVE_STATUS_COLORS} />,

   754→      id: 'actions',
   755→      header: '',
   756→      size: 120,
   757→      cell: ({ row }) => {
   758→        const leave = row.original;
   759→        if (leave.status !== 'pending') return null;
   760→        return (
   761→          <div className="flex items-center gap-1">
   762→            <Button
   763→              variant="outline"
   764→              size="sm"
   765→              className="h-7 text-xs text-emerald-600 border-emerald-200 hover:bg-emerald-50 dark:border-emerald-800 dark:hover:bg-emerald-950"
   766→              onClick={async () => {
   767→                try {
   768→                  await updateRecord(`/api/hr/leave/${leave.id}`, { status: 'approved' });
   769→                  toast.success('Leave request approved');
   770→                  fetchLeaves();
   771→                  fetchStats();
   772→                } catch (e: any) {
   773→                  toast.error(e.message || 'Failed to approve');
   774→                }
   775→              }}
   776→            >
   777→              <ShieldCheck className="size-3.5 mr-1" />Approve
   778→            </Button>
   779→            <Button
   780→              variant="outline"
   781→              size="sm"
   782→              className="h-7 text-xs text-red-600 border-red-200 hover:bg-red-50 dark:border-red-800 dark:hover:bg-red-950"
   783→              onClick={async () => {
   784→                try {
   785→                  await updateRecord(`/api/hr/leave/${leave.id}`, { status: 'rejected' });
   786→                  toast.success('Leave request rejected');
   787→                  fetchLeaves();
   788→                  fetchStats();
   789→                } catch (e: any) {
   790→                  toast.error(e.message || 'Failed to reject');
   791→                }
   792→              }}
   793→            >
   794→              <ShieldX className="size-3.5 mr-1" />Reject

   825→  };

   863→                <SelectContent className="max-h-48 overflow-y-auto">

   865→                    <SelectItem key={u.id} value={u.id}>
   866→                      {u.firstName} {u.lastName}

   871→            </Field>
