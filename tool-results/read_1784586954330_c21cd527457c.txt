   152→   152→
   153→   153→async function apiFetch<T>(url: string, init?: RequestInit): Promise<{ data: T; meta?: ApiMeta }> {
   154→   154→  const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...init });
   155→   155→  const json = await res.json();
   156→   156→  if (json.error) throw new Error(json.error.message || 'Request failed');
   157→   157→  return json;
   158→   158→}
