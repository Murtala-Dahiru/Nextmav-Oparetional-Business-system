   287→function InvoicesTab() {
   288→  const [statusFilter, setStatusFilter] = useState('all');
   289→  const [search, setSearch] = useState('');
   290→
   291→  const summary = useMemo(() => {
   292→    const total = invoices.reduce((s, i) => s + i.total, 0);
   293→    const paid = invoices.filter((i) => i.status === 'paid').reduce((s, i) => s + i.total, 0);
   294→    const pending = invoices.filter((i) => i.status === 'sent').reduce((s, i) => s + i.total, 0);
   295→    const overdue = invoices.filter((i) => i.status === 'overdue').reduce((s, i) => s + i.total, 0);
   296→    const draft = invoices.filter((i) => i.status === 'draft').reduce((s, i) => s + i.total, 0);
   297→    return { total, paid, pending, overdue, draft };
   298→  }, []);
   299→
   300→  const filtered = useMemo(() => {
   301→    return invoices.filter((inv) => {
   302→      const matchesStatus = statusFilter === 'all' || inv.status === statusFilter;
   303→      const q = search.toLowerCase();
   304→      const matchesSearch = !q ||
   305→        inv.invoiceNumber.toLowerCase().includes(q) ||
   306→        inv.contactName.toLowerCase().includes(q) ||
   307→        inv.companyName.toLowerCase().includes(q);
   308→      return matchesStatus && matchesSearch;
   309→    });
   310→  }, [statusFilter, search]);

   312→  const summaryCards = [
   313→    { label: 'Total Invoices', value: formatCurrency(summary.total), color: 'text-emerald-600 bg-emerald-50' },

   316→    { label: 'Overdue', value: formatCurrency(summary.overdue), color: 'text-red-600 bg-red-50' },
   317→    { label: 'Draft', value: formatCurrency(summary.draft), color: 'text-gray-600 bg-gray-50' },

   322→      {/* Summary Cards */}

   326→            <Card className="border-0 shadow-sm">
   327→              <CardContent className="p-4">
   328→                <p className="text-xs text-muted-foreground font-medium">{card.label}</p>

   335→
