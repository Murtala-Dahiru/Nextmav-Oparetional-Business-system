   131→   131→async function fetchList<T>(url: string): Promise<{ data: T[]; meta: ApiMeta }> {
   132→   132→  const res = await fetch(url);
   133→   133→  if (!res.ok) {
   134→   134→    const e = await res.json().catch(() => ({ message: 'Request failed' }));
   135→   135→    throw new Error((e as any).message || `Error ${res.status}`);
   136→   136→  }
   137→   137→  return res.json();
   138→   138→}
   139→   139→
   140→   140→async function createRecord<T>(url: string, data: unknown): Promise<T> {
   141→   141→  const res = await fetch(url, {
   142→   142→    method: 'POST',
   143→   143→    headers: { 'Content-Type': 'application/json' },
   144→   144→    body: JSON.stringify(data),
   145→   145→  });
   146→   146→  if (!res.ok) {
   147→   147→    const e = await res.json().catch(() => ({ message: 'Create failed' }));
   148→   148→    throw new Error((e as any).message || 'Create failed');
   149→   149→  }
   150→   150→  const json = await res.json();
   151→   151→  return json.data;
   152→   152→}

   198→   198→function computeDuration(start: string, end: string): number {
   199→   199→  const s = new Date(start);
   200→   200→  const e = new Date(end);
   201→   201→  const diffMs = e.getTime() - s.getTime();
   202→   202→  return Math.max(1, Math.ceil(diffMs / (1000 * 60 * 60 * 24)) + 1);
   203→   203→}
   204→   204→
   205→   205→function isDateInRange(dateStr: string, start: string, end: string): boolean {
   206→   206→  const d = new Date(dateStr);
   207→   207→  const s = new Date(start);
   208→   208→  const e = new Date(end);
   209→   209→  return d >= s && d <= e;
   210→   210→}
