   125→   125→async function fetchList<T>(url: string): Promise<{ data: T[]; meta: ApiMeta }> {
   126→   126→  const res = await fetch(url);
   127→   127→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Request failed' })); throw new Error((e as any).message || `Error ${res.status}`); }
   128→   128→  return res.json();
   129→   129→}
   130→   130→
   131→   131→async function createRecord<T>(url: string, data: unknown): Promise<T> {
   132→   132→  const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
   133→   133→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Create failed' })); throw new Error((e as any).message || 'Create failed'); }
   134→   134→  const json = await res.json();
   135→   135→  return json.data;
   136→   136→}
   137→   137→
   138→   138→async function updateRecord<T>(url: string, data: unknown): Promise<T> {
   139→   139→  const res = await fetch(url, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
   140→   140→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Update failed' })); throw new Error((e as any).message || 'Update failed'); }
   141→   141→  const json = await res.json();
   142→   142→  return json.data;
   143→   143→}
   144→   144→
   145→   145→async function deleteRecord(url: string): Promise<void> {
   146→   146→  const res = await fetch(url, { method: 'DELETE' });
   147→   147→  if (!res.ok) { const e = await res.json().catch(() => ({ message: 'Delete failed' })); throw new Error((e as any).message || 'Delete failed'); }
   148→   148→}
