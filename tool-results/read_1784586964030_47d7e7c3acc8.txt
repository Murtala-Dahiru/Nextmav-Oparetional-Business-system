    92→
    93→async function apiFetch<T>(url: string, init?: RequestInit): Promise<{ data: T; meta?: { total: number; page: number; pageSize: number; totalPages: number } }> {
    94→  const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...init });
    95→  const json = await res.json();
    96→  if (json.error) throw new Error(json.error.message || 'Request failed');
    97→  return json;
    98→}

   100→// ═══════════════════════════════════════════════════════════════════════════
   101→//  Module
   102→// ═══════════════════════════════════════════════════════════════════════════
   103→
   104→export default function WorkspaceModule() {
   105→  // Data state
   106→  const [pages, setPages] = useState<WorkspacePage[]>([]);
   107→  const [loading, setLoading] = useState(true);
   108→
   109→  // Selection state
   110→  const [selectedPageId, setSelectedPageId] = useState<string | null>(null);
   111→  const [sidebarSearch, setSidebarSearch] = useState('');
   112→  const [collapsedFolders, setCollapsedFolders] = useState<Set<string>>(new Set());
   113→
   114→  // Full page content
   115→  const [fullPage, setFullPage] = useState<WorkspacePage | null>(null);
   116→  const [pageLoading, setPageLoading] = useState(false);
   117→  const [editing, setEditing] = useState(false);
   118→  const [editContent, setEditContent] = useState('');
   119→  const [savingContent, setSavingContent] = useState(false);
   120→
   121→  // Inline title editing
   122→  const [isEditingTitle, setIsEditingTitle] = useState(false);
   123→  const [editTitle, setEditTitle] = useState('');
   124→
   125→  // Dialog state
   126→  const [createDialogOpen, setCreateDialogOpen] = useState(false);
   127→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   128→  const [deletingPage, setDeletingPage] = useState<WorkspacePage | null>(null);
   129→  const [saving, setSaving] = useState(false);
   130→  const [deleting, setDeleting] = useState(false);
   131→
   132→  // Fetch all pages for sidebar (large page size)
   133→  const fetchPages = useCallback(async () => {
   134→    setLoading(true);
   135→    try {
   136→      const res = await apiFetch<WorkspacePage[]>('/api/workspace/pages?page=1&pageSize=200');
   137→      setPages(res.data || []);
   138→    } catch (err: any) {
   139→      toast.error(err.message || 'Failed to load pages');
   140→    } finally {
   141→      setLoading(false);
   142→    }
   143→  }, []);
   144→
   145→  useEffect(() => { fetchPages(); }, [fetchPages]);
   146→
   147→  // Fetch full page
   148→  const fetchFullPage = useCallback(async (id: string) => {
   149→    setPageLoading(true);
   150→    setSelectedPageId(id);
   151→    setEditing(false);
   152→    try {
   153→      const res = await apiFetch<WorkspacePage>(`/api/workspace/pages/${id}`);
   154→      setFullPage(res.data);
   155→      setEditContent(res.data.content);
   156→      setEditTitle(res.data.title);
   157→    } catch (err: any) {
   158→      toast.error(err.message || 'Failed to load page');
   159→    } finally {
   160→      setPageLoading(false);
   161→    }
   162→  }, []);
   163→
   164→  // Derived: filtered pages
   165→  const filteredPages = useMemo(() => {
   166→    if (!sidebarSearch) return pages;
   167→    const q = sidebarSearch.toLowerCase();
   168→    return pages.filter((p) => p.title.toLowerCase().includes(q));
   169→  }, [pages, sidebarSearch]);
   170→
   171→  // Derived: folders and non-folder pages
   172→  const rootPages = useMemo(() => {
   173→    const folders = filteredPages.filter((p) => p.isFolder && !p.parentId);
   174→    const items = filteredPages.filter((p) => !p.isFolder && !p.parentId);
   175→    return { folders, items };
   176→  }, [filteredPages]);
   177→
   178→  // Get children of a folder
   179→  const getFolderChildren = useCallback((folderId: string) => {
   180→    return filteredPages.filter((p) => p.parentId === folderId && !p.isFolder);
   181→  }, [filteredPages]);
   182→
   183→  // Toggle folder collapse
   184→  const toggleFolder = useCallback((folderId: string) => {
   185→    setCollapsedFolders((prev) => {
   186→      const next = new Set(prev);
   187→      if (next.has(folderId)) next.delete(folderId);
   188→      else next.add(folderId);
   189→      return next;
   190→    });
   191→  }, []);
   192→
   193→  // Save page content
   194→  const saveContent = useCallback(async () => {
   195→    if (!fullPage) return;
   196→    setSavingContent(true);
   197→    try {
   198→      const res = await apiFetch<WorkspacePage>(`/api/workspace/pages/${fullPage.id}`, {
   199→        method: 'PUT',
   200→        body: JSON.stringify({ content: editContent }),
   201→      });
   202→      setFullPage(res.data);
   203→      setEditing(false);
   204→      fetchPages();
   205→      toast.success('Page saved');
   206→    } catch (err: any) {
   207→      toast.error(err.message || 'Save failed');
   208→    } finally {
   209→      setSavingContent(false);
   210→    }
   211→  }, [fullPage, editContent, fetchPages]);
   212→
   213→  // Save page title (inline)
   214→  const saveTitle = useCallback(async () => {
   215→    if (!fullPage) return;
   216→    try {
   217→      const res = await apiFetch<WorkspacePage>(`/api/workspace/pages/${fullPage.id}`, {
   218→        method: 'PUT',
   219→        body: JSON.stringify({ title: editTitle }),
   220→      });
   221→      setFullPage(res.data);
   222→      setIsEditingTitle(false);
   223→      fetchPages();
   224→    } catch (err: any) {
   225→      toast.error(err.message || 'Failed to update title');
   226→    }
   227→  }, [fullPage, editTitle, fetchPages]);
   228→
   229→  // Toggle star
   230→  const toggleStar = useCallback(async (page: WorkspacePage) => {
   231→    try {
   232→      await apiFetch(`/api/workspace/pages/${page.id}`, {
   233→        method: 'PUT',
   234→        body: JSON.stringify({ isStarred: !page.isStarred }),
   235→      });
   236→      fetchPages();
   237→      if (selectedPageId === page.id) {
   238→        setFullPage((prev) => prev ? { ...prev, isStarred: !prev.isStarred } : null);
   239→      }
   240→    } catch (err: any) {
   241→      toast.error(err.message || 'Failed to toggle star');
   242→    }
   243→  }, [fetchPages, selectedPageId]);
   244→
   245→  // Delete
   246→  const confirmDelete = useCallback(() => {
   247→    if (!deletingPage) return;
   248→    setDeleting(true);
   249→    apiFetch(`/api/workspace/pages/${deletingPage.id}`, { method: 'DELETE' })
   250→      .then(() => {
   251→        toast.success('Page deleted');
   252→        setDeleteDialogOpen(false);
   253→        setDeletingPage(null);
   254→        if (selectedPageId === deletingPage.id) {
   255→          setSelectedPageId(null);
   256→          setFullPage(null);
   257→        }
   258→        fetchPages();
   259→      })
   260→      .catch((err: any) => toast.error(err.message || 'Delete failed'))
   261→      .finally(() => setDeleting(false));
   262→  }, [deletingPage, fetchPages, selectedPageId]);
   263→
   264→  // Sort: starred first, then by updatedAt desc
   265→  const sortedItems = useMemo(() => {
   266→    return [...rootPages.items].sort((a, b) => {
   267→      if (a.isStarred && !b.isStarred) return -1;
   268→      if (!a.isStarred && b.isStarred) return 1;
   269→      return new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime();
   270→    });
   271→  }, [rootPages.items]);
   272→
   273→  const sortedFolders = useMemo(() => {
   274→    return [...rootPages.folders].sort((a, b) => {
   275→      if (a.isStarred && !b.isStarred) return -1;
   276→      if (!a.isStarred && b.isStarred) return 1;
   277→      return new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime();
   278→    });
   279→  }, [rootPages.folders]);
   280→
   281→  // Form for create dialog
   282→  const { register, handleSubmit, control, reset, watch, formState: { errors } } = useForm<PageFormValues>({
   283→    // eslint-disable-next-line @typescript-eslint/no-explicit-any
   284→    resolver: zodResolver(createPageSchema) as any,
   285→    defaultValues: {
   286→      title: '', content: '', icon: 'file-text', color: '#10b981',
   287→      parentId: '', isFolder: false, isStarred: false, lastEditedBy: '',
   288→    },
   289→  });
   290→
   291→  const watchIsFolder = watch('isFolder');
   292→
   293→  // Open create dialog
   294→  const openCreate = useCallback(() => {
   295→    reset({
   296→      title: '', content: '# New Page\n\nStart writing here...', icon: 'file-text', color: '#10b981',
   297→      parentId: '', isFolder: false, isStarred: false, lastEditedBy: '',
   298→    });
   299→    setCreateDialogOpen(true);
   300→  }, [reset]);
   301→
   302→  // Submit create
   303→  const onSubmit = useCallback(async (values: PageFormValues) => {
   304→    setSaving(true);
   305→    try {
   306→      const payload = {
   307→        ...values,
   308→        parentId: values.parentId || '',
   309→      };
   310→      const res = await apiFetch<WorkspacePage>('/api/workspace/pages', {
   311→        method: 'POST',
   312→        body: JSON.stringify(payload),
   313→      });
   314→      toast.success(values.isFolder ? 'Folder created' : 'Page created');
   315→      setCreateDialogOpen(false);
   316→      fetchPages();
   317→      // Auto-select the new page
   318→      if (!values.isFolder) {
   319→        fetchFullPage(res.data.id);
   320→      }
   321→    } catch (err: any) {
   322→      toast.error(err.message || 'Create failed');
   323→    } finally {
   324→      setSaving(false);
   325→    }
   326→  }, [fetchPages, fetchFullPage]);
   327→
   328→  // Page icon component
   329→  const PageIcon = ({ icon, color, className }: { icon: string; color: string; className?: string }) => {
   330→    const IconComp = ICON_MAP[icon] || FileText;
   331→    return <IconComp className={className} style={{ color }} />;
   332→  };
   333→
   334→  // Render sidebar page item
   335→  const renderPageItem = (page: WorkspacePage, indent = 0) => (
   336→    <button
   337→      key={page.id}
   338→      onClick={() => fetchFullPage(page.id)}
   339→      className={`w-full flex items-center gap-2 px-3 py-1.5 text-sm rounded-md transition-colors text-left hover:bg-accent ${
   340→        selectedPageId === page.id ? 'bg-accent text-accent-foreground' : 'text-foreground'
   341→      }`}
   342→      style={{ paddingLeft: `${12 + indent * 20}px` }}
   343→    >
   344→      <PageIcon icon={page.icon} color={page.color} className="size-4 shrink-0" />
   345→      <span className="truncate flex-1">{page.title}</span>
   346→      {page.isStarred && <Star className="size-3 text-amber-500 fill-amber-500 shrink-0" />}
   347→      <span className="text-xs text-muted-foreground shrink-0 hidden sm:inline">
   348→        {formatRelativeTime(page.updatedAt)}
   349→      </span>
   350→    </button>
   351→  );
   352→
   353→  // Render folder group
   354→  const renderFolder = (folder: WorkspacePage) => {
   355→    const children = getFolderChildren(folder.id);
   356→    const isCollapsed = collapsedFolders.has(folder.id);
   357→
   358→    return (
   359→      <div key={folder.id}>
   360→        <button
   361→          onClick={() => toggleFolder(folder.id)}
   362→          className="w-full flex items-center gap-2 px-3 py-1.5 text-sm rounded-md transition-colors text-left hover:bg-accent text-foreground"
   363→        >
   364→          {isCollapsed ? (
   365→            <ChevronRight className="size-3.5 text-muted-foreground shrink-0" />
   366→          ) : (
   367→            <ChevronDown className="size-3.5 text-muted-foreground shrink-0" />
   368→          )}
   369→          <PageIcon icon={folder.icon} color={folder.color} className="size-4 shrink-0" />
   370→          <span className="truncate flex-1 font-medium">{folder.title}</span>
   371→          {folder.isStarred && <Star className="size-3 text-amber-500 fill-amber-500 shrink-0" />}
   372→          <span className="text-xs text-muted-foreground shrink-0">{children.length}</span>
   373→        </button>
   374→        {!isCollapsed && (
   375→          <div className="mt-0.5 space-y-0.5">
   376→            {children.map((child) => renderPageItem(child, 1))}
   377→            {children.length === 0 && (
   378→              <p className="text-xs text-muted-foreground pl-12 py-1">Empty folder</p>
   379→            )}
   380→          </div>
   381→        )}
   382→      </div>
   383→    );
   384→  };
   385→
   386→  // Current selected page icon for header
   387→  const currentIcon = fullPage ? (ICON_MAP[fullPage.icon] || FileText) : FileText;
   388→
   389→  return (
   390→    <TooltipProvider>
   391→      <div className="flex flex-1 overflow-hidden h-full">
   392→        {/* ─── Left Sidebar ─── */}
   393→        <aside className="w-64 border-r bg-card flex flex-col shrink-0 hidden sm:flex">
   394→          {/* New Page Button */}
   395→          <div className="p-3">
   396→            <Button
   397→              onClick={openCreate}
   398→              variant="outline"
   399→              className="w-full justify-start gap-2 text-emerald-600 border-emerald-200 hover:bg-emerald-50 dark:border-emerald-800 dark:hover:bg-emerald-950/30"
   400→            >
   401→              <Plus className="size-4" /> New Page
   402→            </Button>
   403→          </div>
   404→
   405→          {/* Search */}
   406→          <div className="px-3 pb-2">
   407→            <div className="relative">
   408→              <Search className="text-muted-foreground absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5" />
   409→              <Input
   410→                placeholder="Search pages..."
   411→                value={sidebarSearch}
   412→                onChange={(e) => setSidebarSearch(e.target.value)}
   413→                className="pl-8 h-8 text-sm"
   414→              />
   415→            </div>
   416→          </div>
   417→
   418→          <Separator />
   419→
   420→          {/* Page List */}
   421→          <ScrollArea className="flex-1 px-2 py-2">
   422→            {loading ? (
   423→              <div className="space-y-2 px-1">
   424→                {Array.from({ length: 8 }).map((_, i) => (
   425→                  <Skeleton key={i} className="h-8 w-full" />
   426→                ))}
   427→              </div>
   428→            ) : filteredPages.length === 0 ? (
   429→              <p className="text-xs text-muted-foreground text-center py-8">
   430→                {sidebarSearch ? 'No pages match your search.' : 'No pages yet.'}
   431→              </p>
   432→            ) : (
   433→              <div className="space-y-0.5">
   434→                {/* Folders */}
   435→                {sortedFolders.length > 0 && (
   436→                  <div className="space-y-0.5 mb-2">
   437→                    {sortedFolders.map(renderFolder)}
   438→                  </div>
   439→                )}
   440→
   441→                {/* Divider if both exist */}
   442→                {sortedFolders.length > 0 && sortedItems.length > 0 && (
   443→                  <Separator className="my-2" />
   444→                )}
   445→
   446→                {/* Root pages */}
   447→                {sortedItems.map((page) => renderPageItem(page))}
   448→              </div>
   449→            )}
   450→          </ScrollArea>
   451→        </aside>
   452→
   453→        {/* ─── Right Content Area ─── */}
   454→        <main className="flex-1 flex flex-col overflow-hidden">
   455→          {!selectedPageId ? (
   456→            <EmptyState
   457→              icon={BookMarked}
   458→              title="Select a page or create a new one"
   459→              description="Choose a page from the sidebar or create a new page to get started."
   460→              action={{ label: 'New Page', onClick: openCreate }}
   461→            />
   462→          ) : pageLoading ? (
   463→            <div className="flex-1 p-6 space-y-4">
   464→              <Skeleton className="h-8 w-64" />
   465→              <Skeleton className="h-4 w-96" />
   466→              <div className="space-y-2 mt-6">
   467→                {Array.from({ length: 12 }).map((_, i) => (
   468→                  <Skeleton key={i} className="h-4 w-full" />
   469→                ))}
   470→              </div>
   471→            </div>
   472→          ) : fullPage ? (
   473→            <div className="flex-1 flex flex-col overflow-hidden">
   474→              {/* Page Header */}
   475→              <div className="flex items-center gap-3 px-6 py-4 border-b shrink-0">
   476→                <PageIcon icon={fullPage.icon} color={fullPage.color} className="size-5 shrink-0" />
   477→
   478→                {isEditingTitle ? (
   479→                  <Input
   480→                    value={editTitle}
   481→                    onChange={(e) => setEditTitle(e.target.value)}
   482→                    onBlur={saveTitle}
   483→                    onKeyDown={(e) => { if (e.key === 'Enter') saveTitle(); if (e.key === 'Escape') setIsEditingTitle(false); }}
   484→                    className="h-8 text-lg font-semibold max-w-md"
   485→                    autoFocus
   486→                  />
   487→                ) : (
   488→                  <h1
   489→                    className="text-lg font-semibold text-foreground cursor-pointer hover:text-emerald-600 transition-colors"
   490→                    onClick={() => { setEditTitle(fullPage.title); setIsEditingTitle(true); }}
   491→                  >
   492→                    {fullPage.title}
   493→                  </h1>
   494→                )}
   495→
   496→                <Tooltip>
   497→                  <TooltipTrigger asChild>
   498→                    <Button
   499→                      variant="ghost"
   500→                      size="icon"
   501→                      className="size-8"
   502→                      onClick={() => toggleStar(fullPage)}
   503→                    >
   504→                      {fullPage.isStarred ? (
   505→                        <Star className="size-4 text-amber-500 fill-amber-500" />
   506→                      ) : (
   507→                        <StarOff className="size-4 text-muted-foreground" />
   508→                      )}
   509→                    </Button>
   510→                  </TooltipTrigger>
   511→                  <TooltipContent>{fullPage.isStarred ? 'Unstar' : 'Star'}</TooltipContent>
   512→                </Tooltip>
   513→
   514→                {/* Edit / Save / More */}
   515→                <div className="flex items-center gap-1 ml-auto">
   516→                  {!editing ? (
   517→                    <Button
   518→                      variant="outline"
   519→                      size="sm"
   520→                      className="gap-1.5"
   521→                      onClick={() => { setEditContent(fullPage.content); setEditing(true); }}
   522→                    >
   523→                      <Pencil className="size-3.5" /> Edit
   524→                    </Button>
   525→                  ) : (
   526→                    <>
   527→                      <Button
   528→                        variant="outline"
   529→                        size="sm"
   530→                        onClick={() => { setEditing(false); setEditContent(fullPage.content); }}
   531→                      >
   532→                        Cancel
   533→                      </Button>
   534→                      <Button
   535→                        size="sm"
   536→                        className="bg-emerald-600 text-white hover:bg-emerald-700 gap-1.5"
   537→                        onClick={saveContent}
