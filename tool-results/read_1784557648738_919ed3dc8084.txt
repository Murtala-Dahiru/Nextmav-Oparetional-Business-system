   103→   103→function formatDate(): string {
   104→   104→  return new Date().toLocaleDateString('en-US', {
   105→   105→    weekday: 'long',
   106→   106→    year: 'numeric',
   107→   107→    month: 'long',
   108→   108→    day: 'numeric',
   109→   109→  });
   110→   110→}
   111→   111→
   112→   112→function timeAgo(dateStr: string): string {
   113→   113→  const now = new Date();
   114→   114→  const date = new Date(dateStr);
   115→   115→  const diffMs = now.getTime() - date.getTime();
   116→   116→  const diffMin = Math.floor(diffMs / 60000);
   117→   117→  if (diffMin < 1) return 'Just now';
   118→   118→  if (diffMin < 60) return `${diffMin}m ago`;
   119→   119→  const diffHr = Math.floor(diffMin / 60);
   120→   120→  if (diffHr < 24) return `${diffHr}h ago`;
   121→   121→  const diffDay = Math.floor(diffHr / 24);
   122→   122→  return `${diffDay}d ago`;
   123→   123→}
   124→   124→
   125→   125→function formatTime(dateStr: string): string {
   126→   126→  const date = new Date(dateStr);
   127→   127→  return date.toLocaleTimeString('en-US', {
   128→   128→    hour: 'numeric',
   129→   129→    minute: '2-digit',
   130→   130→    hour12: true,
   131→   131→  });
   132→   132→}
   133→   133→
   134→   134→function formatShortDate(dateStr: string): string {
   135→   135→  const date = new Date(dateStr);

   137→   137→    month: 'short',
   138→   138→    day: 'numeric',
   139→   139→  });
   140→   140→}
