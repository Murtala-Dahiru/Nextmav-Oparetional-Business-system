    43→    43→function formatLastSeen(d: string) {
    44→    44→  const date = new Date(d);
    45→    45→  const now = new Date('2026-07-20T11:00:00Z');
    46→    46→  const diffMs = now.getTime() - date.getTime();
    47→    47→  const diffMins = Math.floor(diffMs / 60000);
    48→    48→  if (diffMins < 60) return `${diffMins}m ago`;
    49→    49→  const diffHrs = Math.floor(diffMins / 60);
    50→    50→  if (diffHrs < 24) return `${diffHrs}h ago`;
    51→    51→  const diffDays = Math.floor(diffHrs / 24);
    52→    52→  return `${diffDays}d ago`;
    53→    53→}
