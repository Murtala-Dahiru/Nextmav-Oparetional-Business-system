   275→   275→export default function DashboardModule() {
   276→   276→  const { setActiveModule } = useAppStore();
   277→   277→  const [loading, setLoading] = useState(true);
   278→   278→  const [data, setData] = useState<DashboardData | null>(null);
   279→   279→  const [error, setError] = useState<string | null>(null);
   280→   280→
   281→   281→  // Fetch all data on mount
   282→   282→  useEffect(() => {
   283→   283→    let cancelled = false;
   284→   284→
   285→   285→    async function fetchAll() {
   286→   286→      try {
   287→   287→        const endpoints = [
   288→   288→          '/api/crm/leads?pageSize=100',
   289→   289→          '/api/crm/deals?pageSize=100',
   290→   290→          '/api/projects/projects?pageSize=100',
   291→   291→          '/api/projects/tasks?pageSize=100',
   292→   292→          '/api/support/tickets?pageSize=100',
   293→   293→          '/api/finance/invoices?pageSize=100',
   294→   294→          '/api/finance/expenses?pageSize=100',
   295→   295→          '/api/activity-log?pageSize=10',
   296→   296→        ];
   297→   297→
   298→   298→        const results = await Promise.all(
   299→   299→          endpoints.map((url) =>
   300→   300→            fetch(url).then((r) => {
   301→   301→              if (!r.ok) throw new Error(`Failed to fetch ${url}`);
   302→   302→              return r.json();
   303→   303→            }),
   304→   304→          ),
   305→   305→        );
   306→   306→
   307→   307→        if (cancelled) return;
   308→   308→
   309→   309→        setData({
   310→   310→          leads: (results[0] as ApiResponse<Lead>).data,
   311→   311→          deals: (results[1] as ApiResponse<Deal>).data,
   312→   312→          projects: (results[2] as ApiResponse<Project>).data,
   313→   313→          tasks: (results[3] as ApiResponse<Task>).data,
   314→   314→          tickets: (results[4] as ApiResponse<Ticket>).data,
   315→   315→          invoices: (results[5] as ApiResponse<Invoice>).data,
   316→   316→          expenses: (results[6] as ApiResponse<Expense>).data,
   317→   317→          activities: (results[7] as ApiResponse<ActivityItem>).data,
   318→   318→        });
   319→   319→      } catch (e: unknown) {
   320→   320→        if (!cancelled) {
   321→   321→          setError(e instanceof Error ? e.message : 'Failed to load dashboard data');
   322→   322→        }
   323→   323→      } finally {
   324→   324→        if (!cancelled) setLoading(false);
   325→   325→      }
   326→   326→    }
   327→   327→
   328→   328→    fetchAll();
   329→   329→    return () => {
   330→   330→      cancelled = true;
   331→   331→    };
   332→   332→  }, []);
   333→   333→
   334→   334→  // ---- Computed KPIs ----
   335→   335→
   336→   336→  const kpis = useMemo(() => {
   337→   337→    if (!data) return null;
   338→   338→
   339→   339→    // Total Revenue: sum of paid invoices
   340→   340→    const paidInvoices = data.invoices.filter((inv) => inv.status === 'paid');
   341→   341→    const totalRevenue = paidInvoices.reduce((sum, inv) => sum + inv.total, 0);
   342→   342→
   343→   343→    // Revenue change: compare this month vs last month paid invoices
   344→   344→    const now = new Date();
   345→   345→    const thisMonthStart = new Date(now.getFullYear(), now.getMonth(), 1);
   346→   346→    const lastMonthStart = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   347→   347→
   348→   348→    const thisMonthRevenue = paidInvoices
   349→   349→      .filter((inv) => inv.paidAt && new Date(inv.paidAt) >= thisMonthStart)
   350→   350→      .reduce((sum, inv) => sum + inv.total, 0);
   351→   351→
   352→   352→    const lastMonthRevenue = paidInvoices
   353→   353→      .filter(
   354→   354→        (inv) =>
   355→   355→          inv.paidAt &&
   356→   356→          new Date(inv.paidAt) >= lastMonthStart &&
   357→   357→          new Date(inv.paidAt) < thisMonthStart,
   358→   358→      )
   359→   359→      .reduce((sum, inv) => sum + inv.total, 0);
   360→   360→
   361→   361→    const revenueChange =
   362→   362→      lastMonthRevenue > 0
   363→   363→        ? ((thisMonthRevenue - lastMonthRevenue) / lastMonthRevenue) * 100
   364→   364→        : thisMonthRevenue > 0
   365→   365→          ? 100
   366→   366→          : 0;
   367→   367→
   368→   368→    // Active Deals: not closed-won/closed-lost
   369→   369→    const activeDeals = data.deals.filter(
   370→   370→      (d) => d.stage !== 'closed-won' && d.stage !== 'closed-lost',
   371→   371→    );
   372→   372→    const activeDealsCount = activeDeals.length;
   373→   373→
   374→   374→    // Deals change: this month vs last month created
   375→   375→    const thisMonthDeals = data.deals.filter(
   376→   376→      (d) => new Date(d.createdAt) >= thisMonthStart,
   377→   377→    ).length;
   378→   378→    const lastMonthDeals = data.deals.filter(
   379→   379→      (d) => new Date(d.createdAt) >= lastMonthStart && new Date(d.createdAt) < thisMonthStart,
   380→   380→    ).length;
   381→   381→    const dealsChange =
   382→   382→      lastMonthDeals > 0
   383→   383→        ? ((thisMonthDeals - lastMonthDeals) / lastMonthDeals) * 100
   384→   384→        : thisMonthDeals > 0
   385→   385→          ? 100
   386→   386→          : 0;
   387→   387→
   388→   388→    // Open Tickets: open / in-progress / pending
   389→   389→    const openTickets = data.tickets.filter(
   390→   390→      (t) => t.status === 'open' || t.status === 'in-progress' || t.status === 'pending',
   391→   391→    );
   392→   392→    const openTicketsCount = openTickets.length;
   393→   393→
   394→   394→    // Tickets change: this month vs last month
   395→   395→    const thisMonthTickets = data.tickets.filter(
   396→   396→      (t) => new Date(t.createdAt) >= thisMonthStart,
   397→   397→    ).length;
   398→   398→    const lastMonthTickets = data.tickets.filter(
   399→   399→      (t) => new Date(t.createdAt) >= lastMonthStart && new Date(t.createdAt) < thisMonthStart,
   400→   400→    ).length;
   401→   401→    const ticketsChange =
   402→   402→      lastMonthTickets > 0
   403→   403→        ? ((thisMonthTickets - lastMonthTickets) / lastMonthTickets) * 100
   404→   404→        : thisMonthTickets > 0
   405→   405→          ? 100
   406→   406→          : 0;
   407→   407→
   408→   408→    // Active Projects
   409→   409→    const activeProjects = data.projects.filter((p) => p.status === 'active');
   410→   410→    const activeProjectsCount = activeProjects.length;
   411→   411→
   412→   412→    const thisMonthProjects = data.projects.filter(
   413→   413→      (p) => new Date(p.createdAt) >= thisMonthStart,
   414→   414→    ).length;
   415→   415→    const lastMonthProjects = data.projects.filter(
   416→   416→      (p) => new Date(p.createdAt) >= lastMonthStart && new Date(p.createdAt) < thisMonthStart,
   417→   417→    ).length;
   418→   418→    const projectsChange =
   419→   419→      lastMonthProjects > 0
   420→   420→        ? ((thisMonthProjects - lastMonthProjects) / lastMonthProjects) * 100
   421→   421→        : thisMonthProjects > 0
   422→   422→          ? 100
   423→   423→          : 0;
   424→   424→
   425→   425→    // Tasks Due This Week
   426→   426→    const weekFromNow = new Date();
   427→   427→    weekFromNow.setDate(weekFromNow.getDate() + 7);
   428→   428→    weekFromNow.setHours(23, 59, 59, 999);
   429→   429→    const todayStart = new Date();
   430→   430→    todayStart.setHours(0, 0, 0, 0);
   431→   431→    const tasksDueThisWeek = data.tasks.filter(
   432→   432→      (t) =>
   433→   433→        t.dueDate &&
   434→   434→        t.status !== 'done' &&
   435→   435→        new Date(t.dueDate) >= todayStart &&
   436→   436→        new Date(t.dueDate) <= weekFromNow,
   437→   437→    ).length;
   438→   438→
   439→   439→    // New Leads This Month
   440→   440→    const thirtyDaysAgo = new Date();
   441→   441→    thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);
   442→   442→    const newLeadsThisMonth = data.leads.filter(
   443→   443→      (l) => new Date(l.createdAt) >= thirtyDaysAgo,
   444→   444→    ).length;
   445→   445→
   446→   446→    const thisMonthLeads = data.leads.filter(
   447→   447→      (l) => new Date(l.createdAt) >= thisMonthStart,
   448→   448→    ).length;
   449→   449→    const lastMonthLeads = data.leads.filter(
   450→   450→      (l) => new Date(l.createdAt) >= lastMonthStart && new Date(l.createdAt) < thisMonthStart,
   451→   451→    ).length;
   452→   452→    const leadsChange =
   453→   453→      lastMonthLeads > 0
   454→   454→        ? ((thisMonthLeads - lastMonthLeads) / lastMonthLeads) * 100
   455→   455→        : thisMonthLeads > 0
   456→   456→          ? 100
   457→   457→          : 0;
   458→   458→
   459→   459→    return {
   460→   460→      totalRevenue,
   461→   461→      revenueChange,
   462→   462→      activeDealsCount,
   463→   463→      dealsChange,
   464→   464→      openTicketsCount,
   465→   465→      ticketsChange,
   466→   466→      activeProjectsCount,
   467→   467→      projectsChange,
   468→   468→      tasksDueThisWeek,
   469→   469→      newLeadsThisMonth,
   470→   470→      leadsChange,
   471→   471→    };
   472→   472→  }, [data]);
   473→   473→
   474→   474→  // ---- Chart: Monthly Revenue ----
   475→   475→
   476→   476→  const revenueChartData = useMemo(() => {
   477→   477→    if (!data) return [];
   478→   478→
   479→   479→    const paidInvoices = data.invoices.filter((inv) => inv.status === 'paid' && inv.paidAt);
   480→   480→
   481→   481→    // Group by month (last 6 months)
   482→   482→    const monthMap = new Map<string, number>();
   483→   483→    const now = new Date();
   484→   484→
   485→   485→    for (let i = 5; i >= 0; i--) {
   486→   486→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   487→   487→      const key = d.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   488→   488→      monthMap.set(key, 0);
   489→   489→    }
   490→   490→
   491→   491→    paidInvoices.forEach((inv) => {
   492→   492→      const d = new Date(inv.paidAt!);
   493→   493→      const key = d.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   494→   494→      if (monthMap.has(key)) {
   495→   495→        monthMap.set(key, monthMap.get(key)! + inv.total);
   496→   496→      }
   497→   497→    });
   498→   498→
   499→   499→    return Array.from(monthMap.entries()).map(([month, revenue]) => ({ month, revenue }));
   500→   500→  }, [data]);
   501→   501→
   502→   502→  // ---- Chart: Deal Pipeline ----
   503→   503→
   504→   504→  const pipelineChartData = useMemo(() => {
   505→   505→    if (!data) return [];
   506→   506→
   507→   507→    const stageMap = new Map<string, { count: number; value: number }>();
   508→   508→
