    36→function formatCurrency(v: number) {
    37→  return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', maximumFractionDigits: 0 }).format(v);
    38→}
    39→
    40→function formatNumber(v: number) {
    41→  return new Intl.NumberFormat('en-US').format(v);
    42→}
    43→
    44→function formatDate(d: string) {
    45→  return new Date(d).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    46→}
    47→
    48→function getInitials(name: string) {
    49→  return name
    50→    .split(' ')
    51→    .map((w) => w[0])
    52→    .join('')
    53→    .slice(0, 2)
    54→    .toUpperCase();
    55→}

   513→  const filtered = useMemo(() => {
   514→    const q = search.toLowerCase();
   515→    if (!q) return opportunities;
   516→    return opportunities.filter((o) =>
   517→      `${o.name} ${o.companyName} ${o.contactName} ${o.ownerName} ${o.stage}`.toLowerCase().includes(q)
   518→    );
   519→  }, [search]);
   520→
   521→  const totalPipeline = filtered.reduce((a, o) => a + o.value, 0);
   522→  const weightedValue = filtered.reduce((a, o) => a + o.value * (o.probability / 100), 0);
   523→  const avgDeal = filtered.length > 0 ? totalPipeline / filtered.length : 0;
   524→
   525→  return (
   526→    <div className="space-y-4">
   527→      <TabHeader title="Deals" description="All opportunities and deal pipeline" />
   528→
   529→      <div className="flex items-center gap-3">
   530→        <div className="relative w-64">
   531→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   532→          <Input
   533→            placeholder="Search deals..."
   534→            value={search}
   535→            onChange={(e) => setSearch(e.target.value)}
   536→            className="pl-8 h-9"
   537→          />
   538→        </div>
   539→        <span className="text-sm text-muted-foreground ml-auto">Showing {filtered.length} deals</span>
   540→      </div>
   541→
   542→      {/* summary cards */}
   543→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   544→        {[
   545→          { label: 'Total Pipeline Value', value: formatCurrency(totalPipeline), icon: DollarSign, accent: 'text-teal-600 dark:text-teal-400', bg: 'bg-teal-50 dark:bg-teal-950/40' },
   546→          { label: 'Weighted Value', value: formatCurrency(weightedValue), icon: TrendingUp, accent: 'text-teal-600 dark:text-teal-400', bg: 'bg-teal-50 dark:bg-teal-950/40' },
   547→          { label: 'Average Deal Size', value: formatCurrency(avgDeal), icon: BarChart3, accent: 'text-amber-600 dark:text-amber-400', bg: 'bg-amber-50 dark:bg-amber-950/40' },
   548→        ].map((card) => (
   549→          <div key={card.label} className={cn('rounded-xl border border-border p-4', card.bg)}>
   550→            <div className="flex items-center gap-2 mb-1">
   551→              <card.icon className={cn('size-4', card.accent)} />
   552→              <span className="text-xs text-muted-foreground font-medium">{card.label}</span>
   553→            </div>
   554→            <p className={cn('text-xl font-bold', card.accent)}>{card.value}</p>
   555→          </div>
   556→        ))}
   557→      </div>
   558→
   559→      <div className="overflow-x-auto"><Table>
   560→        <TableHeader>
   561→          <TableRow className="hover:bg-transparent">
   562→            <TableHead>Deal</TableHead>
   563→            <TableHead>Value</TableHead>
   564→            <TableHead>Stage</TableHead>
   565→            <TableHead>Probability</TableHead>
   566→            <TableHead>Close Date</TableHead>
   567→            <TableHead>Contact</TableHead>
   568→            <TableHead>Company</TableHead>
   569→            <TableHead>Owner</TableHead>
   570→          </TableRow>
   571→        </TableHeader>
   572→        <TableBody>
   573→          {filtered.map((opp) => {
   574→            const stage = pipeline.stages.find((s) => s.id === opp.stageId);
   575→            return (
   576→              <TableRow key={opp.id}>
   577→                <TableCell className="font-medium">{opp.name}</TableCell>
   578→                <TableCell className="font-semibold">{formatCurrency(opp.value)}</TableCell>
   579→                <TableCell>
   580→                  <Badge
   581→                    variant="outline"
   582→                    className="border-0 text-xs"
   583→                    style={{
   584→                      backgroundColor: stage ? `${stage.color}18` : undefined,
   585→                      color: stage?.color,
   586→                    }}
   587→                  >
   588→                    {opp.stage}
   589→                  </Badge>
   590→                </TableCell>
   591→                <TableCell>
   592→                  <div className="flex items-center gap-2 w-28">
   593→                    <Progress value={opp.probability} className="h-1.5 flex-1" />
   594→                    <span className="text-xs text-muted-foreground w-7 text-right">{opp.probability}%</span>
   595→                  </div>
   596→                </TableCell>
   597→                <TableCell className="text-muted-foreground">{formatDate(opp.closeDate)}</TableCell>
   598→                <TableCell>{opp.contactName}</TableCell>
   599→                <TableCell>{opp.companyName}</TableCell>
   600→                <TableCell>
   601→                  <div className="flex items-center gap-1.5">
   602→                    <div className="size-5 rounded-full bg-teal-600 text-white flex items-center justify-center text-[9px] font-bold">
   603→                      {getInitials(opp.ownerName)}
   604→                    </div>
   605→                    <span className="text-sm">{opp.ownerName}</span>
   606→                  </div>
   607→                </TableCell>
   608→              </TableRow>
   609→            );
   610→          })}
   611→          {filtered.length === 0 && (
   612→            <TableRow>
   613→              <TableCell colSpan={8} className="text-center py-8 text-muted-foreground">
   614→                No deals match your search.
   615→              </TableCell>
   616→            </TableRow>
   617→          )}
   618→        </TableBody>
   619→      </Table></div>
   620→    </div>
   621→  );
   622→}
