   331→export default function SupportModule() {
   332→  // ── Data state ──
   333→  const [tickets, setTickets] = useState<Ticket[]>([]);
   334→  const [users, setUsers] = useState<Assignee[]>([]);
   335→  const [stats, setStats] = useState<TicketStats>({ open: 0, inProgress: 0, pending: 0, resolved: 0 });
   336→
   337→  // ── Table state ──
   338→  const [total, setTotal] = useState(0);
   339→  const [page, setPage] = useState(0);
   340→  const [pageSize, setPageSize] = useState(PAGE_SIZE);
   341→  const [search, setSearch] = useState('');
   342→  const [sorting, setSorting] = useState<SortingState>([]);
   343→  const [columnFilters, setColumnFilters] = useState<ColumnFiltersState>([]);
   344→
   345→  // ── Dialog state ──
   346→  const [createOpen, setCreateOpen] = useState(false);
   347→  const [editTicket, setEditTicket] = useState<Ticket | null>(null);
   348→  const [deleteTicket, setDeleteTicket] = useState<Ticket | null>(null);
   349→  const [kbArticle, setKbArticle] = useState<(typeof KB_ARTICLES)[number] | null>(null);
   350→
   351→  // ── Loading states ──
   352→  const [ticketsLoading, setTicketsLoading] = useState(true);
   353→  const [creating, setCreating] = useState(false);
   354→  const [updating, setUpdating] = useState(false);
   355→  const [deleting, setDeleting] = useState(false);
   356→
   357→  // ─── Fetch users ──────────────────────────────────────────────────
   358→  useEffect(() => {
   359→    (async () => {
   360→      try {
   361→        const res = await fetch('/api/admin/users?pageSize=100');
   362→        const json = await res.json();
   363→        if (json.data) setUsers(json.data);
   364→      } catch { /* silent */ }
   365→    })();
   366→  }, []);
   367→
   368→  // ─── Fetch tickets ────────────────────────────────────────────────
   369→  const fetchTickets = useCallback(async () => {
   370→    setTicketsLoading(true);
   371→    try {
   372→      const params = new URLSearchParams();
   373→      params.set('page', String(page + 1));
   374→      params.set('pageSize', String(pageSize));
   375→      if (search) params.set('search', search);
   376→      if (sorting.length > 0) {
   377→        params.set('sort', sorting[0].id);
   378→        params.set('sortDir', sorting[0].desc ? 'desc' : 'asc');
   379→      }
   380→      columnFilters.forEach((f) => {
   381→        params.set(f.id, String(f.value));
   382→      });
   383→
   384→      const res = await fetch(`/api/support/tickets?${params}`);
   385→      const json = await res.json();
   386→      if (json.error) { toast.error(json.error.message); return; }
   387→
   388→      setTickets(json.data ?? []);
   389→      setTotal(json.meta?.total ?? 0);
   390→
   391→      // Fetch stats
   392→      const statsRes = await fetch('/api/support/tickets?pageSize=1000');
   393→      const statsJson = await statsRes.json();
   394→      const allTickets: Ticket[] = statsJson.data ?? [];
   395→      setStats({
   396→        open: allTickets.filter((t) => t.status === 'open').length,
   397→        inProgress: allTickets.filter((t) => t.status === 'in-progress').length,
   398→        pending: allTickets.filter((t) => t.status === 'pending').length,
   399→        resolved: allTickets.filter((t) => t.status === 'resolved').length,
   400→      });
   401→    } catch {
   402→      toast.error('Failed to load tickets');
   403→    } finally {
   404→      setTicketsLoading(false);
   405→    }
   406→  }, [page, pageSize, search, sorting, columnFilters]);
   407→
   408→  useEffect(() => {
   409→    fetchTickets();
   410→  }, [fetchTickets]);
   411→
   412→  // ─── Create ticket ────────────────────────────────────────────────
   413→  const createForm = useForm({
   414→    resolver: zodResolver(createTicketSchema),
   415→    defaultValues: {
   416→      subject: '',
   417→      description: '',
   418→      priority: 'medium',
   419→      status: 'open',
   420→      category: 'general',
   421→      contactName: '',
   422→      contactEmail: '',
   423→      assigneeId: null as string | null,
   424→      dueDate: null as string | null,
   425→      resolution: '',
   426→    },
   427→  });
   428→
   429→  const handleCreate = useCallback(async (values: z.infer<typeof createTicketSchema>) => {
   430→    setCreating(true);
   431→    try {
   432→      const ticketNumber = `TKT-${String(total + 1).padStart(3, '0')}`;
   433→      const res = await fetch('/api/support/tickets', {
   434→        method: 'POST',
   435→        headers: { 'Content-Type': 'application/json' },
   436→        body: JSON.stringify({ ...values, ticketNumber }),
   437→      });
   438→      const json = await res.json();
   439→      if (json.error) { toast.error(json.error.message); return; }
   440→      toast.success('Ticket created');
   441→      setCreateOpen(false);
   442→      createForm.reset();
   443→      fetchTickets();
   444→    } catch {
   445→      toast.error('Failed to create ticket');
   446→    } finally {
   447→      setCreating(false);
   448→    }
   449→  }, [total, createForm, fetchTickets]);
   450→
   451→  // ─── Edit ticket ──────────────────────────────────────────────────
   452→  const editForm = useForm({
   453→    resolver: zodResolver(updateTicketSchema),
   454→  });
   455→
   456→  useEffect(() => {
   457→    if (editTicket) {
   458→      editForm.reset({
   459→        subject: editTicket.subject,
   460→        description: editTicket.description,
   461→        priority: editTicket.priority,
   462→        status: editTicket.status,
   463→        category: editTicket.category,
   464→        contactName: editTicket.contactName,
   465→        contactEmail: editTicket.contactEmail,
   466→        assigneeId: editTicket.assigneeId,
   467→        dueDate: editTicket.dueDate ? editTicket.dueDate.slice(0, 10) : null,
   468→        resolution: editTicket.resolution,
   469→      });
   470→    }
   471→  }, [editTicket, editForm]);
   472→
   473→  const handleUpdate = useCallback(async (values: z.infer<typeof updateTicketSchema>) => {
   474→    if (!editTicket) return;
   475→    setUpdating(true);
   476→    try {
   477→      const data: Record<string, unknown> = { ...values };
   478→      if (data.dueDate && typeof data.dueDate === 'string') {
   479→        data.dueDate = new Date(data.dueDate).toISOString();
   480→      }
   481→      const res = await fetch(`/api/support/tickets/${editTicket.id}`, {
   482→        method: 'PUT',
   483→        headers: { 'Content-Type': 'application/json' },
   484→        body: JSON.stringify(data),
   485→      });
   486→      const json = await res.json();
   487→      if (json.error) { toast.error(json.error.message); return; }
   488→      toast.success('Ticket updated');
   489→      setEditTicket(null);
   490→      editForm.reset();
   491→      fetchTickets();
   492→    } catch {
   493→      toast.error('Failed to update ticket');
   494→    } finally {
   495→      setUpdating(false);
   496→    }
   497→  }, [editTicket, editForm, fetchTickets]);
   498→
   499→  // ─── Delete ticket ────────────────────────────────────────────────
   500→  const handleDelete = useCallback(async () => {
   501→    if (!deleteTicket) return;
   502→    setDeleting(true);
   503→    try {
   504→      const res = await fetch(`/api/support/tickets/${deleteTicket.id}`, { method: 'DELETE' });
   505→      const json = await res.json();
   506→      if (json.error) { toast.error(json.error.message); return; }
   507→      toast.success('Ticket deleted');
   508→      setDeleteTicket(null);
   509→      fetchTickets();
   510→    } catch {
   511→      toast.error('Failed to delete ticket');
   512→    } finally {
   513→      setDeleting(false);
   514→    }
   515→  }, [deleteTicket, fetchTickets]);
   516→
   517→  // ─── Status change (quick) ────────────────────────────────────────
   518→  const handleQuickStatus = useCallback(async (ticket: Ticket, newStatus: string) => {
   519→    try {
   520→      const res = await fetch(`/api/support/tickets/${ticket.id}`, {
   521→        method: 'PUT',
   522→        headers: { 'Content-Type': 'application/json' },
   523→        body: JSON.stringify({ status: newStatus }),
   524→      });
   525→      const json = await res.json();
   526→      if (json.error) { toast.error(json.error.message); return; }
   527→      toast.success(`Status changed to ${newStatus}`);
   528→      fetchTickets();
   529→    } catch {
   530→      toast.error('Failed to update status');
   531→    }
   532→  }, [fetchTickets]);
   533→
   534→  // ─── Table filters ────────────────────────────────────────────────
   535→  const filters: DataTableFilter[] = useMemo(
   536→    () => [
   537→      {
   538→        key: 'status',
   539→        label: 'Status',
   540→        options: TICKET_STATUSES.map((s) => ({
   541→          value: s,
   542→          label: s.replace(/-/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase()),
   543→        })),
   544→      },
   545→      {
   546→        key: 'priority',
   547→        label: 'Priority',
   548→        options: TICKET_PRIORITIES.map((p) => ({
   549→          value: p,
   550→          label: p.charAt(0).toUpperCase() + p.slice(1),
   551→        })),
   552→      },
   553→      {
   554→        key: 'category',
   555→        label: 'Category',
   556→        options: [
