   296→   296→        }

   318→   318→      { label: 'Team Members', value: s.totalEmployees, icon: User },
   319→   319→      { label: 'Pending Invoices', value: s.pendingInvoices, icon: FileText },
   320→   320→    ];
   321→   321→  }, [data]);
   322→   322→
   323→   323→  // Active Deals: not closed-won/closed-lost
   324→   324→  const activeDealsCount = data.deals.filter(
   325→   325→    (d) => d.stage !== 'closed-won' && d.stage !== 'closed-lost',
   326→   326→  ).length;
   327→   327→
   328→   328→    // Deals change: this month vs last month created
   329→   329→    const thisMonthDeals = data.deals.filter(
   330→   330→      (d) => new Date(d.createdAt) >= thisMonthStart,
   331→   331→    ).length;
   332→   332→    const lastMonthDeals = data.deals.filter(
   333→   333→      (d) => new Date(d.createdAt) >= lastMonthStart && new Date(d.createdAt) < thisMonthStart,
   334→   334→    ).length;

   337→   337→        ? ((thisMonthDeals - lastMonthDeals) / lastMonthDeals) * 100
   338→   338→        : thisMonthDeals > 0
   339→   339→          ? 100
   340→   340→          : 0;

   342→   342→    // Open Tickets: open / in-progress / pending
   343→   343→    const openTickets = data.tickets.filter(
   344→   344→      (t) => t.status === 'open' || t.status === 'in-progress' || t.status === 'pending',
   345→   345→    );
   346→   346→    const openTicketsCount = openTickets.length;
   347→   347→
   348→   348→    // Tickets change: this month vs last month
   349→   349→    const thisMonthTickets = data.tickets.filter(
   350→   350→      (t) => new Date(t.createdAt) >= thisMonthStart,
   351→   351→    ).length;
   352→   352→    const lastMonthTickets = data.tickets.filter(
   353→   353→      (t) => new Date(t.createdAt) >= lastMonthStart && new Date(t.createdAt) < thisMonthStart,
   354→   354→    ).length;
   355→   355→    const ticketsChange =
   356→   356→      lastMonthTickets > 0
   357→   357→        ? ((thisMonthTickets - lastMonthTickets) / lastMonthTickets) * 100
   358→   358→        : thisMonthTickets > 0
   359→   359→          ? 100
   360→   360→          : 0;
   361→   361→
   362→   362→    // Active Projects
   363→   363→    const activeProjects = data.projects.filter((p) => p.status === 'active');
   364→   364→    const activeProjectsCount = activeProjects.length;
   365→   365→
   366→   366→    const thisMonthProjects = data.projects.filter(
   367→   367→      (p) => new Date(p.createdAt) >= thisMonthStart,
   368→   368→    ).length;
   369→   369→    const lastMonthProjects = data.projects.filter(
   370→   370→      (p) => new Date(p.createdAt) >= lastMonthStart && new Date(p.createdAt) < thisMonthStart,
   371→   371→    ).length;

   373→   373→      lastMonthProjects > 0

   375→   375→        : thisMonthProjects > 0
   376→   376→          ? 100
   377→   377→          : 0;

   379→   379→    // Tasks Due This Week
   380→   380→    const weekFromNow = new Date();
   381→   381→    weekFromNow.setDate(weekFromNow.getDate() + 7);
   382→   382→    weekFromNow.setHours(23, 59, 59, 999);

   412→   412→
   413→   413→    return {

   449→   449→
   450→   450→  // ---- Chart: Deal Pipeline ----

   465→   465→        entry.value += deal.value;
   466→   466→      }
   467→   467→    });
   468→   468→
   469→   469→    return Array.from(stageMap.entries())
   470→   470→      .filter(([, v]) => v.count > 0)
   471→   471→      .map(([stage, v]) => ({
   472→   472→        stage: stage.replace(/-/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase()),
   473→   473→        count: v.count,
   474→   474→        value: v.value,
   475→   475→      }));
   476→   476→  }, [data]);
   477→   477→

   479→   479→
   480→   480→  const leadChartData = useMemo(() => {
   481→   481→    if (!data?.leadByStatus) return [];
   482→   482→    const statusOrder = [new, contacted, qualified, proposal, negotiation, won, lost];
   483→   483→    return statusOrder

   485→   485→        const found = data.leadByStatus.find((s: any) => s.status === status);
   486→   486→        return { status, count: found?._count ?? 0 };
   487→   487→      });
   488→   488→  }, [data]);
   489→   489→
   490→   490→    return Array.from(statusMap.entries())
   491→   491→      .map(([status, count]) => ({
   492→   492→        name: status.replace(/-/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase()),
   493→   493→        value: count,
   494→   494→        color: LEAD_STATUS_COLORS[status] ?? '#94a3b8',
   495→   495→      }))
   496→   496→      .sort((a, b) => b.value - a.value);
   497→   497→  }, [data]);
   498→   498→
   499→   499→  // ---- Top Deals ----
   500→   500→
