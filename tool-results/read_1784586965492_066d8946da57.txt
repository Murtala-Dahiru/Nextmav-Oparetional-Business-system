    73→async function apiFetch<T>(url: string, init?: RequestInit): Promise<T> {
    74→  const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...init });
    75→  const json = await res.json();
    76→  if (!res.ok) throw new Error(json.error?.message ?? 'Request failed');
    77→  return json;
    78→}

   145→              <Label htmlFor="p-cost">Cost</Label>

   243→//  Main Module
   244→// ═══════════════════════════════════════════════════════════════
   245→
   246→export default function InventoryModule() {
   247→  // ── Products State ──
   248→  const [products, setProducts] = useState<Product[]>([]);
   249→  const [productMeta, setProductMeta] = useState<ApiMeta>({ total: 0, page: 1, pageSize: 20, totalPages: 0 });
   250→  const [productLoading, setProductLoading] = useState(true);
   251→  const [productPage, setProductPage] = useState(0);
   252→  const [productPageSize, setProductPageSize] = useState(20);
   253→  const [productSearch, setProductSearch] = useState('');
   254→  const [productFilters, setProductFilters] = useState<ColumnFiltersState>([]);
   255→  const [productSorting, setProductSorting] = useState<SortingState>([]);
   256→
   257→  // ── Warehouses State ──
   258→  const [warehouses, setWarehouses] = useState<Warehouse[]>([]);
   259→  const [warehouseLoading, setWarehouseLoading] = useState(false);
   260→  const [warehouseProductCounts, setWarehouseProductCounts] = useState<Record<string, number>>({});
   261→
   262→  // ── Dialogs ──
   263→  const [productDialogOpen, setProductDialogOpen] = useState(false);
   264→  const [editingProduct, setEditingProduct] = useState<Product | null>(null);
   265→  const [productSubmitting, setProductSubmitting] = useState(false);
   266→
   267→  const [warehouseDialogOpen, setWarehouseDialogOpen] = useState(false);
   268→  const [editingWarehouse, setEditingWarehouse] = useState<Warehouse | null>(null);
   269→  const [warehouseSubmitting, setWarehouseSubmitting] = useState(false);
   270→
   271→  const [deleteTarget, setDeleteTarget] = useState<{ type: 'product' | 'warehouse'; id: string; name: string } | null>(null);
   272→  const [deleting, setDeleting] = useState(false);
   273→
   274→  const [stats, setStats] = useState<Stats>({ totalProducts: 0, totalValue: 0, lowStock: 0 });
   275→
   276→  // ── Fetch products ──
   277→  const fetchProducts = useCallback(async () => {
   278→    setProductLoading(true);
   279→    try {
   280→      const params = new URLSearchParams({
   281→        page: String(productPage + 1),
   282→        pageSize: String(productPageSize),
   283→        sort: productSorting[0]?.id || 'createdAt',
   284→        sortDir: productSorting[0]?.desc ? 'desc' : 'asc',
   285→      });
   286→      if (productSearch) params.set('search', productSearch);
   287→      productFilters.forEach((f) => { params.set(f.id, String(f.value)); });
   288→
   289→      const res = await apiFetch<{ data: Product[]; meta: ApiMeta }>(`/api/inventory/products?${params}`);
   290→      setProducts(res.data);
   291→      setProductMeta(res.meta);
   292→    } catch (e: any) {
   293→      toast.error(e.message);
   294→    } finally {
   295→      setProductLoading(false);
   296→    }
   297→  }, [productPage, productPageSize, productSearch, productFilters, productSorting]);
   298→
   299→  // ── Fetch stats ──
   300→  const fetchStats = useCallback(async () => {
   301→    try {
   302→      const res = await apiFetch<{ data: Product[]; meta: ApiMeta }>('/api/inventory/products?pageSize=9999');
   303→      const all = res.data;
   304→      const totalProducts = all.length;
   305→      const totalValue = all.reduce((s, p) => s + p.price * p.stock, 0);
   306→      const lowStock = all.filter((p) => p.stock < p.reorderLevel).length;
   307→      setStats({ totalProducts, totalValue, lowStock });
   308→    } catch { /* ignore */ }
   309→  }, []);
   310→
   311→  // ── Fetch warehouses ──
   312→  const fetchWarehouses = useCallback(async () => {
   313→    setWarehouseLoading(true);
   314→    try {
   315→      const res = await apiFetch<{ data: Warehouse[]; meta: ApiMeta }>('/api/inventory/warehouses?pageSize=100');
   316→      setWarehouses(res.data);
   317→
   318→      const allProducts = await apiFetch<{ data: Product[]; meta: ApiMeta }>('/api/inventory/products?pageSize=9999');
   319→      const counts: Record<string, number> = {};
   320→      allProducts.data.forEach((p) => {
   321→        if (p.warehouseId) counts[p.warehouseId] = (counts[p.warehouseId] || 0) + 1;
   322→      });
   323→      setWarehouseProductCounts(counts);
   324→    } catch (e: any) {
   325→      toast.error(e.message);
   326→    } finally {
   327→      setWarehouseLoading(false);
   328→    }
   329→  }, []);
   330→
   331→  useEffect(() => { fetchProducts(); }, [fetchProducts]);
   332→  useEffect(() => { fetchStats(); }, [fetchStats]);
   333→
   334→  // ── Product CRUD ──
   335→  const handleProductSubmit = async (form: ProductFormState) => {
   336→    setProductSubmitting(true);
   337→    try {
   338→      const payload = {
   339→        name: form.name, sku: form.sku, category: form.category,
   340→        price: Number(form.price) || 0, cost: Number(form.cost) || 0,
   341→        stock: Number(form.stock) || 0, unit: form.unit,
   342→        reorderLevel: Number(form.reorderLevel) || 0, isActive: form.isActive,
   343→      };
   344→      if (editingProduct) {
   345→        await apiFetch(`/api/inventory/products/${editingProduct.id}`, { method: 'PUT', body: JSON.stringify(payload) });
   346→        toast.success('Product updated');
   347→      } else {
   348→        await apiFetch('/api/inventory/products', { method: 'POST', body: JSON.stringify(payload) });
   349→        toast.success('Product created');
   350→      }
   351→      setProductDialogOpen(false);
   352→      setEditingProduct(null);
   353→      fetchProducts();
   354→      fetchStats();
   355→    } catch (e: any) {
   356→      toast.error(e.message);
   357→    } finally {
   358→      setProductSubmitting(false);
   359→    }
   360→  };
   361→
   362→  const handleDelete = async () => {
   363→    if (!deleteTarget) return;
   364→    setDeleting(true);
   365→    try {
   366→      const endpoint = deleteTarget.type === 'product'
   367→        ? `/api/inventory/products/${deleteTarget.id}`
   368→        : `/api/inventory/warehouses/${deleteTarget.id}`;
   369→      await apiFetch(endpoint, { method: 'DELETE' });
   370→      toast.success(`${deleteTarget.type === 'product' ? 'Product' : 'Warehouse'} deleted`);
   371→      setDeleteTarget(null);
   372→      if (deleteTarget.type === 'product') { fetchProducts(); fetchStats(); } else { fetchWarehouses(); }
   373→    } catch (e: any) {
   374→      toast.error(e.message);
   375→    } finally {
   376→      setDeleting(false);
   377→    }
   378→  };
   379→
   380→  // ── Warehouse CRUD ──
   381→  const handleWarehouseSubmit = async (form: WarehouseFormState) => {
   382→    setWarehouseSubmitting(true);
   383→    try {
   384→      const payload = {
   385→        name: form.name, location: form.location,
   386→        capacity: Number(form.capacity) || 0, isActive: form.isActive,
   387→      };
   388→      if (editingWarehouse) {
   389→        await apiFetch(`/api/inventory/warehouses/${editingWarehouse.id}`, { method: 'PUT', body: JSON.stringify(payload) });
   390→        toast.success('Warehouse updated');
   391→      } else {
   392→        await apiFetch('/api/inventory/warehouses', { method: 'POST', body: JSON.stringify(payload) });
   393→        toast.success('Warehouse created');
   394→      }
   395→      setWarehouseDialogOpen(false);
   396→      setEditingWarehouse(null);
   397→      fetchWarehouses();
   398→    } catch (e: any) {
   399→      toast.error(e.message);
   400→    } finally {
   401→      setWarehouseSubmitting(false);
   402→    }
   403→  };
   404→
   405→  // ── Product columns ──
   406→  const productColumns: ColumnDef<Product>[] = [
   407→    { accessorKey: 'sku', header: 'SKU', size: 100 },
   408→    { accessorKey: 'name', header: 'Name', size: 180 },
   409→    {
   410→      accessorKey: 'category', header: 'Category', size: 110,
   411→      cell: ({ row }) => {
   412→        const v = row.original.category;
   413→        return <Badge variant="secondary" className={CATEGORY_COLORS[v] || CATEGORY_COLORS.general}>{v}</Badge>;
   414→      },
   415→    },
   416→    {
   417→      accessorKey: 'price', header: 'Price', size: 100,
   418→      cell: ({ row }) => <span>{formatCurrency(row.original.price)}</span>,
   419→    },
   420→    {
   421→      accessorKey: 'cost', header: 'Cost', size: 100,
   422→      cell: ({ row }) => <span>{formatCurrency(row.original.cost)}</span>,
   423→    },
   424→    {
   425→      accessorKey: 'stock', header: 'Stock', size: 80,
   426→      cell: ({ row }) => {
   427→        const r = row.original;
   428→        const isLow = r.stock < r.reorderLevel;
   429→        return (
   430→          <span className={isLow ? 'text-red-600 font-semibold' : 'text-emerald-600 font-semibold'}>
   431→            {r.stock}
   432→          </span>
   433→        );
   434→      },
   435→    },
   436→    { accessorKey: 'unit', header: 'Unit', size: 80 },
   437→    { accessorKey: 'reorderLevel', header: 'Reorder', size: 80 },
   438→    {
   439→      accessorKey: 'isActive', header: 'Status', size: 100,
   440→      cell: ({ row }) => (
   441→        <Badge variant={row.original.isActive ? 'default' : 'secondary'} className={row.original.isActive ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300' : ''}>
   442→          {row.original.isActive ? 'Active' : 'Inactive'}
   443→        </Badge>
   444→      ),
   445→    },
   446→    {
   447→      id: 'actions', size: 60,
   448→      cell: ({ row }) => (
   449→        <DropdownMenu>
   450→          <DropdownMenuTrigger asChild>
   451→            <Button variant="ghost" size="icon" className="size-8"><MoreHorizontal className="size-4" /></Button>
   452→          </DropdownMenuTrigger>
   453→          <DropdownMenuContent align="end">
   454→            <DropdownMenuItem onClick={() => { setEditingProduct(row.original); setProductDialogOpen(true); }}>
   455→              <Pencil className="size-4 mr-2" /> Edit
   456→            </DropdownMenuItem>
   457→            <DropdownMenuItem className="text-red-600" onClick={() => setDeleteTarget({ type: 'product', id: row.original.id, name: row.original.name })}>
   458→              <Trash2 className="size-4 mr-2" /> Delete
   459→            </DropdownMenuItem>
   460→          </DropdownMenuContent>
   461→        </DropdownMenu>
   462→      ),
   463→    },
   464→  ];
   465→
   466→  // ── Product filters ──
   467→  const productFilterDefs: DataTableFilter[] = [
   468→    {
   469→      key: 'category', label: 'Category',
   470→      options: CATEGORIES.map((c) => ({ value: c, label: c })),
   471→    },
   472→    {
   473→      key: 'isActive', label: 'Status',
   474→      options: [
   475→        { value: 'true', label: 'Active' },
