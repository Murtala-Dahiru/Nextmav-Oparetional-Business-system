    94→// ─── Helpers ────────────────────────────────────────────────────────────────
    95→
    96→function getGreeting(): string {
    97→  const hour = new Date().getHours();
    98→  if (hour < 12) return 'Good morning';
    99→  if (hour < 17) return 'Good afternoon';
   100→  return 'Good evening';
   101→}
   102→
   103→function formatDate(): string {
   104→  return new Date().toLocaleDateString('en-US', {
   105→    weekday: 'long',
   106→    year: 'numeric',
   107→    month: 'long',
   108→    day: 'numeric',
   109→  });
   110→}
   111→
   112→function timeAgo(dateStr: string): string {
   113→  const now = new Date();
   114→  const date = new Date(dateStr);
   115→  const diffMs = now.getTime() - date.getTime();
   116→  const diffMin = Math.floor(diffMs / 60000);
   117→  if (diffMin < 1) return 'Just now';
   118→  if (diffMin < 60) return `${diffMin}m ago`;
   119→  const diffHr = Math.floor(diffMin / 60);
   120→  if (diffHr < 24) return `${diffHr}h ago`;
   121→  const diffDay = Math.floor(diffHr / 24);
   122→  return `${diffDay}d ago`;
   123→}
   124→
   125→function formatTime(dateStr: string): string {
   126→  const date = new Date(dateStr);
   127→  return date.toLocaleTimeString('en-US', {
   128→    hour: 'numeric',
   129→    minute: '2-digit',
   130→    hour12: true,
   131→  });
   132→}
   133→
   134→function formatShortDate(dateStr: string): string {
   135→  const date = new Date(dateStr);
   136→  return date.toLocaleDateString('en-US', {
   137→    month: 'short',
   138→    day: 'numeric',
   139→  });
   140→}
   141→
   142→function getInitials(name: string): string {
   143→  return name
   144→    .split(' ')
   145→    .map((n) => n[0])
   146→    .join('')
   147→    .toUpperCase()
   148→    .slice(0, 2);
   149→}

   153→function CustomBarTooltip({ active, payload, label }: {
   154→  active?: boolean;
   155→  payload?: Array<{ name: string; value: number; color: string }>;
   156→  label?: string;
   157→}) {
   158→  if (!active || !payload?.length) return null;
   159→  return (
   160→    <div className="rounded-lg border bg-background p-3 shadow-lg">
   161→      <p className="mb-1 text-sm font-semibold text-foreground">{label}</p>
   162→      {payload.map((entry, idx) => (
   163→        <p key={idx} className="text-sm" style={{ color: entry.color }}>
   164→          {entry.name}: ${entry.value.toLocaleString()}
   165→        </p>
   166→      ))}
   167→    </div>
   168→  );
   169→}
   170→
   171→function CustomPieTooltip({ active, payload }: {
   172→  active?: boolean;
   173→  payload?: Array<{ name: string; value: number; payload: { name: string; value: number } }>;
   174→}) {
   175→  if (!active || !payload?.length) return null;
   176→  return (
   177→    <div className="rounded-lg border bg-background p-3 shadow-lg">
   178→      <p className="text-sm font-semibold text-foreground">{payload[0].name}</p>
   179→      <p className="text-sm text-muted-foreground">{payload[0].value}%</p>
   180→    </div>
   181→  );
   182→}
