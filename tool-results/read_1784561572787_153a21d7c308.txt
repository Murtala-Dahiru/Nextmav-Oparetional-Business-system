    98→function getGreeting(): string {
    99→  const hour = new Date().getHours();
   100→  if (hour < 12) return 'Good morning';
   101→  if (hour < 17) return 'Good afternoon';
   102→  return 'Good evening';
   103→}
   104→
   105→function formatDate(): string {
   106→  return new Date().toLocaleDateString('en-US', {
   107→    weekday: 'long',
   108→    year: 'numeric',
   109→    month: 'long',
   110→    day: 'numeric',
   111→  });
   112→}
   113→
   114→function timeAgo(dateStr: string): string {
   115→  const now = new Date();
   116→  const date = new Date(dateStr);
   117→  const diffMs = now.getTime() - date.getTime();
   118→  const diffMin = Math.floor(diffMs / 60000);
   119→  if (diffMin < 1) return 'Just now';
   120→  if (diffMin < 60) return `${diffMin}m ago`;
   121→  const diffHr = Math.floor(diffMin / 60);
   122→  if (diffHr < 24) return `${diffHr}h ago`;
   123→  const diffDay = Math.floor(diffHr / 24);
   124→  return `${diffDay}d ago`;
   125→}
   126→
   127→function formatTime(dateStr: string): string {
   128→  const date = new Date(dateStr);
   129→  return date.toLocaleTimeString('en-US', {
   130→    hour: 'numeric',
   131→    minute: '2-digit',
   132→    hour12: true,
   133→  });
   134→}
   135→
   136→function formatShortDate(dateStr: string): string {
   137→  const date = new Date(dateStr);
   138→  return date.toLocaleDateString('en-US', {
   139→    month: 'short',
   140→    day: 'numeric',
   141→  });
   142→}
   143→
   144→function getInitials(name: string): string {
   145→  return name
   146→    .split(' ')
   147→    .map((n) => n[0])
   148→    .join('')
   149→    .toUpperCase()
   150→    .slice(0, 2);
   151→}

   153→// ─── Custom Tooltip Components ──────────────────────────────────────────────

   155→function CustomBarTooltip({ active, payload, label }: {
   156→  active?: boolean;
   157→  payload?: Array<{ name: string; value: number; color: string }>;
   158→  label?: string;
   159→}) {
   160→  if (!active || !payload?.length) return null;
   161→  return (
   162→    <div className="rounded-lg border bg-background p-3 shadow-lg">
   163→      <p className="mb-1 text-sm font-semibold text-foreground">{label}</p>
   164→      {payload.map((entry, idx) => (
   165→        <p key={idx} className="text-sm" style={{ color: entry.color }}>
   166→          {entry.name}: ${entry.value.toLocaleString()}
   167→        </p>
   168→      ))}
   169→    </div>
   170→  );
   171→}
   172→
   173→function CustomPieTooltip({ active, payload }: {
   174→  active?: boolean;
   175→  payload?: Array<{ name: string; value: number; payload: { name: string; value: number } }>;
   176→}) {
   177→  if (!active || !payload?.length) return null;
   178→  return (

   181→      <p className="text-sm text-muted-foreground">{payload[0].value}%</p>
   182→    </div>
   183→  );
   184→}
