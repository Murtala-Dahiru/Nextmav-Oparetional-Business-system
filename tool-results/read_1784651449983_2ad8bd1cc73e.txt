   245→function ChartTooltipContent({
   246→  active,
   247→  payload,
   248→  label,
   249→  formatter,
   250→}: {
   251→  active?: boolean;
   252→  payload?: Array<{ value: number; name: string; dataKey: string; color?: string }>;
   253→  label?: string;
   254→  formatter?: (value: number) => string;
   255→}) {
   256→  if (!active || !payload?.length) return null;
   257→  return (
   258→    <div className="border-border/50 bg-background rounded-lg border px-3 py-2 text-xs shadow-xl">
   259→      {label && <p className="font-medium text-foreground mb-1">{label}</p>}
   260→      {payload.map((item, idx) => (
   261→        <div key={idx} className="flex items-center gap-2">
   262→          <span
   263→            className="inline-block h-2 w-2 rounded-full"
   264→            style={{ backgroundColor: item.color ?? CHART_COLORS[0] }}
   265→          />
   266→          <span className="text-muted-foreground">
   267→            {item.name}: {formatter ? formatter(item.value) : item.value.toLocaleString()}
   268→          </span>
   269→        </div>
   270→      ))}
   271→    </div>
   272→  );
   273→}

   279→export default function DashboardModule() {
   280→  const { setActiveModule } = useAppStore();
   281→  const [loading, setLoading] = useState(true);
   282→  const [data, setData] = useState<DashboardData | null>(null);
   283→  const [error, setError] = useState<string | null>(null);
   284→
   285→  // Fetch dashboard data on mount
   286→  useEffect(() => {
   287→    let cancelled = false;
   288→
   289→    async function fetchDashboard() {
   290→      try {
   291→        setLoading(true);
   292→        const res = await fetch('/api/dashboard');
   293→        if (!res.ok) throw new Error(`Failed to fetch dashboard: ${res.status}`);
   294→        const json = await res.json();
   295→        if (cancelled) return;
   296→        setData(json.data);
   297→      } catch (e: unknown) {
   298→        if (!cancelled) {
   299→          setError(e instanceof Error ? e.message : 'Failed to load dashboard data');
   300→        }
   301→      } finally {
   302→        if (!cancelled) setLoading(false);
   303→      }
   304→    }
   305→
   306→    fetchDashboard();
   307→    return () => {
   308→      cancelled = true;
   309→    };
   310→  }, []);
   311→
   312→  // ---- Computed KPIs ----
   313→
   314→  const kpis = useMemo(() => {
   315→    if (!data) return null;
   316→
   317→    const now = new Date();
   318→    const thisMonthStart = new Date(now.getFullYear(), now.getMonth(), 1);
   319→    const lastMonthStart = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   320→
   321→    const calcChange = (thisCount: number, lastCount: number) =>
   322→      lastCount > 0
   323→        ? ((thisCount - lastCount) / lastCount) * 100
   324→        : thisCount > 0
   325→          ? 100
   326→          : 0;
   327→
   328→    // Total Revenue from paid invoices
   329→    const totalRevenue = data.invoices
   330→      .filter((inv) => inv.status === 'paid')
   331→      .reduce((sum, inv) => sum + inv.total, 0);
   332→    const thisMonthRevenue = data.invoices
   333→      .filter((inv) => inv.status === 'paid' && new Date(inv.paidAt ?? inv.createdAt) >= thisMonthStart)
   334→      .reduce((sum, inv) => sum + inv.total, 0);
   335→    const lastMonthRevenue = data.invoices
   336→      .filter((inv) => inv.status === 'paid' && new Date(inv.paidAt ?? inv.createdAt) >= lastMonthStart && new Date(inv.paidAt ?? inv.createdAt) < thisMonthStart)
   337→      .reduce((sum, inv) => sum + inv.total, 0);
   338→    const revenueChange = calcChange(thisMonthRevenue, lastMonthRevenue);
   339→
   340→    // Active Deals
   341→    const activeDealsCount = data.deals.filter(
   342→      (d) => d.stage !== 'closed-won' && d.stage !== 'closed-lost',
   343→    ).length;
   344→    const thisMonthDeals = data.deals.filter(
   345→      (d) => new Date(d.createdAt) >= thisMonthStart,
   346→    ).length;
   347→    const lastMonthDeals = data.deals.filter(
   348→      (d) => new Date(d.createdAt) >= lastMonthStart && new Date(d.createdAt) < thisMonthStart,
   349→    ).length;
   350→    const dealsChange = calcChange(thisMonthDeals, lastMonthDeals);
   351→
   352→    // Open Tickets
   353→    const openTicketsCount = data.tickets.filter(
   354→      (t) => t.status === 'open' || t.status === 'in-progress' || t.status === 'pending',
   355→    ).length;
   356→    const thisMonthTickets = data.tickets.filter(
   357→      (t) => new Date(t.createdAt) >= thisMonthStart,
   358→    ).length;
   359→    const lastMonthTickets = data.tickets.filter(
   360→      (t) => new Date(t.createdAt) >= lastMonthStart && new Date(t.createdAt) < thisMonthStart,
   361→    ).length;
   362→    const ticketsChange = calcChange(thisMonthTickets, lastMonthTickets);
   363→
   364→    // Active Projects
   365→    const activeProjectsCount = data.projects.filter((p) => p.status === 'active').length;
   366→    const thisMonthProjects = data.projects.filter(
   367→      (p) => new Date(p.createdAt) >= thisMonthStart,
   368→    ).length;
   369→    const lastMonthProjects = data.projects.filter(
   370→      (p) => new Date(p.createdAt) >= lastMonthStart && new Date(p.createdAt) < thisMonthStart,
   371→    ).length;
   372→    const projectsChange = calcChange(thisMonthProjects, lastMonthProjects);
   373→
   374→    // Tasks Due This Week
   375→    const weekFromNow = new Date();
   376→    weekFromNow.setDate(weekFromNow.getDate() + 7);
   377→    weekFromNow.setHours(23, 59, 59, 999);
   378→    const todayStart = new Date();
   379→    todayStart.setHours(0, 0, 0, 0);
   380→    const tasksDueThisWeek = data.tasks.filter(
   381→      (t) =>
   382→        t.dueDate &&
   383→        t.status !== 'done' &&
   384→        new Date(t.dueDate) >= todayStart &&
   385→        new Date(t.dueDate) <= weekFromNow,
   386→    ).length;
   387→
   388→    // New Leads This Month
   389→    const thirtyDaysAgo = new Date();
   390→    thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);
   391→    const newLeadsThisMonth = data.leads.filter(
   392→      (l) => new Date(l.createdAt) >= thirtyDaysAgo,
   393→    ).length;
   394→    const thisMonthLeads = data.leads.filter(
   395→      (l) => new Date(l.createdAt) >= thisMonthStart,
   396→    ).length;
   397→    const lastMonthLeads = data.leads.filter(
   398→      (l) => new Date(l.createdAt) >= lastMonthStart && new Date(l.createdAt) < thisMonthStart,
   399→    ).length;
   400→    const leadsChange = calcChange(thisMonthLeads, lastMonthLeads);
   401→
   402→    return {
   403→      totalRevenue,
   404→      revenueChange,
   405→      activeDealsCount,
   406→      dealsChange,
   407→      openTicketsCount,
   408→      ticketsChange,
   409→      activeProjectsCount,
   410→      projectsChange,
   411→      tasksDueThisWeek,
   412→      newLeadsThisMonth,
   413→      leadsChange,
   414→    };
   415→  }, [data]);
   416→
   417→  // ---- Chart: Monthly Revenue ----
   418→
   419→  const revenueChartData = useMemo(() => {
   420→    if (!data) return [];
   421→
   422→    const paidInvoices = data.invoices.filter((inv) => inv.status === 'paid' && inv.paidAt);
   423→
   424→    // Group by month (last 6 months)
   425→    const monthMap = new Map<string, number>();
   426→    const now = new Date();
   427→
   428→    for (let i = 5; i >= 0; i--) {
   429→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   430→      const key = d.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   431→      monthMap.set(key, 0);
   432→    }
   433→
   434→    paidInvoices.forEach((inv) => {
   435→      const d = new Date(inv.paidAt!);
   436→      const key = d.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   437→      if (monthMap.has(key)) {
   438→        monthMap.set(key, monthMap.get(key)! + inv.total);
   439→      }
   440→    });
   441→
   442→    return Array.from(monthMap.entries()).map(([month, revenue]) => ({ month, revenue }));
   443→  }, [data]);
   444→
   445→  // ---- Chart: Deal Pipeline ----
   446→
   447→  const pipelineChartData = useMemo(() => {
   448→    if (!data) return [];
   449→
   450→    const stageMap = new Map<string, { count: number; value: number }>();
   451→
   452→    DEAL_STAGE_ORDER.forEach((stage) => {
   453→      stageMap.set(stage, { count: 0, value: 0 });
   454→    });
   455→
   456→    data.deals.forEach((deal) => {
   457→      const entry = stageMap.get(deal.stage);
   458→      if (entry) {
   459→        entry.count += 1;
   460→        entry.value += deal.value;
   461→      }
   462→    });
   463→
   464→    return Array.from(stageMap.entries())
   465→      .filter(([, v]) => v.count > 0)
   466→      .map(([stage, v]) => ({
   467→        stage: stage.replace(/-/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase()),
   468→        count: v.count,
   469→        value: v.value,
   470→      }));
   471→  }, [data]);
   472→
   473→  // ---- Chart: Lead Status Distribution ----
   474→
   475→  const leadChartData = useMemo(() => {
   476→    if (!data?.leadByStatus) return [];
   477→    return data.leadByStatus
   478→      .map((s: any) => ({
   479→        name: (s.status as string).replace(/-/g, ' ').replace(/\b\w/g, (c: string) => c.toUpperCase()),
   480→        value: s.count,
   481→        color: LEAD_STATUS_COLORS[s.status] ?? '#94a3b8',
   482→      }))
   483→      .sort((a: any, b: any) => b.value - a.value);
   484→  }, [data]);
   485→
   486→  // ---- Top Deals ----
   487→
   488→  const topDeals = useMemo(() => {
   489→    if (!data) return [];
   490→    return [...data.deals]
   491→      .sort((a, b) => b.value - a.value)
   492→      .slice(0, 5);
   493→  }, [data]);
   494→
   495→  // ---- Quick Action Handler ----
   496→
   497→  const handleQuickAction = useCallback(
   498→    (module: ModuleId) => {
   499→      setActiveModule(module);
   500→    },
   501→    [setActiveModule],
   502→  );
   503→
   504→  // ---- Module navigation from activity ----
   505→
   506→  const handleActivityClick = useCallback(
   507→    (moduleName: string) => {
   508→      const moduleMap: Record<string, ModuleId> = {
   509→        crm: 'crm',
   510→        projects: 'projects',
   511→        support: 'support',
   512→        finance: 'finance',
   513→        hr: 'hr',
   514→        inventory: 'inventory',
   515→        calendar: 'calendar',
   516→        communication: 'communication',
   517→        admin: 'admin',
   518→        workspace: 'workspace',
   519→      };
   520→      if (moduleMap[moduleName]) {
   521→        setActiveModule(moduleMap[moduleName]);
   522→      }
   523→    },
   524→    [setActiveModule],
   525→  );
   526→
   527→  // ---- Error State ----
   528→
   529→  if (error && !data) {
   530→    return (
   531→      <div className="space-y-6">
   532→        <PageHeader
   533→          title="Dashboard"
   534→          description="Overview of your business operations"
   535→          icon={TrendingUp}
   536→        />
   537→        <EmptyState
   538→          icon={Activity}
   539→          title="Unable to load dashboard"
   540→          description={error}
   541→        />
   542→      </div>
   543→    );
   544→  }
   545→
   546→  // ---- Render ----
   547→
   548→  return (
   549→    <div className="space-y-6">
   550→      {/* Page Header */}
   551→      <PageHeader
   552→        title="Dashboard"
   553→        description="Overview of your business operations"
   554→        icon={TrendingUp}
   555→      />
   556→
   557→      {/* Section 1: KPI Cards */}
   558→      {loading ? (
   559→        <KpiSkeletons />
   560→      ) : (
   561→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-6 gap-4">
   562→          <StatCard
   563→            label="Total Revenue"
   564→            value={formatCurrency(kpis!.totalRevenue)}
   565→            change={kpis!.revenueChange}
   566→            changeLabel="vs last month"
   567→            icon={DollarSign}
   568→          />
   569→          <StatCard
   570→            label="Active Deals"
   571→            value={kpis!.activeDealsCount}
   572→            change={kpis!.dealsChange}
   573→            changeLabel="vs last month"
   574→            icon={Handshake}
   575→          />
   576→          <StatCard
   577→            label="Open Tickets"
   578→            value={kpis!.openTicketsCount}
   579→            change={kpis!.ticketsChange}
   580→            changeLabel="vs last month"
   581→            icon={TicketCheck}
   582→          />
   583→          <StatCard
   584→            label="Active Projects"
   585→            value={kpis!.activeProjectsCount}
   586→            change={kpis!.projectsChange}
   587→            changeLabel="vs last month"
   588→            icon={FolderKanban}
   589→          />
   590→          <StatCard
   591→            label="Tasks Due This Week"
   592→            value={kpis!.tasksDueThisWeek}
   593→            icon={ListTodo}
   594→          />
   595→          <StatCard
   596→            label="New Leads"
   597→            value={kpis!.newLeadsThisMonth}
   598→            change={kpis!.leadsChange}
   599→            changeLabel="vs last month"
   600→            icon={UserPlus}
   601→          />
   602→        </div>
   603→      )}
   604→
   605→      {/* Section 2: Revenue Chart */}
   606→      {loading ? (
   607→        <ChartSkeleton />
   608→      ) : revenueChartData.length === 0 ? (
   609→        <Card>
   610→          <CardHeader>
   611→            <CardTitle>Revenue Overview</CardTitle>
   612→          </CardHeader>
   613→          <CardContent>
   614→            <EmptyState
   615→              icon={DollarSign}
   616→              title="No revenue data yet"
   617→              description="Paid invoices will appear as monthly revenue here."
   618→            />
   619→          </CardContent>
   620→        </Card>
   621→      ) : (
   622→        <Card>
   623→          <CardHeader>
   624→            <CardTitle>Revenue Overview</CardTitle>
   625→          </CardHeader>
   626→          <CardContent>
   627→            <div className="h-72">
   628→              <ResponsiveContainer width="100%" height="100%">
   629→                <AreaChart data={revenueChartData} margin={{ top: 5, right: 20, left: 10, bottom: 5 }}>
   630→                  <defs>
   631→                    <linearGradient id="revenueGradient" x1="0" y1="0" x2="0" y2="1">
   632→                      <stop offset="0%" stopColor="#10b981" stopOpacity={0.3} />
