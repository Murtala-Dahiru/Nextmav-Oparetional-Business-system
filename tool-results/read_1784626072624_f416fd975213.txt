    73→    73→async function apiFetch<T>(url: string, init?: RequestInit): Promise<T> {
    74→    74→  const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...init });
    75→    75→  const json = await res.json();
    76→    76→  if (!res.ok) throw new Error(json.error?.message ?? 'Request failed');
    77→    77→  return json;
    78→    78→}

   243→   243→//  Main Module
   244→   244→// ═══════════════════════════════════════════════════════════════
   245→   245→
   246→   246→export default function InventoryModule() {
   247→   247→  // ── Products State ──
   248→   248→  const [products, setProducts] = useState<Product[]>([]);
   249→   249→  const [productMeta, setProductMeta] = useState<ApiMeta>({ total: 0, page: 1, pageSize: 20, totalPages: 0 });
   250→   250→  const [productLoading, setProductLoading] = useState(true);
   251→   251→  const [productPage, setProductPage] = useState(0);
   252→   252→  const [productPageSize, setProductPageSize] = useState(20);
   253→   253→  const [productSearch, setProductSearch] = useState('');
   254→   254→  const [productFilters, setProductFilters] = useState<ColumnFiltersState>([]);
   255→   255→  const [productSorting, setProductSorting] = useState<SortingState>([]);
   256→   256→
   257→   257→  // ── Warehouses State ──
   258→   258→  const [warehouses, setWarehouses] = useState<Warehouse[]>([]);
   259→   259→  const [warehouseLoading, setWarehouseLoading] = useState(false);
   260→   260→  const [warehouseProductCounts, setWarehouseProductCounts] = useState<Record<string, number>>({});
   261→   261→
   262→   262→  // ── Dialogs ──
   263→   263→  const [productDialogOpen, setProductDialogOpen] = useState(false);
   264→   264→  const [editingProduct, setEditingProduct] = useState<Product | null>(null);
   265→   265→  const [productSubmitting, setProductSubmitting] = useState(false);
   266→   266→
   267→   267→  const [warehouseDialogOpen, setWarehouseDialogOpen] = useState(false);
   268→   268→  const [editingWarehouse, setEditingWarehouse] = useState<Warehouse | null>(null);
   269→   269→  const [warehouseSubmitting, setWarehouseSubmitting] = useState(false);
   270→   270→
   271→   271→  const [deleteTarget, setDeleteTarget] = useState<{ type: 'product' | 'warehouse'; id: string; name: string } | null>(null);
   272→   272→  const [deleting, setDeleting] = useState(false);
   273→   273→
   274→   274→  const [stats, setStats] = useState<Stats>({ totalProducts: 0, totalValue: 0, lowStock: 0 });
   275→   275→
   276→   276→  // ── Fetch products ──
   277→   277→  const fetchProducts = useCallback(async () => {
   278→   278→    setProductLoading(true);
   279→   279→    try {
   280→   280→      const params = new URLSearchParams({
   281→   281→        page: String(productPage + 1),
   282→   282→        pageSize: String(productPageSize),
   283→   283→        sort: productSorting[0]?.id || 'createdAt',
   284→   284→        sortDir: productSorting[0]?.desc ? 'desc' : 'asc',
   285→   285→      });
   286→   286→      if (productSearch) params.set('search', productSearch);
   287→   287→      productFilters.forEach((f) => { params.set(f.id, String(f.value)); });
   288→   288→
   289→   289→      const res = await apiFetch<{ data: Product[]; meta: ApiMeta }>(`/api/inventory/products?${params}`);
   290→   290→      setProducts(res.data);
   291→   291→      setProductMeta(res.meta);
   292→   292→    } catch (e: any) {
   293→   293→      toast.error(e.message);
   294→   294→    } finally {
   295→   295→      setProductLoading(false);
   296→   296→    }
   297→   297→  }, [productPage, productPageSize, productSearch, productFilters, productSorting]);
   298→   298→
   299→   299→  // ── Fetch stats ──
   300→   300→  const fetchStats = useCallback(async () => {
   301→   301→    try {
   302→   302→      const res = await apiFetch<{ data: Product[]; meta: ApiMeta }>('/api/inventory/products?pageSize=9999');
   303→   303→      const all = res.data;
   304→   304→      const totalProducts = all.length;
   305→   305→      const totalValue = all.reduce((s, p) => s + p.price * p.stock, 0);
   306→   306→      const lowStock = all.filter((p) => p.stock < p.reorderLevel).length;
   307→   307→      setStats({ totalProducts, totalValue, lowStock });
   308→   308→    } catch { /* ignore */ }
   309→   309→  }, []);
   310→   310→
   311→   311→  // ── Fetch warehouses ──
   312→   312→  const fetchWarehouses = useCallback(async () => {
   313→   313→    setWarehouseLoading(true);
   314→   314→    try {
   315→   315→      const res = await apiFetch<{ data: Warehouse[]; meta: ApiMeta }>('/api/inventory/warehouses?pageSize=100');
   316→   316→      setWarehouses(res.data);
   317→   317→
   318→   318→      const allProducts = await apiFetch<{ data: Product[]; meta: ApiMeta }>('/api/inventory/products?pageSize=9999');
   319→   319→      const counts: Record<string, number> = {};
   320→   320→      allProducts.data.forEach((p) => {
   321→   321→        if (p.warehouseId) counts[p.warehouseId] = (counts[p.warehouseId] || 0) + 1;
   322→   322→      });
   323→   323→      setWarehouseProductCounts(counts);
   324→   324→    } catch (e: any) {
   325→   325→      toast.error(e.message);
   326→   326→    } finally {
   327→   327→      setWarehouseLoading(false);
   328→   328→    }
   329→   329→  }, []);
   330→   330→
   331→   331→  useEffect(() => { fetchProducts(); }, [fetchProducts]);
   332→   332→  useEffect(() => { fetchStats(); }, [fetchStats]);
   333→   333→
   334→   334→  // ── Product CRUD ──
   335→   335→  const handleProductSubmit = async (form: ProductFormState) => {
   336→   336→    setProductSubmitting(true);
   337→   337→    try {
   338→   338→      const payload = {
   339→   339→        name: form.name, sku: form.sku, category: form.category,
   340→   340→        price: Number(form.price) || 0, cost: Number(form.cost) || 0,
   341→   341→        stock: Number(form.stock) || 0, unit: form.unit,
   342→   342→        reorderLevel: Number(form.reorderLevel) || 0, isActive: form.isActive,
   343→   343→      };
   344→   344→      if (editingProduct) {
   345→   345→        await apiFetch(`/api/inventory/products/${editingProduct.id}`, { method: 'PUT', body: JSON.stringify(payload) });
   346→   346→        toast.success('Product updated');
   347→   347→      } else {
   348→   348→        await apiFetch('/api/inventory/products', { method: 'POST', body: JSON.stringify(payload) });
   349→   349→        toast.success('Product created');
   350→   350→      }
   351→   351→      setProductDialogOpen(false);
   352→   352→      setEditingProduct(null);
   353→   353→      fetchProducts();
   354→   354→      fetchStats();
   355→   355→    } catch (e: any) {
   356→   356→      toast.error(e.message);
   357→   357→    } finally {
   358→   358→      setProductSubmitting(false);
   359→   359→    }
   360→   360→  };
   361→   361→
   362→   362→  const handleDelete = async () => {
   363→   363→    if (!deleteTarget) return;
   364→   364→    setDeleting(true);
   365→   365→    try {
   366→   366→      const endpoint = deleteTarget.type === 'product'
   367→   367→        ? `/api/inventory/products/${deleteTarget.id}`
   368→   368→        : `/api/inventory/warehouses/${deleteTarget.id}`;
   369→   369→      await apiFetch(endpoint, { method: 'DELETE' });
   370→   370→      toast.success(`${deleteTarget.type === 'product' ? 'Product' : 'Warehouse'} deleted`);
   371→   371→      setDeleteTarget(null);
   372→   372→      if (deleteTarget.type === 'product') { fetchProducts(); fetchStats(); } else { fetchWarehouses(); }
   373→   373→    } catch (e: any) {
   374→   374→      toast.error(e.message);
   375→   375→    } finally {
   376→   376→      setDeleting(false);
   377→   377→    }
   378→   378→  };
   379→   379→
   380→   380→  // ── Warehouse CRUD ──
   381→   381→  const handleWarehouseSubmit = async (form: WarehouseFormState) => {
   382→   382→    setWarehouseSubmitting(true);
   383→   383→    try {
   384→   384→      const payload = {
   385→   385→        name: form.name, location: form.location,
   386→   386→        capacity: Number(form.capacity) || 0, isActive: form.isActive,
   387→   387→      };
   388→   388→      if (editingWarehouse) {
   389→   389→        await apiFetch(`/api/inventory/warehouses/${editingWarehouse.id}`, { method: 'PUT', body: JSON.stringify(payload) });
   390→   390→        toast.success('Warehouse updated');
   391→   391→      } else {
   392→   392→        await apiFetch('/api/inventory/warehouses', { method: 'POST', body: JSON.stringify(payload) });
   393→   393→        toast.success('Warehouse created');
   394→   394→      }
   395→   395→      setWarehouseDialogOpen(false);
   396→   396→      setEditingWarehouse(null);
   397→   397→      fetchWarehouses();
   398→   398→    } catch (e: any) {
   399→   399→      toast.error(e.message);
