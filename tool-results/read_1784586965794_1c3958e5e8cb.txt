   241→function ChartTooltipContent({
   242→  active,
   243→  payload,
   244→  label,
   245→  formatter,
   246→}: {
   247→  active?: boolean;
   248→  payload?: Array<{ value: number; name: string; dataKey: string; color?: string }>;
   249→  label?: string;
   250→  formatter?: (value: number) => string;
   251→}) {
   252→  if (!active || !payload?.length) return null;
   253→  return (
   254→    <div className="border-border/50 bg-background rounded-lg border px-3 py-2 text-xs shadow-xl">
   255→      {label && <p className="font-medium text-foreground mb-1">{label}</p>}
   256→      {payload.map((item, idx) => (
   257→        <div key={idx} className="flex items-center gap-2">
   258→          <span
   259→            className="inline-block h-2 w-2 rounded-full"
   260→            style={{ backgroundColor: item.color ?? CHART_COLORS[0] }}
   261→          />
   262→          <span className="text-muted-foreground">
   263→            {item.name}: {formatter ? formatter(item.value) : item.value.toLocaleString()}
   264→          </span>
   265→        </div>
   266→      ))}
   267→    </div>
   268→  );
   269→}

   275→export default function DashboardModule() {
   276→  const { setActiveModule } = useAppStore();
   277→  const [loading, setLoading] = useState(true);
   278→  const [data, setData] = useState<DashboardData | null>(null);
   279→  const [error, setError] = useState<string | null>(null);
   280→
   281→  // Fetch all data on mount
   282→  useEffect(() => {
   283→    let cancelled = false;
   284→
   285→    async function fetchAll() {
   286→      try {
   287→        const endpoints = [
   288→          '/api/crm/leads?pageSize=100',
   289→          '/api/crm/deals?pageSize=100',
   290→          '/api/projects/projects?pageSize=100',
   291→          '/api/projects/tasks?pageSize=100',
   292→          '/api/support/tickets?pageSize=100',
   293→          '/api/finance/invoices?pageSize=100',
   294→          '/api/finance/expenses?pageSize=100',
   295→          '/api/activity-log?pageSize=10',
   296→        ];
   297→
   298→        const results = await Promise.all(
   299→          endpoints.map((url) =>
   300→            fetch(url).then((r) => {
   301→              if (!r.ok) throw new Error(`Failed to fetch ${url}`);
   302→              return r.json();
   303→            }),
   304→          ),
   305→        );
   306→
   307→        if (cancelled) return;
   308→
   309→        setData({
   310→          leads: (results[0] as ApiResponse<Lead>).data,
   311→          deals: (results[1] as ApiResponse<Deal>).data,
   312→          projects: (results[2] as ApiResponse<Project>).data,
   313→          tasks: (results[3] as ApiResponse<Task>).data,
   314→          tickets: (results[4] as ApiResponse<Ticket>).data,
   315→          invoices: (results[5] as ApiResponse<Invoice>).data,
   316→          expenses: (results[6] as ApiResponse<Expense>).data,
   317→          activities: (results[7] as ApiResponse<ActivityItem>).data,
   318→        });
   319→      } catch (e: unknown) {
   320→        if (!cancelled) {
   321→          setError(e instanceof Error ? e.message : 'Failed to load dashboard data');
   322→        }
   323→      } finally {
   324→        if (!cancelled) setLoading(false);
   325→      }
   326→    }
   327→
   328→    fetchAll();
   329→    return () => {
   330→      cancelled = true;
   331→    };
   332→  }, []);
   333→
   334→  // ---- Computed KPIs ----
   335→
   336→  const kpis = useMemo(() => {
   337→    if (!data) return null;
   338→
   339→    // Total Revenue: sum of paid invoices
   340→    const paidInvoices = data.invoices.filter((inv) => inv.status === 'paid');
   341→    const totalRevenue = paidInvoices.reduce((sum, inv) => sum + inv.total, 0);
   342→
   343→    // Revenue change: compare this month vs last month paid invoices
   344→    const now = new Date();
   345→    const thisMonthStart = new Date(now.getFullYear(), now.getMonth(), 1);
   346→    const lastMonthStart = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   347→
   348→    const thisMonthRevenue = paidInvoices
   349→      .filter((inv) => inv.paidAt && new Date(inv.paidAt) >= thisMonthStart)
   350→      .reduce((sum, inv) => sum + inv.total, 0);
   351→
   352→    const lastMonthRevenue = paidInvoices
   353→      .filter(
   354→        (inv) =>
   355→          inv.paidAt &&
   356→          new Date(inv.paidAt) >= lastMonthStart &&
   357→          new Date(inv.paidAt) < thisMonthStart,
   358→      )
   359→      .reduce((sum, inv) => sum + inv.total, 0);
   360→
   361→    const revenueChange =
   362→      lastMonthRevenue > 0
   363→        ? ((thisMonthRevenue - lastMonthRevenue) / lastMonthRevenue) * 100
   364→        : thisMonthRevenue > 0
   365→          ? 100
   366→          : 0;
   367→
   368→    // Active Deals: not closed-won/closed-lost
   369→    const activeDeals = data.deals.filter(
   370→      (d) => d.stage !== 'closed-won' && d.stage !== 'closed-lost',
   371→    );
   372→    const activeDealsCount = activeDeals.length;
   373→
   374→    // Deals change: this month vs last month created
   375→    const thisMonthDeals = data.deals.filter(
   376→      (d) => new Date(d.createdAt) >= thisMonthStart,
   377→    ).length;
   378→    const lastMonthDeals = data.deals.filter(
   379→      (d) => new Date(d.createdAt) >= lastMonthStart && new Date(d.createdAt) < thisMonthStart,
   380→    ).length;
   381→    const dealsChange =
   382→      lastMonthDeals > 0
   383→        ? ((thisMonthDeals - lastMonthDeals) / lastMonthDeals) * 100
   384→        : thisMonthDeals > 0
   385→          ? 100
   386→          : 0;
   387→
   388→    // Open Tickets: open / in-progress / pending
   389→    const openTickets = data.tickets.filter(
   390→      (t) => t.status === 'open' || t.status === 'in-progress' || t.status === 'pending',
   391→    );
   392→    const openTicketsCount = openTickets.length;
   393→
   394→    // Tickets change: this month vs last month
   395→    const thisMonthTickets = data.tickets.filter(
   396→      (t) => new Date(t.createdAt) >= thisMonthStart,
   397→    ).length;
   398→    const lastMonthTickets = data.tickets.filter(
   399→      (t) => new Date(t.createdAt) >= lastMonthStart && new Date(t.createdAt) < thisMonthStart,
   400→    ).length;
   401→    const ticketsChange =
   402→      lastMonthTickets > 0
   403→        ? ((thisMonthTickets - lastMonthTickets) / lastMonthTickets) * 100
   404→        : thisMonthTickets > 0
   405→          ? 100
   406→          : 0;
   407→
   408→    // Active Projects
   409→    const activeProjects = data.projects.filter((p) => p.status === 'active');
   410→    const activeProjectsCount = activeProjects.length;
   411→
   412→    const thisMonthProjects = data.projects.filter(
   413→      (p) => new Date(p.createdAt) >= thisMonthStart,
   414→    ).length;
   415→    const lastMonthProjects = data.projects.filter(
   416→      (p) => new Date(p.createdAt) >= lastMonthStart && new Date(p.createdAt) < thisMonthStart,
   417→    ).length;
   418→    const projectsChange =
   419→      lastMonthProjects > 0
   420→        ? ((thisMonthProjects - lastMonthProjects) / lastMonthProjects) * 100
   421→        : thisMonthProjects > 0
   422→          ? 100
   423→          : 0;
   424→
   425→    // Tasks Due This Week
   426→    const weekFromNow = new Date();
   427→    weekFromNow.setDate(weekFromNow.getDate() + 7);
   428→    weekFromNow.setHours(23, 59, 59, 999);
   429→    const todayStart = new Date();
   430→    todayStart.setHours(0, 0, 0, 0);
   431→    const tasksDueThisWeek = data.tasks.filter(
   432→      (t) =>
   433→        t.dueDate &&
   434→        t.status !== 'done' &&
   435→        new Date(t.dueDate) >= todayStart &&
   436→        new Date(t.dueDate) <= weekFromNow,
   437→    ).length;
   438→
   439→    // New Leads This Month
   440→    const thirtyDaysAgo = new Date();
   441→    thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);
   442→    const newLeadsThisMonth = data.leads.filter(
   443→      (l) => new Date(l.createdAt) >= thirtyDaysAgo,
   444→    ).length;
   445→
   446→    const thisMonthLeads = data.leads.filter(
   447→      (l) => new Date(l.createdAt) >= thisMonthStart,
   448→    ).length;
   449→    const lastMonthLeads = data.leads.filter(
   450→      (l) => new Date(l.createdAt) >= lastMonthStart && new Date(l.createdAt) < thisMonthStart,
   451→    ).length;
   452→    const leadsChange =
   453→      lastMonthLeads > 0
   454→        ? ((thisMonthLeads - lastMonthLeads) / lastMonthLeads) * 100
   455→        : thisMonthLeads > 0
   456→          ? 100
   457→          : 0;
   458→
   459→    return {
   460→      totalRevenue,
   461→      revenueChange,
   462→      activeDealsCount,
   463→      dealsChange,
   464→      openTicketsCount,
   465→      ticketsChange,
   466→      activeProjectsCount,
   467→      projectsChange,
   468→      tasksDueThisWeek,
   469→      newLeadsThisMonth,
   470→      leadsChange,
   471→    };
   472→  }, [data]);
   473→
   474→  // ---- Chart: Monthly Revenue ----
   475→
   476→  const revenueChartData = useMemo(() => {
   477→    if (!data) return [];
   478→
   479→    const paidInvoices = data.invoices.filter((inv) => inv.status === 'paid' && inv.paidAt);
   480→
   481→    // Group by month (last 6 months)
   482→    const monthMap = new Map<string, number>();
   483→    const now = new Date();
   484→
   485→    for (let i = 5; i >= 0; i--) {
   486→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   487→      const key = d.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   488→      monthMap.set(key, 0);
   489→    }
   490→
   491→    paidInvoices.forEach((inv) => {
   492→      const d = new Date(inv.paidAt!);
   493→      const key = d.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   494→      if (monthMap.has(key)) {
   495→        monthMap.set(key, monthMap.get(key)! + inv.total);
   496→      }
   497→    });
   498→
   499→    return Array.from(monthMap.entries()).map(([month, revenue]) => ({ month, revenue }));
   500→  }, [data]);
   501→
   502→  // ---- Chart: Deal Pipeline ----
   503→
   504→  const pipelineChartData = useMemo(() => {
   505→    if (!data) return [];
   506→
   507→    const stageMap = new Map<string, { count: number; value: number }>();
   508→
   509→    DEAL_STAGE_ORDER.forEach((stage) => {
   510→      stageMap.set(stage, { count: 0, value: 0 });
   511→    });
   512→
   513→    data.deals.forEach((deal) => {
   514→      const entry = stageMap.get(deal.stage);
   515→      if (entry) {
   516→        entry.count += 1;
   517→        entry.value += deal.value;
   518→      }
   519→    });
   520→
   521→    return Array.from(stageMap.entries())
   522→      .filter(([, v]) => v.count > 0)
   523→      .map(([stage, v]) => ({
   524→        stage: stage.replace(/-/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase()),
   525→        count: v.count,
   526→        value: v.value,
   527→      }));
   528→  }, [data]);
   529→
   530→  // ---- Chart: Lead Status Distribution ----
   531→
   532→  const leadStatusData = useMemo(() => {
   533→    if (!data) return [];
   534→
   535→    const statusMap = new Map<string, number>();
   536→    data.leads.forEach((lead) => {
   537→      statusMap.set(lead.status, (statusMap.get(lead.status) ?? 0) + 1);
   538→    });
   539→
   540→    return Array.from(statusMap.entries())
   541→      .map(([status, count]) => ({
   542→        name: status.replace(/-/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase()),
   543→        value: count,
   544→        color: LEAD_STATUS_COLORS[status] ?? '#94a3b8',
   545→      }))
   546→      .sort((a, b) => b.value - a.value);
   547→  }, [data]);
   548→
   549→  // ---- Top Deals ----
   550→
   551→  const topDeals = useMemo(() => {
   552→    if (!data) return [];
   553→    return [...data.deals]
   554→      .sort((a, b) => b.value - a.value)
   555→      .slice(0, 5);
   556→  }, [data]);
   557→
   558→  // ---- Quick Action Handler ----
   559→
   560→  const handleQuickAction = useCallback(
   561→    (module: ModuleId) => {
   562→      setActiveModule(module);
   563→    },
   564→    [setActiveModule],
   565→  );
   566→
   567→  // ---- Module navigation from activity ----
   568→
   569→  const handleActivityClick = useCallback(
   570→    (moduleName: string) => {
   571→      const moduleMap: Record<string, ModuleId> = {
   572→        crm: 'crm',
   573→        projects: 'projects',
   574→        support: 'support',
   575→        finance: 'finance',
   576→        hr: 'hr',
   577→        inventory: 'inventory',
   578→        calendar: 'calendar',
   579→        communication: 'communication',
   580→        admin: 'admin',
   581→        workspace: 'workspace',
   582→      };
   583→      if (moduleMap[moduleName]) {
   584→        setActiveModule(moduleMap[moduleName]);
   585→      }
   586→    },
   587→    [setActiveModule],
   588→  );
   589→
   590→  // ---- Error State ----
   591→
   592→  if (error && !data) {
   593→    return (
   594→      <div className="space-y-6">
   595→        <PageHeader
   596→          title="Dashboard"
   597→          description="Overview of your business operations"
   598→          icon={TrendingUp}
   599→        />
   600→        <EmptyState
   601→          icon={Activity}
   602→          title="Unable to load dashboard"
   603→          description={error}
   604→        />
   605→      </div>
   606→    );
   607→  }
   608→
   609→  // ---- Render ----
   610→
   611→  return (
   612→    <div className="space-y-6">
   613→      {/* Page Header */}
   614→      <PageHeader
   615→        title="Dashboard"
   616→        description="Overview of your business operations"
   617→        icon={TrendingUp}
   618→      />
   619→
   620→      {/* Section 1: KPI Cards */}
   621→      {loading ? (
   622→        <KpiSkeletons />
   623→      ) : (
   624→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-6 gap-4">
   625→          <StatCard
   626→            label="Total Revenue"
   627→            value={formatCurrency(kpis!.totalRevenue)}
   628→            change={kpis!.revenueChange}
   629→            changeLabel="vs last month"
   630→            icon={DollarSign}
   631→          />
   632→          <StatCard
   633→            label="Active Deals"
   634→            value={kpis!.activeDealsCount}
   635→            change={kpis!.dealsChange}
