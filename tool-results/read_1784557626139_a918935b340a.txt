    35→function formatDateTime(d: string) {
    36→  return new Date(d).toLocaleString('en-US', {
    37→    month: 'short', day: 'numeric', hour: 'numeric', minute: '2-digit',
    38→  });
    39→}
    40→
    41→function timeAgo(d: string) {
    42→  const diff = Date.now() - new Date(d).getTime();
    43→  const mins = Math.floor(diff / 60000);
    44→  if (mins < 60) return `${mins}m ago`;
    45→  const hrs = Math.floor(mins / 60);
    46→  if (hrs < 24) return `${hrs}h ago`;
    47→  const days = Math.floor(hrs / 24);
    48→  return `${days}d ago`;
    49→}
