   237→function ListViewTab({ tasks }: { tasks: ProjectTaskItem[] }) {
   238→  const [search, setSearch] = useState('');
   239→  const [sortKey, setSortKey] = useState<'priority' | 'dueDate' | 'title'>('priority');
   240→  const priorityOrder: Record<string, number> = { critical: 0, high: 1, medium: 2, low: 3 };
   241→
   242→  const filtered = useMemo(() => {
   243→    let result = tasks;
   244→    if (search) {
   245→      const q = search.toLowerCase();
   246→      result = result.filter(
   247→        (t) =>
   248→          t.title.toLowerCase().includes(q) ||
   249→          t.projectName.toLowerCase().includes(q) ||
   250→          t.assigneeName.toLowerCase().includes(q)
   251→      );
   252→    }
   253→    return [...result].sort((a, b) => {
   254→      if (sortKey === 'priority') return (priorityOrder[a.priority] ?? 9) - (priorityOrder[b.priority] ?? 9);
   255→      if (sortKey === 'dueDate') return new Date(a.dueDate).getTime() - new Date(b.dueDate).getTime();
   256→      return a.title.localeCompare(b.title);
   257→    });

   261→    <div>

   420→/* ---- Tab 4: Timeline ---- */
   421→
   422→function TimelineTab({ items }: { items: ProjectItem[] }) {
   423→  const allDates = useMemo(() => {
   424→    const starts = items.map((p) => new Date(p.startDate).getTime());
   425→    const ends = items.map((p) => new Date(p.endDate).getTime());
   426→    return { min: Math.min(...starts), max: Math.max(...ends) };
   427→  }, [items]);
   428→
   429→  const totalSpan = allDates.max - allDates.min;
   430→
   431→  function getLeft(startDate: string) {
   432→    const ms = new Date(startDate).getTime() - allDates.min;
   433→    return totalSpan > 0 ? (ms / totalSpan) * 100 : 0;
   434→  }
   435→
   436→  function getWidth(startDate: string, endDate: string) {
   437→    const ms = new Date(endDate).getTime() - new Date(startDate).getTime();
   438→    return totalSpan > 0 ? Math.max((ms / totalSpan) * 100, 3) : 3;
   439→  }
   440→
   441→  const months = useMemo(() => {
   442→    const start = new Date(allDates.min);
   443→    const end = new Date(allDates.max);
   444→    const result: string[] = [];
   445→    const current = new Date(start.getFullYear(), start.getMonth(), 1);
   446→    while (current <= end) {
   447→      result.push(current.toLocaleDateString('en-US', { month: 'short', year: '2-digit' }));
   448→      current.setMonth(current.getMonth() + 1);
   449→    }
   450→    return result;
   451→  }, [allDates]);
   452→
   453→  return (
   454→    <div>
   455→      <TabHeader title="Timeline" description="Visualize project schedules and progress on a Gantt-like timeline" actionLabel="Add Project" />
   456→      <div className="bg-card rounded-xl border border-border overflow-hidden">
   457→        {/* Month header */}
   458→        <div className="flex border-b border-border">
   459→          <div className="w-52 shrink-0 px-4 py-3 border-r border-border">
   460→            <span className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Project</span>
   461→          </div>
   462→          <div className="flex-1 flex relative">
   463→            {months.map((m, i) => {
   464→              const monthStart = new Date(allDates.min);
   465→              monthStart.setMonth(monthStart.getMonth() + i);
   466→              const monthStartMs = monthStart.getTime() - allDates.min;
   467→              const nextMonth = new Date(monthStart);
   468→              nextMonth.setMonth(nextMonth.getMonth() + 1);
   469→              const nextMonthMs = Math.min(nextMonth.getTime() - allDates.min, totalSpan);
   470→              const widthPct = totalSpan > 0 ? ((nextMonthMs - monthStartMs) / totalSpan) * 100 : 0;
   471→              return (
   472→                <div
   473→                  key={m}
   474→                  className="flex items-center justify-center text-[11px] text-muted-foreground font-medium border-r border-border last:border-r-0"
   475→                  style={{ width: `${widthPct}%` }}
   476→                >
   477→                  {m}
   478→                </div>
   479→              );
   480→            })}
   481→          </div>
   482→        </div>
   483→        {/* Rows */}
   484→        {items.map((project, i) => (
   485→          <div
   486→            key={project.id}
   487→            className={cn('flex items-center border-b border-border last:border-b-0', i % 2 === 0 ? 'bg-background' : 'bg-muted/20')}
   488→          >
   489→            <div className="w-52 shrink-0 px-4 py-3.5 border-r border-border">
   490→              <div className="flex items-center gap-2">
   491→                <ProjectStatusBadge status={project.status} />
   492→                <span className="text-sm font-medium text-foreground truncate">{project.name}</span>
   493→              </div>
   494→            </div>
   495→            <div className="flex-1 py-3.5 px-2 relative" style={{ minHeight: '44px' }}>
   496→              <div
   497→                className="relative h-7 rounded-md overflow-hidden"
   498→                style={{
   499→                  position: 'absolute',
   500→                  left: `${getLeft(project.startDate)}%`,
   501→                  width: `${getWidth(project.startDate, project.endDate)}%`,
   502→                  top: '50%',
   503→                  transform: 'translateY(-50%)',
   504→                }}
   505→              >
   506→                {/* Background bar */}
   507→                <div className={cn('absolute inset-0 rounded-md opacity-20', projectBarColors[project.status] ?? 'bg-gray-400')} />
   508→                {/* Progress fill */}
   509→                <div
   510→                  className={cn('absolute inset-y-0 left-0 rounded-md', projectBarColors[project.status] ?? 'bg-gray-400')}
   511→                  style={{ width: `${project.progress}%` }}
   512→                />
   513→                {/* Label */}
   514→                <span className="absolute inset-0 flex items-center px-2 text-[10px] font-semibold text-foreground truncate">
   515→                  {project.progress}% — {formatDate(project.startDate).replace(/,\s*\d{4}$/, '')} to {formatDate(project.endDate).replace(/,\s*\d{4}$/, '')}
   516→                </span>
   517→              </div>
   518→            </div>
   519→          </div>
   520→        ))}
   521→      </div>
   522→    </div>
   523→  );
   524→}
