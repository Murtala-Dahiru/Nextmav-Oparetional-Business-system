   133→   133→
   134→   134→async function fetchList<T>(url: string): Promise<{ data: T[]; meta: ApiMeta }> {
   135→   135→  const res = await fetch(url);
   136→   136→  if (!res.ok) {
   137→   137→    const e = await res.json().catch(() => ({ message: 'Request failed' }));
   138→   138→    throw new Error((e as any).message || `Error ${res.status}`);
   139→   139→  }
   140→   140→  return res.json();
   141→   141→}
   142→   142→
   143→   143→async function createRecord<T>(url: string, data: unknown): Promise<T> {
   144→   144→  const res = await fetch(url, {
   145→   145→    method: 'POST',
   146→   146→    headers: { 'Content-Type': 'application/json' },
   147→   147→    body: JSON.stringify(data),
   148→   148→  });
   149→   149→  if (!res.ok) {
   150→   150→    const e = await res.json().catch(() => ({ message: 'Create failed' }));
   151→   151→    throw new Error((e as any).message || 'Create failed');
   152→   152→  }
   153→   153→  const json = await res.json();
   154→   154→  return json.data;
   155→   155→}

   210→   210→function OverviewTab({ onSwitchTab }: { onSwitchTab: (tab: string) => void }) {
   211→   211→  const [invoices, setInvoices] = useState<InvoiceRecord[]>([]);
   212→   212→  const [expenses, setExpenses] = useState<ExpenseRecord[]>([]);
   213→   213→  const [loading, setLoading] = useState(true);
   214→   214→
   215→   215→  useEffect(() => {
   216→   216→    (async () => {
   217→   217→      try {
   218→   218→        const [invRes, expRes] = await Promise.all([
   219→   219→          fetchList<InvoiceRecord>('/api/finance/invoices?pageSize=1000'),
   220→   220→          fetchList<ExpenseRecord>('/api/finance/expenses?pageSize=1000'),
   221→   221→        ]);
   222→   222→        setInvoices(invRes.data);
   223→   223→        setExpenses(expRes.data);
   224→   224→      } catch (e: any) {
   225→   225→        toast.error(e.message || 'Failed to load finance data');
   226→   226→      } finally {
   227→   227→        setLoading(false);
   228→   228→      }
   229→   229→    })();
   230→   230→  }, []);
   231→   231→
   232→   232→  // Computed stats
   233→   233→  const totalRevenue = useMemo(() => invoices.filter((i) => i.status === 'paid').reduce((s, i) => s + i.total, 0), [invoices]);
   234→   234→  const outstanding = useMemo(() => invoices.filter((i) => i.status === 'sent' || i.status === 'overdue').reduce((s, i) => s + i.total, 0), [invoices]);
   235→   235→  const totalExpenses = useMemo(() => expenses.reduce((s, e) => s + e.amount, 0), [expenses]);
   236→   236→  const netIncome = totalRevenue - totalExpenses;
   237→   237→
   238→   238→  // Revenue by month (AreaChart data)
   239→   239→  const revenueByMonth = useMemo(() => {
   240→   240→    const monthMap: Record<string, number> = {};
   241→   241→    invoices.forEach((inv) => {
   242→   242→      if (inv.status !== 'paid') return;
   243→   243→      const d = new Date(inv.createdAt);
   244→   244→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   245→   245→      monthMap[key] = (monthMap[key] || 0) + inv.total;
   246→   246→    });
   247→   247→    return Object.entries(monthMap)
   248→   248→      .sort(([a], [b]) => a.localeCompare(b))
   249→   249→      .map(([key, revenue]) => {
   250→   250→        const [y, m] = key.split('-');
   251→   251→        return { month: `${MONTH_NAMES[parseInt(m) - 1]} ${y}`, revenue };
   252→   252→      });
   253→   253→  }, [invoices]);
   254→   254→
   255→   255→  // Expense by category (PieChart data)
   256→   256→  const expenseByCategory = useMemo(() => {
   257→   257→    const catMap: Record<string, number> = {};
   258→   258→    expenses.forEach((exp) => {
   259→   259→      catMap[exp.category] = (catMap[exp.category] || 0) + exp.amount;
   260→   260→    });
   261→   261→    return Object.entries(catMap).map(([name, value]) => ({ name, value }));
   262→   262→  }, [expenses]);
   263→   263→
   264→   264→  // Recent invoices (top 5 by date)
   265→   265→  const recentInvoices = useMemo(
   266→   266→    () => [...invoices].sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()).slice(0, 5),
   267→   267→    [invoices],
   268→   268→  );
   269→   269→
   270→   270→  if (loading) {
   271→   271→    return (
   272→   272→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   273→   273→        {[0, 1, 2, 3].map((i) => (
   274→   274→          <Card key={i}><CardContent className="p-4"><div className="h-20 animate-pulse bg-muted rounded" /></CardContent></Card>
   275→   275→        ))}
   276→   276→      </div>
   277→   277→    );
   278→   278→  }
   279→   279→
   280→   280→  return (
   281→   281→    <div className="flex flex-col gap-6">
   282→   282→      <PageHeader title="Finance" icon={DollarSign} />
   283→   283→
   284→   284→      {/* Stat Cards */}
   285→   285→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   286→   286→        <StatCard label="Total Revenue" value={formatCurrency(totalRevenue)} icon={TrendingUp} />
   287→   287→        <StatCard label="Outstanding" value={formatCurrency(outstanding)} icon={Wallet} />
   288→   288→        <StatCard label="Total Expenses" value={formatCurrency(totalExpenses)} icon={Receipt} />
   289→   289→        <StatCard label="Net Income" value={formatCurrency(netIncome)} icon={netIncome >= 0 ? TrendingUp : TrendingDown} />
   290→   290→      </div>
   291→   291→
   292→   292→      {/* Charts */}
   293→   293→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   294→   294→        {/* Revenue by month */}
   295→   295→        <Card>
   296→   296→          <CardHeader className="pb-2">
   297→   297→            <CardTitle className="text-base font-semibold">Revenue by Month</CardTitle>
   298→   298→          </CardHeader>
   299→   299→          <CardContent>
   300→   300→            {revenueByMonth.length === 0 ? (
   301→   301→              <div className="flex items-center justify-center h-[250px] text-muted-foreground text-sm">
   302→   302→                No paid invoices to display
   303→   303→              </div>
   304→   304→            ) : (
   305→   305→              <ResponsiveContainer width="100%" height={250}>
   306→   306→                <AreaChart data={revenueByMonth}>
   307→   307→                  <defs>
   308→   308→                    <linearGradient id="revenueGrad" x1="0" y1="0" x2="0" y2="1">
   309→   309→                      <stop offset="5%" stopColor="#10b981" stopOpacity={0.3} />
   310→   310→                      <stop offset="95%" stopColor="#10b981" stopOpacity={0} />
   311→   311→                    </linearGradient>
   312→   312→                  </defs>
   313→   313→                  <CartesianGrid strokeDasharray="3 3" className="stroke-muted" />
   314→   314→                  <XAxis dataKey="month" tick={{ fontSize: 12 }} className="text-muted-foreground" />
   315→   315→                  <YAxis tick={{ fontSize: 12 }} className="text-muted-foreground" tickFormatter={(v) => `$${(v / 1000).toFixed(0)}k`} />
   316→   316→                  <Tooltip formatter={(value: number) => [formatCurrency(value), 'Revenue']} />
   317→   317→                  <Area type="monotone" dataKey="revenue" stroke="#10b981" fill="url(#revenueGrad)" strokeWidth={2} />
   318→   318→                </AreaChart>
   319→   319→              </ResponsiveContainer>
   320→   320→            )}
   321→   321→          </CardContent>
   322→   322→        </Card>
   323→   323→
   324→   324→        {/* Expense breakdown */}
   325→   325→        <Card>
   326→   326→          <CardHeader className="pb-2">
   327→   327→            <CardTitle className="text-base font-semibold">Expense Breakdown</CardTitle>
   328→   328→          </CardHeader>
   329→   329→          <CardContent>
   330→   330→            {expenseByCategory.length === 0 ? (
   331→   331→              <div className="flex items-center justify-center h-[250px] text-muted-foreground text-sm">
   332→   332→                No expenses to display
   333→   333→              </div>
   334→   334→            ) : (
   335→   335→              <div className="flex items-center gap-6">
   336→   336→                <ResponsiveContainer width="50%" height={250}>
   337→   337→                  <PieChart>
   338→   338→                    <Pie
   339→   339→                      data={expenseByCategory}
   340→   340→                      cx="50%"
   341→   341→                      cy="50%"
   342→   342→                      innerRadius={50}
   343→   343→                      outerRadius={90}
   344→   344→                      paddingAngle={3}
   345→   345→                      dataKey="value"
   346→   346→                    >
   347→   347→                      {expenseByCategory.map((entry) => (
   348→   348→                        <Cell key={entry.name} fill={PIE_COLORS[entry.name] || '#94a3b8'} />
   349→   349→                      ))}
   350→   350→                    </Pie>
   351→   351→                    <Tooltip formatter={(value: number) => [formatCurrency(value), 'Amount']} />
   352→   352→                  </PieChart>
   353→   353→                </ResponsiveContainer>
   354→   354→                <div className="flex flex-col gap-2.5 flex-1">
   355→   355→                  {expenseByCategory.map((item) => (
   356→   356→                    <div key={item.name} className="flex items-center gap-2 text-sm">
   357→   357→                      <div className="size-3 rounded-full shrink-0" style={{ backgroundColor: PIE_COLORS[item.name] || '#94a3b8' }} />
   358→   358→                      <span className="text-muted-foreground capitalize flex-1">{item.name}</span>
   359→   359→                      <span className="font-medium">{formatCurrency(item.value)}</span>
   360→   360→                    </div>
   361→   361→                  ))}
   362→   362→                </div>
   363→   363→              </div>
   364→   364→            )}
   365→   365→          </CardContent>
   366→   366→        </Card>
   367→   367→      </div>
   368→   368→
   369→   369→      {/* Recent Invoices */}
   370→   370→      <Card>
   371→   371→        <CardHeader className="pb-2 flex flex-row items-center justify-between">
   372→   372→          <CardTitle className="text-base font-semibold">Recent Invoices</CardTitle>
   373→   373→          <Button variant="ghost" size="sm" onClick={() => onSwitchTab('invoices')} className="text-emerald-600">
   374→   374→            View all <ArrowUpRight className="size-3.5 ml-1" />
   375→   375→          </Button>
   376→   376→        </CardHeader>
   377→   377→        <CardContent>
   378→   378→          {recentInvoices.length === 0 ? (
   379→   379→            <p className="text-muted-foreground text-sm py-8 text-center">No invoices yet</p>
   380→   380→          ) : (
   381→   381→            <div className="rounded-lg border overflow-hidden">
   382→   382→              <table className="w-full text-sm">
   383→   383→                <thead>
   384→   384→                  <tr className="border-b bg-muted/50">
   385→   385→                    <th className="text-left font-medium px-4 py-2.5">Invoice #</th>
   386→   386→                    <th className="text-left font-medium px-4 py-2.5 hidden sm:table-cell">Contact</th>
   387→   387→                    <th className="text-left font-medium px-4 py-2.5 hidden md:table-cell">Company</th>
   388→   388→                    <th className="text-right font-medium px-4 py-2.5">Amount</th>
   389→   389→                    <th className="text-left font-medium px-4 py-2.5">Status</th>
   390→   390→                    <th className="text-left font-medium px-4 py-2.5 hidden lg:table-cell">Date</th>
   391→   391→                  </tr>
   392→   392→                </thead>
   393→   393→                <tbody>
   394→   394→                  {recentInvoices.map((inv) => (
   395→   395→                    <tr key={inv.id} className="border-b last:border-0">
   396→   396→                      <td className="px-4 py-2.5 font-medium">{inv.invoiceNumber}</td>
   397→   397→                      <td className="px-4 py-2.5 hidden sm:table-cell">{inv.contactName || '—'}</td>
   398→   398→                      <td className="px-4 py-2.5 hidden md:table-cell">{inv.companyName || '—'}</td>
