   181→   181→function BoardTab({ tasks }: { tasks: ProjectTaskItem[] }) {
   182→   182→  const [search, setSearch] = useState('');
   183→   183→  const filtered = useMemo(() => {
   184→   184→    if (!search) return tasks;
   185→   185→    const q = search.toLowerCase();
   186→   186→    return tasks.filter(
   187→   187→      (t) =>
   188→   188→        t.title.toLowerCase().includes(q) ||
   189→   189→        t.projectName.toLowerCase().includes(q) ||
   190→   190→        t.assigneeName.toLowerCase().includes(q)
   191→   191→    );
   192→   192→  }, [tasks, search]);

   201→   201→
   202→   202→  return (

   204→   204→      <TabHeader title="Task Board" description="Drag tasks across columns to update their status" actionLabel="New Task" />

   237→   237→function ListViewTab({ tasks }: { tasks: ProjectTaskItem[] }) {
   238→   238→  const [search, setSearch] = useState('');
   239→   239→  const [sortKey, setSortKey] = useState<'priority' | 'dueDate' | 'title'>('priority');
   240→   240→  const priorityOrder: Record<string, number> = { critical: 0, high: 1, medium: 2, low: 3 };
   241→   241→
   242→   242→  const filtered = useMemo(() => {
   243→   243→    let result = tasks;
   244→   244→    if (search) {
   245→   245→      const q = search.toLowerCase();
   246→   246→      result = result.filter(
   247→   247→        (t) =>
   248→   248→          t.title.toLowerCase().includes(q) ||
   249→   249→          t.projectName.toLowerCase().includes(q) ||
   250→   250→          t.assigneeName.toLowerCase().includes(q)

   254→   254→      if (sortKey === 'priority') return (priorityOrder[a.priority] ?? 9) - (priorityOrder[b.priority] ?? 9);

   427→   427→/* ---- Tab 4: Timeline ---- */
   428→   428→
   429→   429→function TimelineTab({ items }: { items: ProjectItem[] }) {
   430→   430→  const allDates = useMemo(() => {
   431→   431→    const starts = items.map((p) => new Date(p.startDate).getTime());
   432→   432→    const ends = items.map((p) => new Date(p.endDate).getTime());
   433→   433→    return { min: Math.min(...starts), max: Math.max(...ends) };
   434→   434→  }, [items]);
   435→   435→
   436→   436→  const totalSpan = allDates.max - allDates.min;
   437→   437→
   438→   438→  function getLeft(startDate: string) {
   439→   439→    const ms = new Date(startDate).getTime() - allDates.min;
   440→   440→    return totalSpan > 0 ? (ms / totalSpan) * 100 : 0;
   441→   441→  }
   442→   442→
   443→   443→  function getWidth(startDate: string, endDate: string) {
   444→   444→    const ms = new Date(endDate).getTime() - new Date(startDate).getTime();
   445→   445→    return totalSpan > 0 ? Math.max((ms / totalSpan) * 100, 3) : 3;
   446→   446→  }
   447→   447→
   448→   448→  const months = useMemo(() => {
   449→   449→    const start = new Date(allDates.min);
   450→   450→    const end = new Date(allDates.max);
   451→   451→    const result: string[] = [];
   452→   452→    const current = new Date(start.getFullYear(), start.getMonth(), 1);
   453→   453→    while (current <= end) {
   454→   454→      result.push(current.toLocaleDateString('en-US', { month: 'short', year: '2-digit' }));
   455→   455→      current.setMonth(current.getMonth() + 1);
   456→   456→    }
   457→   457→    return result;
   458→   458→  }, [allDates]);
   459→   459→
   460→   460→  return (
   461→   461→    <div>
   462→   462→      <TabHeader title="Timeline" description="Visualize project schedules and progress on a Gantt-like timeline" actionLabel="Add Project" />
   463→   463→      <div className="bg-card rounded-xl border border-border overflow-hidden">
   464→   464→        {/* Month header */}
   465→   465→        <div className="flex border-b border-border">
   466→   466→          <div className="w-52 shrink-0 px-4 py-3 border-r border-border">
   467→   467→            <span className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Project</span>
   468→   468→          </div>
   469→   469→          <div className="flex-1 flex relative">
   470→   470→            {months.map((m, i) => {
   471→   471→              const monthStart = new Date(allDates.min);
   472→   472→              monthStart.setMonth(monthStart.getMonth() + i);
   473→   473→              const monthStartMs = monthStart.getTime() - allDates.min;
   474→   474→              const nextMonth = new Date(monthStart);
   475→   475→              nextMonth.setMonth(nextMonth.getMonth() + 1);
   476→   476→              const nextMonthMs = Math.min(nextMonth.getTime() - allDates.min, totalSpan);
   477→   477→              const widthPct = totalSpan > 0 ? ((nextMonthMs - monthStartMs) / totalSpan) * 100 : 0;
   478→   478→              return (
   479→   479→                <div
   480→   480→                  key={m}
   481→   481→                  className="flex items-center justify-center text-[11px] text-muted-foreground font-medium border-r border-border last:border-r-0"
   482→   482→                  style={{ width: `${widthPct}%` }}
   483→   483→                >
   484→   484→                  {m}
   485→   485→                </div>
   486→   486→              );
   487→   487→            })}
   488→   488→          </div>
   489→   489→        </div>
   490→   490→        {/* Rows */}
   491→   491→        {items.map((project, i) => (
   492→   492→          <div
   493→   493→            key={project.id}
   494→   494→            className={cn('flex items-center border-b border-border last:border-b-0', i % 2 === 0 ? 'bg-background' : 'bg-muted/20')}
   495→   495→          >
   496→   496→            <div className="w-52 shrink-0 px-4 py-3.5 border-r border-border">
   497→   497→              <div className="flex items-center gap-2">
   498→   498→                <ProjectStatusBadge status={project.status} />
   499→   499→                <span className="text-sm font-medium text-foreground truncate">{project.name}</span>
   500→   500→              </div>
   501→   501→            </div>
   502→   502→            <div className="flex-1 py-3.5 px-2 relative" style={{ minHeight: '44px' }}>
   503→   503→              <div
   504→   504→                className="relative h-7 rounded-md overflow-hidden"
   505→   505→                style={{
   506→   506→                  position: 'absolute',
   507→   507→                  left: `${getLeft(project.startDate)}%`,
   508→   508→                  width: `${getWidth(project.startDate, project.endDate)}%`,
   509→   509→                  top: '50%',
   510→   510→                  transform: 'translateY(-50%)',
   511→   511→                }}
   512→   512→              >
   513→   513→                {/* Background bar */}
   514→   514→                <div className={cn('absolute inset-0 rounded-md opacity-20', projectBarColors[project.status] ?? 'bg-gray-400')} />
   515→   515→                {/* Progress fill */}
   516→   516→                <div
   517→   517→                  className={cn('absolute inset-y-0 left-0 rounded-md', projectBarColors[project.status] ?? 'bg-gray-400')}
   518→   518→                  style={{ width: `${project.progress}%` }}
   519→   519→                />
   520→   520→                {/* Label */}
   521→   521→                <span className="absolute inset-0 flex items-center px-2 text-[10px] font-semibold text-foreground truncate">
   522→   522→                  {project.progress}% — {formatDate(project.startDate).replace(/,\s*\d{4}$/, '')} to {formatDate(project.endDate).replace(/,\s*\d{4}$/, '')}
   523→   523→                </span>
   524→   524→              </div>
   525→   525→            </div>
   526→   526→          </div>
   527→   527→        ))}
   528→   528→      </div>
   529→   529→    </div>
   530→   530→  );
   531→   531→}
