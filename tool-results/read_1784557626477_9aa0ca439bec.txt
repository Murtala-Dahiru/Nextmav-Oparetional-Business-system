   188→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ name: string; value: number; color: string }>; label?: string }) {
   189→  if (!active || !payload?.length) return null;
   190→  return (
   191→    <div className="rounded-lg border border-gray-200 bg-white/95 px-3 py-2 shadow-lg dark:border-gray-700 dark:bg-gray-900/95">
   192→      <p className="mb-1 text-sm font-semibold text-gray-900 dark:text-gray-100">{label}</p>
   193→      {payload.map((entry, i) => (
   194→        <p key={i} className="text-xs text-gray-600 dark:text-gray-400">
   195→          <span className="mr-1.5 inline-block h-2 w-2 rounded-full" style={{ backgroundColor: entry.color }} />
   196→          {entry.name}: <span className="font-medium text-gray-900 dark:text-gray-100">${(entry.value / 1000).toFixed(1)}K</span>
   197→        </p>
   198→      ))}
   199→    </div>
   200→  );
   201→}

   283→function LeadConversionFunnel() {
   284→  const maxCount = funnelStages[0].count;
   285→  return (
   286→    <Card className="border-0 shadow-sm">
   287→      <CardHeader className="pb-2">
   288→        <CardTitle className="flex items-center gap-2 text-base font-semibold text-gray-900 dark:text-gray-100">
   289→          <Filter className="h-4 w-4 text-emerald-600" />
   290→          Lead Conversion Funnel
   291→        </CardTitle>
   292→        <CardDescription>Visual representation of your pipeline from lead to close</CardDescription>
   293→      </CardHeader>
   294→      <CardContent className="pt-2">
   295→        <div className="relative flex flex-col items-center gap-0">
   296→          {funnelStages.map((stage, idx) => {
   297→            const widthPercent = 30 + ((stage.count / maxCount) * 70);
   298→            const conversionFromPrev = idx > 0 ? ((stage.count / funnelStages[idx - 1].count) * 100).toFixed(1) : '100';
   299→            const isLast = idx === funnelStages.length - 1;
   300→            return (
   301→              <div key={stage.label} className="relative flex w-full flex-col items-center">
   302→                {/* Connector arrow */}
   303→                {idx > 0 && (
   304→                  <motion.div
   305→                    initial={{ scaleY: 0 }}
   306→                    animate={{ scaleY: 1 }}
   307→                    transition={{ duration: 0.3, delay: idx * 0.1 }}
   308→                    className="flex items-center gap-2 py-1"
   309→                  >
   310→                    <div className="h-4 w-px bg-gradient-to-b from-gray-300 to-gray-400 dark:from-gray-600 dark:to-gray-500" />
   311→                    <ChevronRight className="h-3 w-3 -rotate-90 text-gray-400" />
   312→                    <span className="text-[10px] font-bold text-emerald-600 dark:text-emerald-400">
   313→                      {conversionFromPrev}%
   314→                    </span>
   315→                  </motion.div>
   316→                )}
   317→                {/* Funnel bar */}
   318→                <motion.div
   319→                  initial={{ scaleX: 0, opacity: 0 }}
   320→                  animate={{ scaleX: 1, opacity: 1 }}
   321→                  transition={{ duration: 0.5, delay: idx * 0.1, ease: 'easeOut' }}
   322→                  style={{ width: `${widthPercent}%` }}
   323→                  className="relative"
   324→                >
   325→                  <div
   326→                    className={cn(
   327→                      'group relative flex items-center justify-between rounded-sm px-5 py-3 transition-all duration-300',
   328→                      isLast
   329→                        ? 'bg-gradient-to-r from-emerald-500 to-teal-400 shadow-lg shadow-emerald-500/25'
   330→                        : 'hover:brightness-110',
   331→                    )}
   332→                    style={
   333→                      isLast
   334→                        ? undefined
   335→                        : {
   336→                            background: `linear-gradient(135deg, ${stage.color}, ${stage.color}dd)`,
   337→                            boxShadow: `0 2px 8px ${stage.color}30`,
   338→                          }
   339→                    }
   340→                  >
   341→                    {/* Animated shimmer effect */}
   342→                    <div className="absolute inset-0 overflow-hidden rounded-sm">
   343→                      <div className="absolute inset-0 -translate-x-full bg-gradient-to-r from-transparent via-white/10 to-transparent animate-[shimmer_3s_infinite]" />
   344→                    </div>
   345→                    {/* Stage label and count */}
   346→                    <div className="relative z-10 flex items-center gap-3">
   347→                      <div className={cn(
   348→                        'flex h-7 w-7 items-center justify-center rounded-full text-xs font-bold',
   349→                        isLast ? 'bg-white/25 text-white' : 'bg-white/15 text-white/90',
   350→                      )}>
   351→                        {idx + 1}
   352→                      </div>
   353→                      <div>
   354→                        <span className={cn(
   355→                          'block text-sm font-semibold tracking-wide',
   356→                          isLast ? 'text-white' : 'text-white/95',
   357→                        )}>
   358→                          {stage.label}
   359→                        </span>
   360→                      </div>
   361→                    </div>
   362→                    <div className="relative z-10 text-right">
   363→                      <span className={cn(
   364→                        'block text-xl font-bold tabular-nums',
   365→                        isLast ? 'text-white' : 'text-white/95',
   366→                      )}>
   367→                        {stage.count.toLocaleString()}
   368→                      </span>
   369→                    </div>
   370→                    {/* Drop percentage */}
   371→                    {idx > 0 && (
   372→                      <div className="absolute -right-1 top-1/2 -translate-y-1/2 translate-x-full px-2 py-0.5">
   373→                        <Badge variant="secondary" className="text-[10px] font-bold text-emerald-700 bg-emerald-50 dark:text-emerald-400 dark:bg-emerald-950/50 border-0">
   374→                          ↓ {((1 - stage.count / funnelStages[idx - 1].count) * 100).toFixed(1)}%
   375→                        </Badge>
   376→                      </div>
   377→                    )}
   378→                  </div>
   379→                </motion.div>
   380→              </div>
   381→            );
   382→          })}
   383→          {/* Overall conversion rate callout */}
   384→          <motion.div
   385→            initial={{ opacity: 0, scale: 0.9 }}
   386→            animate={{ opacity: 1, scale: 1 }}
   387→            transition={{ duration: 0.5, delay: 0.8 }}
   388→            className="mt-4 rounded-xl border border-emerald-200 bg-gradient-to-r from-emerald-50 to-teal-50 px-6 py-3 text-center dark:border-emerald-800 dark:from-emerald-950/40 dark:to-teal-950/40"
   389→          >
   390→            <p className="text-xs font-medium uppercase tracking-wider text-emerald-600 dark:text-emerald-400">Overall Conversion Rate</p>

   392→              {((funnelStages[funnelStages.length - 1].count / funnelStages[0].count) * 100).toFixed(1)}%
   393→            </p>
   394→            <p className="text-xs text-emerald-600/70 dark:text-emerald-400/60">From Lead to Won</p>
   395→          </motion.div>
   396→        </div>
   397→      </CardContent>
   398→    </Card>
