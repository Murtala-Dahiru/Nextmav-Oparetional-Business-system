   241→function ChartTooltipContent({
   242→  active,
   243→  payload,
   244→  label,
   245→  formatter,
   246→}: {
   247→  active?: boolean;
   248→  payload?: Array<{ value: number; name: string; dataKey: string; color?: string }>;
   249→  label?: string;
   250→  formatter?: (value: number) => string;
   251→}) {
   252→  if (!active || !payload?.length) return null;
   253→  return (
   254→    <div className="border-border/50 bg-background rounded-lg border px-3 py-2 text-xs shadow-xl">
   255→      {label && <p className="font-medium text-foreground mb-1">{label}</p>}
   256→      {payload.map((item, idx) => (
   257→        <div key={idx} className="flex items-center gap-2">
   258→          <span
   259→            className="inline-block h-2 w-2 rounded-full"
   260→            style={{ backgroundColor: item.color ?? CHART_COLORS[0] }}
   261→          />
   262→          <span className="text-muted-foreground">
   263→            {item.name}: {formatter ? formatter(item.value) : item.value.toLocaleString()}
   264→          </span>
   265→        </div>
   266→      ))}
   267→    </div>
   268→  );
   269→}

   275→export default function DashboardModule() {
   276→  const { setActiveModule } = useAppStore();
   277→  const [loading, setLoading] = useState(true);
   278→  const [data, setData] = useState<DashboardData | null>(null);
   279→  const [error, setError] = useState<string | null>(null);
   280→
   281→  // Fetch dashboard data on mount
   282→  useEffect(() => {
   283→    let cancelled = false;
   284→
   285→    async function fetchDashboard() {
   286→      try {
   287→        setLoading(true);
   288→        const res = await fetch('/api/dashboard');
   289→        if (!res.ok) throw new Error(`Failed to fetch dashboard: ${res.status}`);
   290→        const json = await res.json();
   291→        if (cancelled) return;
   292→        setData(json.data);
   293→      } catch (e: unknown) {
   294→        if (!cancelled) {
   295→          setError(e instanceof Error ? e.message : 'Failed to load dashboard data');
   296→        }
   297→      } finally {
   298→        if (!cancelled) setLoading(false);
   299→      }
   300→    }
   301→
   302→    fetchDashboard();
   303→    return () => {
   304→      cancelled = true;
   305→    };
   306→  }, []);
   307→
   308→  // ---- Computed KPIs ----
   309→
   310→  const kpis = useMemo(() => {
   311→    if (!data) return null;
   312→    const s = data.stats;
   313→    return [
   314→      { label: 'Active Leads', value: s.activeLeads, change: 12.5, changeLabel: 'vs last month', icon: UserPlus },
   315→      { label: 'Total Revenue', value: formatCurrency(s.revenue), change: 8.3, changeLabel: 'vs last month', icon: DollarSign },
   316→      { label: 'Open Tickets', value: s.openTickets, change: -2.1, changeLabel: 'vs last week', icon: TicketCheck },
   317→      { label: 'Active Projects', value: s.activeProjects, change: 5.7, changeLabel: 'vs last month', icon: FolderKanban },
   318→      { label: 'Team Members', value: s.totalEmployees, icon: User },
   319→      { label: 'Pending Invoices', value: s.pendingInvoices, icon: FileText },
   320→    ];
   321→  }, [data]);
   322→
   323→  // Active Deals: not closed-won/closed-lost
   324→  const activeDealsCount = data.deals.filter(
   325→    (d) => d.stage !== 'closed-won' && d.stage !== 'closed-lost',
   326→  ).length;
   327→
   328→    // Deals change: this month vs last month created
   329→    const thisMonthDeals = data.deals.filter(
   330→      (d) => new Date(d.createdAt) >= thisMonthStart,
   331→    ).length;
   332→    const lastMonthDeals = data.deals.filter(
   333→      (d) => new Date(d.createdAt) >= lastMonthStart && new Date(d.createdAt) < thisMonthStart,
   334→    ).length;
   335→    const dealsChange =
   336→      lastMonthDeals > 0
   337→        ? ((thisMonthDeals - lastMonthDeals) / lastMonthDeals) * 100
   338→        : thisMonthDeals > 0
   339→          ? 100
   340→          : 0;
   341→
   342→    // Open Tickets: open / in-progress / pending
   343→    const openTickets = data.tickets.filter(
   344→      (t) => t.status === 'open' || t.status === 'in-progress' || t.status === 'pending',
   345→    );
   346→    const openTicketsCount = openTickets.length;
   347→
   348→    // Tickets change: this month vs last month
   349→    const thisMonthTickets = data.tickets.filter(
   350→      (t) => new Date(t.createdAt) >= thisMonthStart,
   351→    ).length;
   352→    const lastMonthTickets = data.tickets.filter(
   353→      (t) => new Date(t.createdAt) >= lastMonthStart && new Date(t.createdAt) < thisMonthStart,
   354→    ).length;
   355→    const ticketsChange =
   356→      lastMonthTickets > 0
   357→        ? ((thisMonthTickets - lastMonthTickets) / lastMonthTickets) * 100
   358→        : thisMonthTickets > 0
   359→          ? 100
   360→          : 0;
   361→
   362→    // Active Projects
   363→    const activeProjects = data.projects.filter((p) => p.status === 'active');
   364→    const activeProjectsCount = activeProjects.length;
   365→
   366→    const thisMonthProjects = data.projects.filter(
   367→      (p) => new Date(p.createdAt) >= thisMonthStart,
   368→    ).length;
   369→    const lastMonthProjects = data.projects.filter(
   370→      (p) => new Date(p.createdAt) >= lastMonthStart && new Date(p.createdAt) < thisMonthStart,
   371→    ).length;
   372→    const projectsChange =
   373→      lastMonthProjects > 0
   374→        ? ((thisMonthProjects - lastMonthProjects) / lastMonthProjects) * 100
   375→        : thisMonthProjects > 0
   376→          ? 100
   377→          : 0;
   378→
   379→    // Tasks Due This Week
   380→    const weekFromNow = new Date();
   381→    weekFromNow.setDate(weekFromNow.getDate() + 7);
   382→    weekFromNow.setHours(23, 59, 59, 999);
   383→    const todayStart = new Date();
   384→    todayStart.setHours(0, 0, 0, 0);
   385→    const tasksDueThisWeek = data.tasks.filter(
   386→      (t) =>
   387→        t.dueDate &&
   388→        t.status !== 'done' &&
   389→        new Date(t.dueDate) >= todayStart &&
   390→        new Date(t.dueDate) <= weekFromNow,
   391→    ).length;
   392→
   393→    // New Leads This Month
   394→    const thirtyDaysAgo = new Date();
   395→    thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);
   396→    const newLeadsThisMonth = data.leads.filter(
   397→      (l) => new Date(l.createdAt) >= thirtyDaysAgo,
   398→    ).length;
   399→
   400→    const thisMonthLeads = data.leads.filter(
   401→      (l) => new Date(l.createdAt) >= thisMonthStart,
   402→    ).length;
   403→    const lastMonthLeads = data.leads.filter(
   404→      (l) => new Date(l.createdAt) >= lastMonthStart && new Date(l.createdAt) < thisMonthStart,
   405→    ).length;
   406→    const leadsChange =
   407→      lastMonthLeads > 0
   408→        ? ((thisMonthLeads - lastMonthLeads) / lastMonthLeads) * 100
   409→        : thisMonthLeads > 0
   410→          ? 100
   411→          : 0;
   412→
   413→    return {
   414→      activeDealsCount,
   415→      openTicketsCount,
   416→      activeProjectsCount,
   417→      tasksDueThisWeek,
   418→      newLeadsThisMonth,
   419→    };
   420→  }, [data]);
   421→
   422→  // ---- Chart: Monthly Revenue ----
   423→
   424→  const revenueChartData = useMemo(() => {
   425→    if (!data) return [];
   426→
   427→    const paidInvoices = data.invoices.filter((inv) => inv.status === 'paid' && inv.paidAt);
   428→
   429→    // Group by month (last 6 months)
   430→    const monthMap = new Map<string, number>();
   431→    const now = new Date();
   432→
   433→    for (let i = 5; i >= 0; i--) {
   434→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   435→      const key = d.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   436→      monthMap.set(key, 0);
   437→    }
   438→
   439→    paidInvoices.forEach((inv) => {
   440→      const d = new Date(inv.paidAt!);
   441→      const key = d.toLocaleDateString('en-US', { month: 'short', year: '2-digit' });
   442→      if (monthMap.has(key)) {
   443→        monthMap.set(key, monthMap.get(key)! + inv.total);
   444→      }
   445→    });
   446→
   447→    return Array.from(monthMap.entries()).map(([month, revenue]) => ({ month, revenue }));
   448→  }, [data]);
   449→
   450→  // ---- Chart: Deal Pipeline ----
   451→
   452→  const pipelineChartData = useMemo(() => {
   453→    if (!data) return [];
   454→
   455→    const stageMap = new Map<string, { count: number; value: number }>();
   456→
   457→    DEAL_STAGE_ORDER.forEach((stage) => {
   458→      stageMap.set(stage, { count: 0, value: 0 });
   459→    });
   460→
   461→    data.deals.forEach((deal) => {
   462→      const entry = stageMap.get(deal.stage);
   463→      if (entry) {
   464→        entry.count += 1;
   465→        entry.value += deal.value;
   466→      }
   467→    });
   468→
   469→    return Array.from(stageMap.entries())
   470→      .filter(([, v]) => v.count > 0)
   471→      .map(([stage, v]) => ({
   472→        stage: stage.replace(/-/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase()),
   473→        count: v.count,
   474→        value: v.value,
   475→      }));
   476→  }, [data]);
   477→
   478→  // ---- Chart: Lead Status Distribution ----
   479→
   480→  const leadChartData = useMemo(() => {
   481→    if (!data?.leadByStatus) return [];
   482→    const statusOrder = [new, contacted, qualified, proposal, negotiation, won, lost];
   483→    return statusOrder
   484→      .map((status) => {
   485→        const found = data.leadByStatus.find((s: any) => s.status === status);
   486→        return { status, count: found?._count ?? 0 };
   487→      });
   488→  }, [data]);
   489→
   490→    return Array.from(statusMap.entries())
   491→      .map(([status, count]) => ({
   492→        name: status.replace(/-/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase()),
   493→        value: count,
   494→        color: LEAD_STATUS_COLORS[status] ?? '#94a3b8',
   495→      }))
   496→      .sort((a, b) => b.value - a.value);
   497→  }, [data]);
   498→
   499→  // ---- Top Deals ----
   500→
   501→  const topDeals = useMemo(() => {
   502→    if (!data) return [];
   503→    return [...data.deals]
   504→      .sort((a, b) => b.value - a.value)
   505→      .slice(0, 5);
   506→  }, [data]);
   507→
   508→  // ---- Quick Action Handler ----
   509→
   510→  const handleQuickAction = useCallback(
   511→    (module: ModuleId) => {
   512→      setActiveModule(module);
   513→    },
   514→    [setActiveModule],
   515→  );
   516→
   517→  // ---- Module navigation from activity ----
   518→
   519→  const handleActivityClick = useCallback(
   520→    (moduleName: string) => {
   521→      const moduleMap: Record<string, ModuleId> = {
   522→        crm: 'crm',
   523→        projects: 'projects',
   524→        support: 'support',
   525→        finance: 'finance',
   526→        hr: 'hr',
   527→        inventory: 'inventory',
   528→        calendar: 'calendar',
   529→        communication: 'communication',
   530→        admin: 'admin',
   531→        workspace: 'workspace',
   532→      };
   533→      if (moduleMap[moduleName]) {
   534→        setActiveModule(moduleMap[moduleName]);
   535→      }
   536→    },
   537→    [setActiveModule],
   538→  );
   539→
   540→  // ---- Error State ----
   541→
   542→  if (error && !data) {
   543→    return (
   544→      <div className="space-y-6">
   545→        <PageHeader
   546→          title="Dashboard"
   547→          description="Overview of your business operations"
   548→          icon={TrendingUp}
   549→        />
   550→        <EmptyState
   551→          icon={Activity}
   552→          title="Unable to load dashboard"
   553→          description={error}
   554→        />
   555→      </div>
   556→    );
   557→  }
   558→
   559→  // ---- Render ----
   560→
   561→  return (
   562→    <div className="space-y-6">
   563→      {/* Page Header */}
   564→      <PageHeader
   565→        title="Dashboard"
   566→        description="Overview of your business operations"
   567→        icon={TrendingUp}
   568→      />
   569→
   570→      {/* Section 1: KPI Cards */}
   571→      {loading ? (
   572→        <KpiSkeletons />
   573→      ) : (
   574→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-6 gap-4">
   575→          <StatCard
   576→            label="Total Revenue"
   577→            value={formatCurrency(kpis!.totalRevenue)}
   578→            change={kpis!.revenueChange}
   579→            changeLabel="vs last month"
   580→            icon={DollarSign}
   581→          />
   582→          <StatCard
   583→            label="Active Deals"
   584→            value={kpis!.activeDealsCount}
   585→            change={kpis!.dealsChange}
   586→            changeLabel="vs last month"
   587→            icon={Handshake}
   588→          />
   589→          <StatCard
   590→            label="Open Tickets"
   591→            value={kpis!.openTicketsCount}
   592→            change={kpis!.ticketsChange}
   593→            changeLabel="vs last month"
   594→            icon={TicketCheck}
   595→          />
   596→          <StatCard
   597→            label="Active Projects"
   598→            value={kpis!.activeProjectsCount}
   599→            change={kpis!.projectsChange}
   600→            changeLabel="vs last month"
   601→            icon={FolderKanban}
   602→          />
   603→          <StatCard
   604→            label="Tasks Due This Week"
   605→            value={kpis!.tasksDueThisWeek}
   606→            icon={ListTodo}
   607→          />
   608→          <StatCard
   609→            label="New Leads"
   610→            value={kpis!.newLeadsThisMonth}
   611→            change={kpis!.leadsChange}
   612→            changeLabel="vs last month"
   613→            icon={UserPlus}
   614→          />
   615→        </div>
   616→      )}
   617→
   618→      {/* Section 2: Revenue Chart */}
   619→      {loading ? (
   620→        <ChartSkeleton />
   621→      ) : revenueChartData.length === 0 ? (
   622→        <Card>
   623→          <CardHeader>
   624→            <CardTitle>Revenue Overview</CardTitle>
   625→          </CardHeader>
   626→          <CardContent>
   627→            <EmptyState
   628→              icon={DollarSign}
   629→              title="No revenue data yet"
   630→              description="Paid invoices will appear as monthly revenue here."
   631→            />
   632→          </CardContent>
   633→        </Card>
   634→      ) : (
   635→        <Card>
   636→          <CardHeader>
   637→            <CardTitle>Revenue Overview</CardTitle>
   638→          </CardHeader>
   639→          <CardContent>
