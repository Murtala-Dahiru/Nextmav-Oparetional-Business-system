   105→async function apiFetch<T>(url: string, init?: RequestInit): Promise<T> {
   106→  const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...init });
   107→  const json = await res.json();
   108→  if (!res.ok) throw new Error(json.error?.message ?? 'Request failed');
   109→  return json;
   110→}
